// Instancia aislada de Express para las medidas de navegador (fase 0). Es el
// startInstance() de la auditoría (e1/harness.mjs) sin la dependencia de Playwright:
// sirve dist/ y la API en el mismo origen, en modo demo, con las variables de
// .env.example y sin leer ningún .env (el repo de trabajo es un `git archive`).
import { spawn } from 'node:child_process';
import fs from 'node:fs';
import path from 'node:path';

export const REPO = process.env.ROCKY_REPO;
const NODE = process.env.ROCKY_NODE || process.execPath;
if (!REPO) throw new Error('Falta ROCKY_REPO (carpeta del git archive con dist/ construido).');

function exampleEnv() {
  const out = {};
  for (const line of fs.readFileSync(path.join(REPO, '.env.example'), 'utf8').split('\n')) {
    const match = line.match(/^([A-Z][A-Z0-9_]*)=(.*)$/);
    if (match) out[match[1]] = match[2];
  }
  for (const key of ['RESEND_API_KEY', 'SHOPIFY_ADMIN_ACCESS_TOKEN', 'BEATS_ADMIN_TOKEN', 'AVISOS_FROM', 'AVISOS_REPLY_TO']) {
    out[key] = '';
  }
  return out;
}

export async function startInstance(port, env = {}) {
  const origin = `http://localhost:${port}`;
  const child = spawn(NODE, ['server/index.mjs'], {
    cwd: REPO,
    env: {
      PATH: process.env.PATH,
      HOME: process.env.HOME,
      ...exampleEnv(),
      PORT: String(port),
      PUBLIC_ORIGIN: origin,
      API_ALLOWED_ORIGINS: origin,
      ...env,
    },
    stdio: ['ignore', 'pipe', 'pipe'],
  });
  const log = [];
  child.stdout.on('data', (data) => log.push(String(data)));
  child.stderr.on('data', (data) => log.push(String(data)));
  for (let attempt = 0; attempt < 60; attempt += 1) {
    try {
      const response = await fetch(`${origin}/api/health`);
      if (response.ok) return { origin, child, log, stop: () => child.kill('SIGTERM') };
    } catch {
      /* todavía arrancando */
    }
    if (child.exitCode !== null) break;
    await new Promise((resolve) => setTimeout(resolve, 250));
  }
  child.kill('SIGTERM');
  throw new Error(`La instancia de ${origin} no arrancó:\n${log.join('').slice(-2000)}`);
}
