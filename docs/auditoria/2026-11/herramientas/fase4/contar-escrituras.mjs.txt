// Cuántas veces vuelca a disco el almacén cifrado cada tipo de petición (fase 0).
//
//   node contar-escrituras.mjs <carpeta-de-salida>
//
// Arranca el servidor real contra el doble local de Shopify y OpenRouter de la
// auditoría (e2sim) y cuenta los volcados con cuenta-escrituras-preload.mjs. El doble
// no simula las consultas Admin de los drops, así que la instancia va sin credenciales
// Admin (sin drops, sin Taller, sin pared). Después crea 1.000 sesiones para ver cómo
// crecen el fichero y el coste de cada volcado.
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath, pathToFileURL } from 'node:url';
import { encodePattern, emptyPattern } from '../repo/shared/beat-codec.mjs';
import { PRELOAD, REPO, startE2Instance } from './e2sim/start-e2sim.mjs';

const here = path.dirname(fileURLToPath(import.meta.url));
const outDir = process.argv[2];
if (!outDir) throw new Error('Uso: contar-escrituras.mjs <salida>');
const writeLog = path.join(outDir, 'escrituras.jsonl');
fs.writeFileSync(writeLog, '');

const inst = await startE2Instance(3411, {
  SHOPIFY_CLIENT_ID: '',
  SHOPIFY_CLIENT_SECRET: '',
  SHOPIFY_CUSTOMER_ACCOUNT_CLIENT_ID: '',
  SHOPIFY_RATE_LIMIT_MAX: '2000',
  CHAT_RATE_LIMIT_MAX: '50',
  NODE_OPTIONS: `--import ${PRELOAD} --import ${path.join(here, 'cuenta-escrituras-preload.mjs')}`,
  ROCKY_STORE_MODULE: pathToFileURL(path.join(REPO, 'server/storage/encrypted-file-store.mjs')).href,
  ROCKY_WRITE_LOG: writeLog,
});

const writes = () =>
  fs
    .readFileSync(writeLog, 'utf8')
    .split('\n')
    .filter(Boolean)
    .map((line) => JSON.parse(line));

// Las escrituras diferidas llegan al disco hasta 5 s después: para contar los volcados
// de una petición hay que esperar a que pase la ventana (ROCKY_FLUSH_WAIT_MS, 5.300 ms).
const flushWaitMs = Number(process.env.ROCKY_FLUSH_WAIT_MS ?? 5300);
const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));
let cookie = '';
async function call(label, method, urlPath, { body, useCookie = true, keepCookie = true, settle = true } = {}) {
  const before = writes().length;
  const headers = {};
  if (method !== 'GET') headers.Origin = inst.origin;
  if (body) headers['Content-Type'] = 'application/json';
  if (useCookie && cookie) headers.Cookie = cookie;
  const startedAt = performance.now();
  const response = await fetch(`${inst.origin}${urlPath}`, {
    method,
    headers,
    body: body ? JSON.stringify(body) : undefined,
  });
  const text = await response.text();
  const ms = Math.round((performance.now() - startedAt) * 10) / 10;
  const setCookie = response.headers.getSetCookie?.() ?? [];
  if (keepCookie && setCookie.length) cookie = setCookie.map((entry) => entry.split(';')[0]).join('; ');
  let json = null;
  try {
    json = JSON.parse(text);
  } catch {
    /* no es JSON */
  }
  if (settle && flushWaitMs > 0 && method !== 'GET') await wait(flushWaitMs);
  const after = writes();
  return {
    label,
    request: `${method} ${urlPath}`,
    status: response.status,
    ms,
    writes: after.length - before,
    stateBytes: after.at(-1)?.bytes ?? 0,
    json,
  };
}

const rows = [];
const run = async (...args) => {
  const row = await call(...args);
  rows.push(row);
  return row;
};
const operationId = () => `lb-${Math.random().toString(36).slice(2, 12)}${Date.now()}`;
const variant = (n, k) => `gid://shopify/ProductVariant/90350${String(n).padStart(2, '0')}${String(k).padStart(2, '0')}`;
const asset = fs.readdirSync(path.join(REPO, 'dist/assets')).find((file) => /^index-.*\.js$/.test(file));

try {
  await run('Documento de la SPA', 'GET', '/');
  await run('Estático con hash', 'GET', `/assets/${asset}`);
  await run('Sonda de salud', 'GET', '/api/health');
  await run('Estado de la tienda', 'GET', '/api/shopify/status');
  await run('Catálogo (primera lectura)', 'GET', '/api/shopify/products');
  await run('Catálogo (en caché)', 'GET', '/api/shopify/products');
  await run('Drops', 'GET', '/api/drops');
  await run('Carrito sin cookie', 'GET', '/api/shopify/cart');
  await run('Ranking de beats sin cookie', 'GET', '/api/beats');
  await run('Cuenta sin cookie', 'GET', '/api/shopify/account');

  const added = await run('Añadir al carrito (sesión y carrito nuevos)', 'POST', '/api/shopify/cart/lines', {
    body: { variantId: variant(1, 1), quantity: 1, operationId: operationId() },
  });
  await run('Leer el carrito con sesión', 'GET', '/api/shopify/cart');
  await run('Añadir otra línea (sesión existente)', 'POST', '/api/shopify/cart/lines', {
    body: { variantId: variant(2, 1), quantity: 1, operationId: operationId() },
  });
  const lineId = added.json?.cart?.lines?.[0]?.id ?? added.json?.lines?.[0]?.id ?? null;
  if (lineId) {
    await run('Cambiar cantidad', 'PATCH', '/api/shopify/cart/lines', {
      body: { lineId, quantity: 2, operationId: operationId() },
    });
    await run('Quitar línea', 'DELETE', '/api/shopify/cart/lines', {
      body: { lineId, operationId: operationId() },
    });
  }
  await run('Cuenta con sesión', 'GET', '/api/shopify/account');
  await run('Ranking de beats con sesión', 'GET', '/api/beats');

  const pattern = emptyPattern();
  pattern[0][0] = 3;
  pattern[1][4] = 2;
  const beat = await run('Publicar un beat', 'POST', '/api/beats', {
    body: { code: encodePattern(pattern), bpm: 92, swing: 0.25, tono: 're', name: 'Linea base', alias: 'medida' },
  });
  const beatId = beat.json?.beat?.id;
  if (beatId) await run('«Me mola» a un beat', 'POST', `/api/beats/${beatId}/like`);
  await run('Rocky IA: primer mensaje', 'POST', '/api/chat', { body: { message: 'hola, ¿qué camisetas hay?' } });
  await run('Rocky IA: segundo mensaje', 'POST', '/api/chat', { body: { message: '¿y en talla L?' } });
  await run('Apuntarse a un aviso', 'POST', '/api/avisos', {
    body: { producto: 'rocky-racing', email: 'linea-base@example.com', consentimiento: true },
  });

  // Crecimiento: 1.000 visitantes nuevos que añaden algo al carrito.
  const growth = [];
  await wait(flushWaitMs);
  const writesBeforeGrowth = writes().length;
  const growthStartedAt = performance.now();
  for (let index = 1; index <= 1000; index += 1) {
    const row = await call('nueva sesión', 'POST', '/api/shopify/cart/lines', {
      body: { variantId: variant(1, 1), quantity: 1, operationId: operationId() },
      useCookie: false,
      keepCookie: false,
      settle: false,
    });
    if (row.status !== 200) throw new Error(`Alta ${index}: ${row.status} ${JSON.stringify(row.json)}`);
    growth.push(row);
  }
  const growthSeconds = (performance.now() - growthStartedAt) / 1000;
  await wait(flushWaitMs);
  const all = writes();
  const growthWrites = all.slice(writesBeforeGrowth);
  const median = (values) => [...values].sort((a, b) => a - b)[Math.floor(values.length / 2)];
  const slice = (from, to) => ({
    sessions: `${from + 1}–${to}`,
    requestMs: median(growth.slice(from, to).map((row) => row.ms)),
  });
  const report = {
    date: new Date().toISOString(),
    flushWaitMs,
    rows: rows.map(({ json, ...row }) => row),
    growth: [slice(0, 50), slice(225, 275), slice(475, 525), slice(950, 1000)],
    growthTotals: {
      requests: growth.length,
      seconds: Math.round(growthSeconds * 10) / 10,
      writes: growthWrites.length,
      writtenMB: Math.round((growthWrites.reduce((sum, entry) => sum + entry.bytes, 0) / 1048576) * 100) / 100,
      finalStateKB: Math.round((all.at(-1)?.bytes ?? 0) / 1024),
      medianWriteMs: median(growthWrites.map((entry) => entry.ms)),
    },
    totalWrites: all.length,
  };
  fs.writeFileSync(path.join(outDir, 'escrituras.json'), `${JSON.stringify(report, null, 2)}\n`);
  console.log(JSON.stringify(report, null, 2));
  const unexpected = inst.unexpectedCalls?.() ?? [];
  console.log(`llamadas no simuladas por el doble: ${unexpected.length}`);
  for (const entry of unexpected.slice(0, 5)) console.log(JSON.stringify(entry));
} catch (error) {
  console.error(error);
  console.error(rows.map(({ json, ...row }) => JSON.stringify(row)).join('\n'));
  console.error(inst.log.join('').slice(-3000));
  process.exitCode = 1;
} finally {
  await inst.stop();
}
