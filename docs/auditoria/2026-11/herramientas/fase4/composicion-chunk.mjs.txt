// Reparte los bytes de un chunk entre sus módulos de origen leyendo su sourcemap.
//   node composicion-chunk.mjs <fichero.js> [<fichero.js> ...]
import fs from 'node:fs';
const B64 = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/';
function decodeLine(line) {
  const segments = [];
  for (const part of line.split(',')) {
    if (!part) continue;
    const values = [];
    let shift = 0;
    let value = 0;
    for (const char of part) {
      const digit = B64.indexOf(char);
      value += (digit & 31) << shift;
      if (digit & 32) shift += 5;
      else {
        values.push(value & 1 ? -(value >> 1) : value >> 1);
        shift = 0;
        value = 0;
      }
    }
    segments.push(values);
  }
  return segments;
}
const group = (source) => {
  const clean = source.replace(/^(\.\.\/)+/, '').replace(/^(.*\/)?repo\//, '');
  if (clean.includes('node_modules/')) return `vendor: ${clean.split('node_modules/')[1].split('/').slice(0, clean.includes('node_modules/@') ? 2 : 1).join('/')}`;
  const parts = clean.split('/');
  if (parts[0] === 'src' && (parts[1] === 'features' || parts[1] === 'components')) return parts.slice(0, parts[1] === 'features' ? 3 : 2).join('/');
  if (parts[0] === 'src') return parts.slice(0, 2).join('/');
  return parts.slice(0, 1).join('/');
};
for (const file of process.argv.slice(2)) {
  const code = fs.readFileSync(file, 'utf8').split('\n');
  const map = JSON.parse(fs.readFileSync(`${file}.map`, 'utf8'));
  const bytes = new Map();
  let source = 0;
  map.mappings.split(';').forEach((line, lineIndex) => {
    let column = 0;
    const segments = decodeLine(line).map((values) => {
      column += values[0];
      if (values.length > 1) source += values[1];
      return { column, source: values.length > 1 ? source : null };
    });
    segments.forEach((segment, index) => {
      const end = segments[index + 1]?.column ?? (code[lineIndex]?.length ?? segment.column);
      if (segment.source === null) return;
      const key = group(map.sources[segment.source]);
      bytes.set(key, (bytes.get(key) ?? 0) + (end - segment.column));
    });
  });
  const total = [...bytes.values()].reduce((sum, value) => sum + value, 0);
  console.log(`\n${file.split('/').pop()} — ${(fs.statSync(file).size / 1000).toFixed(1)} kB (${(total / 1000).toFixed(1)} kB atribuidos)`);
  for (const [key, value] of [...bytes.entries()].sort((a, b) => b[1] - a[1]).slice(0, 28)) {
    console.log(`${(value / 1000).toFixed(1).padStart(7)} kB  ${((value / total) * 100).toFixed(1).padStart(5)} %  ${key}`);
  }
}
