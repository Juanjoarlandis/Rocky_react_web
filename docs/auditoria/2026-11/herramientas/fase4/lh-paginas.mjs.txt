// Lighthouse por páginas: node lh.mjs <snapshot> <salida> <puerto> "/ /crew …"
import { execFileSync } from 'node:child_process';
import fs from 'node:fs';
process.env.ROCKY_REPO = process.argv[2];
const { startInstance } = await import('./instancia.mjs');
const [, , , outDir, port, pages] = process.argv;
fs.mkdirSync(outDir, { recursive: true });
const inst = await startInstance(Number(port), { SHOPIFY_RATE_LIMIT_MAX: '1000', SITE_ACCESS_ENABLED: 'false', TRUST_PROXY_HOPS: '1' });
const median = (values) => [...values].sort((a, b) => a - b)[Math.floor(values.length / 2)];
const rows = [];
try {
  for (const page of pages.split(/\s+/).filter(Boolean)) {
    const slug = page === '/' ? 'portada' : page.slice(1).replace(/\//g, '_');
    for (const mode of ['movil', 'escritorio']) {
      const runs = [];
      for (let run = 1; run <= 3; run += 1) {
        const target = `${outDir}/${slug}__${mode}__${run}.json`;
        try {
          execFileSync('npx', ['-y', 'lighthouse@13.5.0', inst.origin + page, ...(mode === 'escritorio' ? ['--preset=desktop'] : []),
            '--only-categories=performance', '--chrome-path=/usr/bin/google-chrome', '--chrome-flags=--headless=new --no-sandbox --disable-gpu',
            '--output=json', `--output-path=${target}`, '--quiet'], { stdio: 'ignore' });
          const r = JSON.parse(fs.readFileSync(target, 'utf8'));
          const a = r.audits;
          const items = a['resource-summary']?.details?.items ?? [];
          const net = a['network-requests']?.details?.items ?? [];
          const bytes = (filter) => net.filter(filter).reduce((t, i) => t + (i.transferSize || 0), 0) / 1000;
          runs.push({ perf: Math.round(r.categories.performance.score * 100), fcp: a['first-contentful-paint'].numericValue / 1000, lcp: a['largest-contentful-paint'].numericValue / 1000,
            cls: a['cumulative-layout-shift'].numericValue, tbt: a['total-blocking-time'].numericValue, tot: bytes(() => true), img: bytes((i) => i.resourceType === 'Image'), js: bytes((i) => i.resourceType === 'Script'), req: net.length });
        } catch (error) { console.log('FALLO', page, mode, run, String(error).slice(0, 120)); }
      }
      if (runs.length) rows.push({ page: slug, mode, n: runs.length, ...Object.fromEntries(Object.keys(runs[0]).map((key) => [key, +median(runs.map((r) => r[key])).toFixed(3)])) });
    }
  }
} finally { inst.stop(); }
fs.writeFileSync(`${outDir}/medianas.json`, JSON.stringify(rows, null, 1));
for (const r of rows) console.log(r.page.padEnd(18), r.mode.padEnd(10), 'perf', String(r.perf).padStart(3), 'FCP', r.fcp.toFixed(2), 'LCP', r.lcp.toFixed(2), 'CLS', r.cls.toFixed(3), 'TBT', Math.round(r.tbt), 'total', Math.round(r.tot), 'kB img', Math.round(r.img), 'kB js', Math.round(r.js), 'req', r.req);
