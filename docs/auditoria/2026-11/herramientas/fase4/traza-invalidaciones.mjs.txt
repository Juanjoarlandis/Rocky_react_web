process.env.ROCKY_REPO = process.argv[2];
const { startInstance } = await import('./instancia.mjs');
const { chromium } = await import(process.env.ROCKY_PLAYWRIGHT);
const wait = (ms) => new Promise((r) => setTimeout(r, ms));
const inst = await startInstance(3293, { SHOPIFY_RATE_LIMIT_MAX: '1000' });
try {
  const browser = await chromium.launch({ channel: 'chrome', headless: true, args: ['--host-resolver-rules=MAP * ~NOTFOUND, EXCLUDE localhost', '--autoplay-policy=no-user-gesture-required'] });
  const context = await browser.newContext({ viewport: { width: 375, height: 812 }, isMobile: true, hasTouch: true, deviceScaleFactor: 2, locale: 'es-ES' });
  await context.addInitScript(() => sessionStorage.setItem('rocky-splash-seen', '1'));
  const page = await context.newPage();
  await page.goto(`${inst.origin}/estudio`, { waitUntil: 'load' });
  await wait(4000);
  const play = page.getByRole('button', { name: 'Reproducir el beat' }).first();
  await play.scrollIntoViewIfNeeded();
  await play.click();
  await wait(2000);
  await browser.startTracing(page, { categories: ['devtools.timeline', 'disabled-by-default-devtools.timeline', 'disabled-by-default-devtools.timeline.stack', 'disabled-by-default-devtools.timeline.invalidationTracking'] });
  await wait(4000);
  const events = JSON.parse((await browser.stopTracing()).toString()).traceEvents;
  const count = {};
  const dur = {};
  for (const e of events) if (e.ph === 'X') { count[e.name] = (count[e.name] || 0) + 1; dur[e.name] = (dur[e.name] || 0) + (e.dur || 0) / 1000; }
  const top = Object.entries(dur).sort((a, b) => b[1] - a[1]).slice(0, 14);
  for (const [n, d] of top) console.log(n.padEnd(34), String(count[n]).padStart(5), d.toFixed(1), 'ms');
  const why = new Map();
  for (const e of events) {
    if (e.name === 'LayoutInvalidationTracking' || e.name === 'InvalidateLayout' || e.name === 'ScheduleStyleRecalculation' || e.name === 'StyleRecalcInvalidationTracking') {
      const d = e.args?.data ?? {};
      const st = (d.stackTrace ?? []).slice(0, 2).map((f) => `${f.functionName}@${(f.url || '').split('/').pop()}:${f.lineNumber}`).join(' < ');
      const k = `${e.name} | ${d.nodeName ?? ''} | ${d.reason ?? ''} ${d.extraData ?? ''} | ${st}`;
      why.set(k, (why.get(k) || 0) + 1);
    }
  }
  for (const [k, n] of [...why].sort((a, b) => b[1] - a[1]).slice(0, 25)) console.log(String(n).padStart(5), k.slice(0, 260));
  await browser.close();
} finally { inst.stop(); }
