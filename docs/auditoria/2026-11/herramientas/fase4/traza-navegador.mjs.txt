// Traza de navegador de la línea base (fase 0): coste de carga y de ejecución en
// portada, modo neón y /estudio, en móvil (CPU ×4 más lenta) y en escritorio.
//
//   ROCKY_REPO=<git archive con dist/> ROCKY_PLAYWRIGHT=<playwright-core/index.mjs> \
//     node traza-navegador.mjs <carpeta-de-salida>
//
// Mide con Chrome del sistema en headless: tareas largas, LCP y CLS (PerformanceObserver),
// ocupación del hilo principal (CDP Performance.getMetrics), cadencia de fotogramas
// (requestAnimationFrame), memoria de los procesos de Chrome (/proc) y el tamaño de las
// imágenes decodificadas frente al tamaño al que se pintan. Bloquea toda salida que no
// sea localhost. Sirve para comparar antes y después en este mismo montaje, no como
// cifra de un teléfono real.
import fs from 'node:fs';
import path from 'node:path';
import { startInstance } from './instancia.mjs';

const { chromium } = await import(process.env.ROCKY_PLAYWRIGHT);
const outDir = process.argv[2];
if (!outDir) throw new Error('Uso: traza-navegador.mjs <salida>');
const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

const PROFILES = {
  movil: {
    context: { viewport: { width: 375, height: 812 }, isMobile: true, hasTouch: true, deviceScaleFactor: 2 },
    cpuThrottle: 4,
  },
  escritorio: { context: { viewport: { width: 1440, height: 900 } }, cpuThrottle: 1 },
};

// Se inyecta antes de cualquier script de la página.
function instrument(skipSplash) {
  if (skipSplash) {
    try {
      sessionStorage.setItem('rocky-splash-seen', '1');
    } catch {
      /* sin sessionStorage */
    }
  }
  const lb = { longTasks: [], lcp: null, cls: 0, shifts: [] };
  window.__lb = lb;
  const observe = (type, handler) => {
    try {
      new PerformanceObserver((list) => list.getEntries().forEach(handler)).observe({ type, buffered: true });
    } catch {
      /* tipo no soportado */
    }
  };
  observe('longtask', (entry) => lb.longTasks.push({ start: entry.startTime, duration: entry.duration }));
  observe('largest-contentful-paint', (entry) => {
    lb.lcp = {
      time: entry.startTime,
      size: entry.size,
      url: (entry.url || '').split('/').pop(),
      element: entry.element ? `${entry.element.tagName.toLowerCase()}.${String(entry.element.className).split(' ')[0]}` : null,
    };
  });
  observe('layout-shift', (entry) => {
    if (entry.hadRecentInput) return;
    lb.cls += entry.value;
    const source = entry.sources?.[0]?.node;
    lb.shifts.push({
      value: entry.value,
      node: source?.nodeType === 1 ? `${source.tagName.toLowerCase()}.${String(source.className).split(' ')[0]}` : null,
    });
  });
  window.__lbFrames = (ms) =>
    new Promise((resolve) => {
      const deltas = [];
      let last = performance.now();
      const end = last + ms;
      const tick = (now) => {
        deltas.push(now - last);
        last = now;
        if (now < end) requestAnimationFrame(tick);
        else resolve(deltas);
      };
      requestAnimationFrame(tick);
    });
  window.__lbScroll = (ms) =>
    new Promise((resolve) => {
      const distance = document.documentElement.scrollHeight - window.innerHeight;
      const startedAt = performance.now();
      const tick = (now) => {
        const progress = Math.min(1, (now - startedAt) / ms);
        window.scrollTo(0, distance * progress);
        if (progress < 1) requestAnimationFrame(tick);
        else resolve(distance);
      };
      requestAnimationFrame(tick);
    });
}

const round = (value, digits = 1) => Math.round(value * 10 ** digits) / 10 ** digits;

function frameStats(deltas) {
  const sorted = [...deltas].sort((a, b) => a - b);
  const total = deltas.reduce((sum, delta) => sum + delta, 0);
  return {
    frames: deltas.length,
    fps: round((deltas.length / total) * 1000),
    p50Ms: round(sorted[Math.floor(sorted.length * 0.5)]),
    p95Ms: round(sorted[Math.floor(sorted.length * 0.95)]),
    maxMs: round(sorted.at(-1)),
    over25Ms: deltas.filter((delta) => delta > 25).length,
    over50Ms: deltas.filter((delta) => delta > 50).length,
  };
}

async function cdpMetrics(cdp) {
  const { metrics } = await cdp.send('Performance.getMetrics');
  return Object.fromEntries(metrics.map((metric) => [metric.name, metric.value]));
}

// CPU acumulada (segundos, todos los hilos) de los procesos renderer y GPU de Chrome.
async function processCpu(browser) {
  const session = await browser.newBrowserCDPSession();
  const { processInfo } = await session.send('SystemInfo.getProcessInfo');
  await session.detach();
  const cpu = { renderer: 0, GPU: 0 };
  for (const info of processInfo) if (info.type in cpu) cpu[info.type] += info.cpuTime;
  return cpu;
}

// Ocupación del hilo principal, CPU de los procesos de Chrome y, si `frames`, cadencia de
// fotogramas mientras ocurre `action`. En reposo se mide sin `frames`: el muestreo con
// requestAnimationFrame obliga a producir un fotograma por refresco y falsearía el reposo.
async function measure(page, cdp, browser, ms, action = null, { frames = true } = {}) {
  const before = await cdpMetrics(cdp);
  const cpuBefore = await processCpu(browser);
  const tasksBefore = await page.evaluate(() => window.__lb.longTasks.length);
  const startedAt = performance.now();
  const sampling = frames ? page.evaluate((duration) => window.__lbFrames(duration), ms) : wait(ms);
  if (action) await action();
  const deltas = await sampling;
  const elapsed = (performance.now() - startedAt) / 1000;
  const after = await cdpMetrics(cdp);
  const cpuAfter = await processCpu(browser);
  const longTasks = await page.evaluate((from) => window.__lb.longTasks.slice(from), tasksBefore);
  const busy = (name) => round(((after[name] - before[name]) / elapsed) * 100);
  return {
    seconds: round(elapsed),
    mainThreadBusyPct: busy('TaskDuration'),
    scriptPct: busy('ScriptDuration'),
    layoutPct: busy('LayoutDuration'),
    stylePct: busy('RecalcStyleDuration'),
    rendererCpuPct: round(((cpuAfter.renderer - cpuBefore.renderer) / elapsed) * 100),
    gpuCpuPct: round(((cpuAfter.GPU - cpuBefore.GPU) / elapsed) * 100),
    layouts: after.LayoutCount - before.LayoutCount,
    styleRecalcs: after.RecalcStyleCount - before.RecalcStyleCount,
    longTasks: longTasks.length,
    longTaskMs: round(longTasks.reduce((sum, task) => sum + task.duration, 0)),
    ...(frames ? frameStats(deltas) : {}),
  };
}

async function chromeMemory(browser) {
  const session = await browser.newBrowserCDPSession();
  const { processInfo } = await session.send('SystemInfo.getProcessInfo');
  await session.detach();
  const byType = {};
  for (const info of processInfo) {
    let rss = 0;
    try {
      const status = fs.readFileSync(`/proc/${info.id}/status`, 'utf8');
      rss = Number(status.match(/VmRSS:\s+(\d+)/)?.[1] ?? 0) / 1024;
    } catch {
      /* el proceso ya no existe */
    }
    byType[info.type] = round((byType[info.type] ?? 0) + rss);
  }
  return byType;
}

async function pageSnapshot(page, cdp, browser) {
  const inPage = await page.evaluate(() => {
    const dpr = window.devicePixelRatio;
    const seen = new Map();
    for (const image of document.images) {
      if (!image.complete || !image.naturalWidth) continue;
      const name = (image.currentSrc || image.src).split('/').pop();
      const shown = Math.max(image.clientWidth * dpr, 1) * Math.max(image.clientHeight * dpr, 1);
      const natural = image.naturalWidth * image.naturalHeight;
      const previous = seen.get(name);
      if (!previous || shown > previous.shown) {
        seen.set(name, { name, w: image.naturalWidth, h: image.naturalHeight, natural, shown, visible: image.clientWidth > 0 });
      }
    }
    const images = [...seen.values()];
    const resources = {};
    for (const entry of performance.getEntriesByType('resource')) {
      const extension = entry.name.split('?')[0].split('.').pop().toLowerCase();
      const kind =
        { js: 'js', css: 'css', woff2: 'fuentes', webp: 'imagen', png: 'imagen', svg: 'imagen', m4a: 'audio' }[extension] ??
        (entry.name.includes('/api/') ? 'api' : 'otros');
      resources[kind] ??= { n: 0, kB: 0 };
      resources[kind].n += 1;
      resources[kind].kB += (entry.transferSize || entry.encodedBodySize || 0) / 1024;
    }
    for (const kind of Object.keys(resources)) resources[kind].kB = Math.round(resources[kind].kB);
    return {
      theme: document.documentElement.dataset.theme ?? 'light',
      lcp: window.__lb.lcp,
      cls: Math.round(window.__lb.cls * 1000) / 1000,
      worstShift: [...window.__lb.shifts].sort((a, b) => b.value - a.value)[0] ?? null,
      longTasks: window.__lb.longTasks.length,
      longTaskMs: Math.round(window.__lb.longTasks.reduce((sum, task) => sum + task.duration, 0)),
      blockingMs: Math.round(window.__lb.longTasks.reduce((sum, task) => sum + Math.max(0, task.duration - 50), 0)),
      resources,
      images: {
        loaded: images.length,
        decodedMB: Math.round((images.reduce((sum, image) => sum + image.natural * 4, 0) / 1024 / 1024) * 10) / 10,
        neededMB: Math.round((images.filter((image) => image.visible).reduce((sum, image) => sum + Math.min(image.natural, image.shown) * 4, 0) / 1024 / 1024) * 10) / 10,
        oversized: images
          .filter((image) => image.visible && image.natural > image.shown * 4)
          .sort((a, b) => b.natural - a.natural)
          .slice(0, 6)
          .map((image) => `${image.name} ${image.w}×${image.h} (×${Math.round(image.natural / image.shown)} píxeles de más)`),
      },
      domNodes: document.getElementsByTagName('*').length,
      animations: document.getAnimations().filter((animation) => animation.playState === 'running').length,
    };
  });
  const metrics = await cdpMetrics(cdp);
  return {
    ...inPage,
    jsHeapMB: round(metrics.JSHeapUsedSize / 1024 / 1024),
    taskSecondsSinceLoad: round(metrics.TaskDuration, 2),
    scriptSecondsSinceLoad: round(metrics.ScriptDuration, 2),
    chromeRssMB: await chromeMemory(browser),
  };
}

async function open(profileName, { skipSplash }) {
  const profile = PROFILES[profileName];
  const browser = await chromium.launch({
    channel: 'chrome',
    headless: true,
    args: ['--host-resolver-rules=MAP * ~NOTFOUND, EXCLUDE localhost', '--autoplay-policy=no-user-gesture-required'],
  });
  const context = await browser.newContext({ ...profile.context, locale: 'es-ES' });
  await context.addInitScript(instrument, skipSplash);
  const page = await context.newPage();
  const cdp = await context.newCDPSession(page);
  await cdp.send('Performance.enable');
  if (profile.cpuThrottle > 1) await cdp.send('Emulation.setCPUThrottlingRate', { rate: profile.cpuThrottle });
  const errors = [];
  page.on('pageerror', (error) => errors.push(String(error).slice(0, 200)));
  return { browser, page, cdp, errors };
}

const scenarios = {
  // Primera visita a la portada, con la intro de 3 s y la caché vacía.
  async portada(origin, profileName) {
    const { browser, page, cdp, errors } = await open(profileName, { skipSplash: false });
    await page.goto(`${origin}/`, { waitUntil: 'load' });
    await wait(6000);
    const result = { carga: await pageSnapshot(page, cdp, browser) };
    result.reposo = await measure(page, cdp, browser, 10_000, null, { frames: false });
    result.scroll = await measure(page, cdp, browser, 8000, () => page.evaluate(() => window.__lbScroll(7500)));
    await wait(1500);
    result.trasScroll = await pageSnapshot(page, cdp, browser);
    result.errors = errors;
    await browser.close();
    return result;
  },

  // Portada con el modo neón encendido: reposo, puntero (linterna) y scroll.
  async neon(origin, profileName) {
    const { browser, page, cdp, errors } = await open(profileName, { skipSplash: true });
    await page.goto(`${origin}/`, { waitUntil: 'load' });
    await wait(2500);
    const bulb = page.getByRole('button', { name: 'Activar el descontrol' }).first();
    if (!(await bulb.isVisible())) await page.getByRole('button', { name: 'Abrir el menú' }).click();
    const encendido = await measure(page, cdp, browser, 5000, async () => {
      await page.getByRole('button', { name: 'Activar el descontrol' }).last().click();
    });
    await wait(2000);
    const result = { encendido, carga: await pageSnapshot(page, cdp, browser) };
    // Encender el neón pone a sonar su cinta: el primer reposo es con ella y el segundo,
    // con la radio parada, para separar lo que cuesta el neón de lo que cuesta la música.
    result.radio = await page.evaluate(() => ({
      sonando: !document.querySelector('audio')?.paused,
      groove: document.body.hasAttribute('data-groove') || document.documentElement.hasAttribute('data-groove'),
    }));
    result.reposo = await measure(page, cdp, browser, 10_000, null, { frames: false });
    await page.evaluate(() => document.querySelector('audio')?.pause());
    await wait(1500);
    result.reposoSinCinta = await measure(page, cdp, browser, 10_000, null, { frames: false });
    result.animacionesSinCinta = await page.evaluate(() =>
      document
        .getAnimations()
        .filter((animation) => animation.playState === 'running')
        .map((animation) => `${animation.animationName || 'transición'} → ${animation.effect?.target?.className?.toString().split(' ')[0] ?? '?'}`)
    );
    if (profileName === 'escritorio') {
      result.puntero = await measure(page, cdp, browser, 6000, async () => {
        for (let step = 0; step < 240; step += 1) {
          const x = 100 + ((step * 37) % 1240);
          const y = 150 + ((step * 23) % 600);
          await page.mouse.move(x, y);
          await wait(20);
        }
      });
    }
    result.scroll = await measure(page, cdp, browser, 8000, () => page.evaluate(() => window.__lbScroll(7500)));
    await wait(1500);
    result.trasScroll = await pageSnapshot(page, cdp, browser);
    result.errors = errors;
    await browser.close();
    return result;
  },

  // El Estudio: carga, reposo y la mesa de beats sonando.
  async estudio(origin, profileName) {
    const { browser, page, cdp, errors } = await open(profileName, { skipSplash: true });
    await page.goto(`${origin}/estudio`, { waitUntil: 'load' });
    await wait(4000);
    const result = { carga: await pageSnapshot(page, cdp, browser) };
    result.reposo = await measure(page, cdp, browser, 10_000, null, { frames: false });
    const play = page.getByRole('button', { name: 'Reproducir el beat' }).first();
    await play.scrollIntoViewIfNeeded();
    await wait(500);
    result.sonando = await measure(page, cdp, browser, 10_000, async () => {
      await play.click();
      await wait(9000);
    });
    result.sonabaAlAcabar = await page.getByRole('button', { name: 'Parar el beat' }).count();
    result.trasSonar = await pageSnapshot(page, cdp, browser);
    result.errors = errors;
    await browser.close();
    return result;
  },
};

const inst = await startInstance(3292, { SHOPIFY_RATE_LIMIT_MAX: '1000' });
const report = { date: new Date().toISOString(), chrome: null, results: {} };
try {
  // ROCKY_ESCENARIOS=neon,estudio limita la traza a esos escenarios.
  const wanted = (process.env.ROCKY_ESCENARIOS || '').split(',').filter(Boolean);
  for (const [scenarioName, scenario] of Object.entries(scenarios)) {
    if (wanted.length && !wanted.includes(scenarioName)) continue;
    for (const profileName of Object.keys(PROFILES)) {
      const key = `${scenarioName}__${profileName}`;
      try {
        report.results[key] = await scenario(inst.origin, profileName);
      } catch (error) {
        report.results[key] = { error: String(error).slice(0, 600) };
      }
      console.error(`hecho: ${key}`);
    }
  }
} finally {
  inst.stop();
}
fs.writeFileSync(path.join(outDir, process.env.ROCKY_ESCENARIOS ? `traza-navegador-${process.env.ROCKY_ESCENARIOS.replace(/,/g, '-')}.json` : 'traza-navegador.json'), `${JSON.stringify(report, null, 2)}\n`);
console.log(JSON.stringify(report, null, 2));
