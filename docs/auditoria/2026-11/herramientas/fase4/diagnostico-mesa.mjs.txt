// Diagnóstico del neón en reposo: node neon-diag.mjs <snapshot> <puerto> [perfil]
// Enciende el neón en la portada, para la radio y mide 6 s en reposo desactivando
// los candidatos de uno en uno (cada variante en una página nueva).
process.env.ROCKY_REPO = process.argv[2];
const { startInstance } = await import('./instancia.mjs');
const { chromium } = await import(process.env.ROCKY_PLAYWRIGHT);
const port = Number(process.argv[3]);
const only = process.argv[4];
const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

const VARIANTS = process.env.EXTRA ? JSON.parse(process.env.EXTRA) : {
  base: '',
  'sin tubo-roto': "[data-theme='neon'] .page-title{animation:none}",
  'sin lata-patrulla': '.ticker-lata{animation:none}',
  'sin ticker-scroll': '.ticker-track{animation:none}',
  'sin uv-wall': '.uv-wall{display:none}',
  'sin uv-oscuridad': '.uv-oscuridad{display:none}',
  'sin lienzo': 'canvas{display:none}',
  'sin animaciones CSS': '*,*::before,*::after{animation:none}',
  'sin webp animados': '__img__',
};
const PROFILES = {
  movil: { context: { viewport: { width: 375, height: 812 }, isMobile: true, hasTouch: true, deviceScaleFactor: 2 }, rate: 4 },
  escritorio: { context: { viewport: { width: 1440, height: 900 } }, rate: 1 },
};

async function metrics(cdp, browser) {
  const { metrics: list } = await cdp.send('Performance.getMetrics');
  const m = Object.fromEntries(list.map((x) => [x.name, x.value]));
  const session = await browser.newBrowserCDPSession();
  const { processInfo } = await session.send('SystemInfo.getProcessInfo');
  await session.detach();
  m.cpu = processInfo.filter((p) => p.type === 'renderer' || p.type === 'GPU').reduce((t, p) => t + p.cpuTime, 0);
  m.gpu = processInfo.filter((p) => p.type === 'GPU').reduce((t, p) => t + p.cpuTime, 0);
  return m;
}

const inst = await startInstance(port, { SHOPIFY_RATE_LIMIT_MAX: '1000' });
try {
  for (const [profileName, profile] of Object.entries(PROFILES)) {
    if (only && only !== profileName) continue;
    for (const [name, css] of Object.entries(VARIANTS)) {
      const browser = await chromium.launch({
        channel: 'chrome',
        headless: true,
        args: ['--host-resolver-rules=MAP * ~NOTFOUND, EXCLUDE localhost', '--autoplay-policy=no-user-gesture-required'],
      });
      const context = await browser.newContext({ ...profile.context, locale: 'es-ES', bypassCSP: true });
      await context.addInitScript(() => sessionStorage.setItem('rocky-splash-seen', '1'));
      const page = await context.newPage();
      const cdp = await context.newCDPSession(page);
      await cdp.send('Performance.enable');
      if (profile.rate > 1) await cdp.send('Emulation.setCPUThrottlingRate', { rate: profile.rate });
      await page.goto(`${inst.origin}${process.env.RUTA || '/'}`, { waitUntil: 'load' });
      await wait(2500);
      if (process.env.PRE && css && css !== '__img__') await page.addStyleTag({ content: css });
      if (process.env.RUTA === '/estudio') {
        await wait(1500);
        const play = page.getByRole('button', { name: 'Reproducir el beat' }).first();
        await play.scrollIntoViewIfNeeded();
        if (css) await page.addStyleTag({ content: css });
        await play.click();
      } else {
      const bulb = page.getByRole('button', { name: 'Activar el descontrol' }).first();
        if (!(await bulb.isVisible())) await page.getByRole('button', { name: 'Abrir el menú' }).click();
        await page.getByRole('button', { name: 'Activar el descontrol' }).last().click();
        await wait(6000);
        await page.evaluate(() => document.querySelector('audio')?.pause());
        if (css === '__img__') {
          await page.evaluate(() => document.querySelectorAll('img').forEach((img) => { if (/-h\d+|anim|walk|run/.test(img.currentSrc)) img.style.display = 'none'; }));
        } else if (css && !process.env.PRE) await page.addStyleTag({ content: css });
}
      await wait(1500);
      const before = await metrics(cdp, browser);
      const t0 = performance.now();
      if (process.env.MODO === 'puntero') {
        for (let step = 0; step < 240; step += 1) {
          await page.mouse.move(100 + ((step * 37) % 1240), 150 + ((step * 23) % 600));
          await wait(20);
        }
      } else if (process.env.MODO === 'scroll') {
        await page.evaluate(() => new Promise((resolve) => {
          const distance = document.documentElement.scrollHeight - window.innerHeight;
          const startedAt = performance.now();
          const tick = (now) => {
            const progress = Math.min(1, (now - startedAt) / 5500);
            window.scrollTo(0, distance * progress);
            if (progress < 1) requestAnimationFrame(tick); else resolve();
          };
          requestAnimationFrame(tick);
        }));
      } else await wait(6000);
      const s = (performance.now() - t0) / 1000;
      const after = await metrics(cdp, browser);
      const pct = (k) => (((after[k] - before[k]) / s) * 100).toFixed(1).padStart(5);
      const info = await page.evaluate(() => ({
        anims: document.getAnimations().filter((a) => a.playState === 'running').map((a) => a.animationName || 'tr').join(','),
        canvas: document.querySelectorAll('canvas').length,
      }));
      console.log(profileName.padEnd(10), name.padEnd(20), 'hilo', pct('TaskDuration'), '% estilo', pct('RecalcStyleDuration'), '% layout', pct('LayoutDuration'), '% cpu', pct('cpu'), '% (gpu', pct('gpu'), ')', '% recálculos', String(after.RecalcStyleCount - before.RecalcStyleCount).padStart(4), 'layouts', String(after.LayoutCount - before.LayoutCount).padStart(4), '|', info.canvas, info.anims);
      await browser.close();
    }
  }
} finally {
  inst.stop();
}
