// Memoria del servidor en un contenedor limitado (servidor-ligero).
//
//   ROCKY_REPO=<repo con dist/ y server/> node medir-memoria.mjs <imagen> <salida> [memoria=128m] [puerto=3392]
//
// Variables: ROCKY_CONTAINER_ENV="NODE_OPTIONS=--max-old-space-size=64;OTRA=valor" (entorno
// extra del contenedor), ROCKY_ROUNDS (tandas de carga seguidas, 3), ROCKY_TAG (sufijo del
// fichero de salida), ROCKY_BIG_STORE=1 (siembra 5.000 sesiones y lanza 200 mutaciones que
// reescriben el almacén), ROCKY_IDLE=1 (además, 120 s de reposo con la sonda de la imagen).
import { execFileSync, spawnSync } from 'node:child_process';
import crypto from 'node:crypto';
import fs from 'node:fs';
import path from 'node:path';

const [image, outArg, memory = '128m', portArg = '3392'] = process.argv.slice(2);
const outDir = outArg && path.resolve(outArg);
if (!image || !outDir) throw new Error('Uso: medir-memoria.mjs <imagen> <salida> [memoria] [puerto]');
const repo = process.env.ROCKY_REPO;
const port = Number(portArg);
const base = `http://127.0.0.1:${port}`;
const tag = process.env.ROCKY_TAG || 'sin-etiqueta';
const name = `rocky-memoria-${tag}`;
const dataDir = path.join(outDir, `data-${tag}`);
const rounds = Number(process.env.ROCKY_ROUNDS || 3);
const extraEnv = (process.env.ROCKY_CONTAINER_ENV || '').split(';').filter(Boolean);
const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));
const docker = (...args) => execFileSync('docker', args, { encoding: 'utf8' }).trim();
const inContainer = (command) => docker('exec', name, 'sh', '-c', command);
const key = crypto.randomBytes(32).toString('base64');
const mb = (bytes) => Math.round((Number(bytes) / 1048576) * 10) / 10;

async function seedBigStore() {
  const { EncryptedFileStore } = await import(path.join(repo, 'server/storage/encrypted-file-store.mjs'));
  const store = new EncryptedFileStore({ filePath: path.join(dataDir, 'rocky-state.enc'), key });
  await store.set('sessions', 'semilla', { createdAt: Date.now() });
  const expiresAt = Date.now() + 14 * 86_400_000;
  for (let i = 0; i < 5000; i += 1) {
    const id = crypto.randomBytes(32).toString('hex');
    store.state.namespaces.sessions[id] = {
      value: { createdAt: Date.now(), lastSeenAt: Date.now(), csrf: crypto.randomBytes(24).toString('hex'), cartId: `gid://shopify/Cart/${crypto.randomBytes(24).toString('hex')}?key=${crypto.randomBytes(16).toString('hex')}`, pared: crypto.randomBytes(16).toString('hex') },
      expiresAt,
    };
    (store.state.namespaces.cartOperations ??= {})[`${id}:${crypto.randomUUID()}`] = {
      value: { status: 'completed', result: { cart: { totalQuantity: 1, lines: [{ id: `gid://shopify/CartLine/${crypto.randomUUID()}?cart=${crypto.randomBytes(12).toString('hex')}`, quantity: 1, variant: { id: 'gid://shopify/ProductVariant/1', title: 'M', product: { handle: 'rocky-racing', title: 'Rocky Racing' }, price: { amount: '35.00', currencyCode: 'EUR' } } }] } } },
      expiresAt,
    };
  }
  await store.set('sessions', 'semilla', { createdAt: Date.now() });
  return fs.statSync(path.join(dataDir, 'rocky-state.enc')).size;
}

function run({ healthcheck }) {
  spawnSync('docker', ['rm', '-f', name], { stdio: 'ignore' });
  const args = ['run', '-d', '--name', name, '--init', '--read-only', '--tmpfs', '/tmp:rw,noexec,nosuid,nodev,size=16m',
    '--cap-drop', 'ALL', '--security-opt', 'no-new-privileges:true', '--pids-limit', '256',
    '--cpus=1', `--memory=${memory}`, `--memory-swap=${memory}`, '-p', `127.0.0.1:${port}:3001`, '-v', `${dataDir}:/app/.data`];
  const containerEnv = { PUBLIC_ORIGIN: 'https://rocky.test', API_ALLOWED_ORIGINS: 'https://rocky.test', TRUST_PROXY_HOPS: '1', SITE_ACCESS_ENABLED: 'false', SHOPIFY_RATE_LIMIT_MAX: '2000', AVISOS_RATE_LIMIT_MAX: '100' };
  containerEnv.APP_ENCRYPTION_KEY = key;
  for (const [k, v] of Object.entries(containerEnv)) args.push('-e', `${k}=${v}`);
  for (const pair of extraEnv) args.push('-e', pair);
  if (!healthcheck) args.push('--no-healthcheck');
  args.push(image);
  docker(...args);
}
async function serving() {
  for (let i = 0; i < 1500; i += 1) {
    try { if ((await fetch(`${base}/api/health`)).ok) return; } catch { /* arrancando */ }
    await wait(20);
  }
  throw new Error(`No arranca:\n${docker('logs', name)}`);
}
const running = () => docker('inspect', '-f', '{{.State.Running}}', name) === 'true';
const snapshot = () => ({ cgroupMB: mb(inContainer('cat /sys/fs/cgroup/memory.current')), peakMB: mb(inContainer('cat /sys/fs/cgroup/memory.peak')) });
const cpuUsec = () => Number(inContainer('grep usage_usec /sys/fs/cgroup/cpu.stat').split(' ')[1]);
function ab(urlPath, n, headers = []) {
  const args = ['-q', '-k', '-n', String(n), '-c', '20'];
  for (const header of headers) args.push('-H', header);
  args.push(`${base}${urlPath}`);
  const text = spawnSync('ab', args, { encoding: 'utf8', maxBuffer: 32 * 1024 * 1024 }).stdout || '';
  const number = (pattern) => Number(text.match(pattern)?.[1] ?? NaN);
  return { failed: number(/Failed requests:\s+(\d+)/), non2xx: number(/Non-2xx responses:\s+(\d+)/) || 0, complete: number(/Complete requests:\s+(\d+)/), rps: number(/Requests per second:\s+([\d.]+)/), bytes: number(/Document Length:\s+(\d+)/) };
}

fs.rmSync(dataDir, { recursive: true, force: true });
fs.mkdirSync(dataDir, { recursive: true, mode: 0o700 });
const report = { image, memory, tag, env: extraEnv, date: new Date().toISOString() };
if (process.env.ROCKY_BIG_STORE === '1') report.seededStoreMB = mb(await seedBigStore());

const assets = fs.readdirSync(path.join(repo, 'dist', 'assets'));
const asset = (pattern) => `/assets/${assets.find((file) => pattern.test(file))}`;
run({ healthcheck: false });
await serving();
await wait(3000);
report.idle = snapshot();
// El límite de heap no se lee aquí: lanzar otro Node dentro del contenedor medido sube su pico.

if (process.env.ROCKY_BIG_STORE === '1') {
  // 200 altas de aviso seguidas: cada una reescribe el almacén entero, como un alta de carrito.
  const PRODUCTOS = ['rocky-airwave', 'rocky-night-runner', 'rocky-pit-crew', 'rocky-signal-ghost', 'rocky-solar-club', 'rocky-marea-035'];
  const results = { ok: 0, other: {} };
  for (let i = 0; i < 200; i += 1) {
    if (!running()) break;
    try {
      const response = await fetch(`${base}/api/avisos`, { method: 'POST', headers: { 'Content-Type': 'application/json', Origin: 'https://rocky.test', 'X-Forwarded-For': `198.51.100.${(i % 250) + 1}` }, body: JSON.stringify({ producto: PRODUCTOS[i % PRODUCTOS.length], email: `persona${i}@example.test`, consentimiento: true }) });
      if (response.ok) results.ok += 1; else results.other[response.status] = (results.other[response.status] || 0) + 1;
    } catch (error) { results.other.red = (results.other.red || 0) + 1; }
  }
  report.bigStore = { ...results, alive: running(), memory: running() ? snapshot() : null };
  if (results.ok !== 200) console.error(`AVISO: solo ${results.ok} de 200 mutaciones respondieron 200.`);
} else {
  const steps = [['/', 5000], [asset(/^index-.*\.js$/), 2000], [asset(/^index-.*\.js$/), 2000, ['Accept-Encoding: br, gzip']], [asset(/^index-.*\.css$/), 3000], [asset(/^lata-spray-walk-seedance-224-.*\.webp$/), 1000], ['/api/drops', 5000], ['/api/shopify/status', 600]];
  report.rounds = [];
  for (let round = 1; round <= rounds && running(); round += 1) {
    const results = [];
    for (const [urlPath, n, headers] of steps) {
      const result = ab(urlPath, n, headers);
      results.push({ path: urlPath.replace(/-[A-Za-z0-9_-]{8}\./, '.'), enc: headers ? 'br' : '', ...result });
      if (!running()) break;
    }
    report.rounds.push({ round, failed: results.reduce((t, r) => t + (r.failed || 0) + (r.non2xx || 0), 0), incomplete: results.filter((r) => !(r.complete > 0)).length, memory: running() ? snapshot() : null, results });
  }
}
report.oomKilled = docker('inspect', '-f', '{{.State.OOMKilled}}', name);
report.alive = running();
docker('rm', '-f', name);

if (process.env.ROCKY_IDLE === '1') {
  fs.rmSync(dataDir, { recursive: true, force: true });
  fs.mkdirSync(dataDir, { recursive: true, mode: 0o700 });
  run({ healthcheck: true });
  await serving();
  await wait(15_000);
  const before = cpuUsec();
  await wait(120_000);
  report.idleWithProbe = { seconds: 120, cpuMs: Math.round((cpuUsec() - before) / 1000), ...snapshot(), health: docker('inspect', '-f', '{{.State.Health.Status}}', name) };
  docker('rm', '-f', name);
}
fs.rmSync(dataDir, { recursive: true, force: true });
fs.writeFileSync(path.join(outDir, `memoria-${memory}-${tag}.json`), `${JSON.stringify(report, null, 2)}\n`);
const brief = { tag, env: extraEnv, idle: report.idle, oomKilled: report.oomKilled, alive: report.alive };
if (report.rounds) brief.rounds = report.rounds.map((r) => ({ round: r.round, failed: r.failed, incomplete: r.incomplete, ...r.memory, jsBrBytes: r.results[2]?.bytes, jsRps: r.results[1]?.rps }));
if (report.bigStore) brief.bigStore = { seededMB: report.seededStoreMB, ...report.bigStore };
if (report.idleWithProbe) brief.idleWithProbe = report.idleWithProbe;
console.log(JSON.stringify(brief));
