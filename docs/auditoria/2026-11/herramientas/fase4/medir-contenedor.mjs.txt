// Línea base del servidor en un contenedor limitado (fase 0 de la optimización).
//
//   ROCKY_REPO=<git archive con dist/> node medir-contenedor.mjs <imagen> <salida> [memoria=256m] [puerto=3391]
//
// Arranca la imagen con las mismas restricciones que compose.production.yaml más
// --cpus=1 y --memory, y mide: arranque, memoria y CPU en reposo (con y sin el
// HEALTHCHECK de la imagen), latencia de estáticos y de la API con ApacheBench, y
// memoria tras la carga. No lee .env: fija lo mínimo que exige NODE_ENV=production.
import { execFileSync, spawnSync } from 'node:child_process';
import crypto from 'node:crypto';
import fs from 'node:fs';
import path from 'node:path';

const [image, outDir, memory = '256m', portArg = '3391'] = process.argv.slice(2);
if (!image || !outDir) throw new Error('Uso: medir-contenedor.mjs <imagen> <salida> [memoria] [puerto]');
const port = Number(portArg);
const base = `http://127.0.0.1:${port}`;
const name = `rocky-lineabase-${memory}`;
const dataDir = path.join(outDir, `data-${memory}`);
const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));
// ROCKY_CONCURRENCY: conexiones simultáneas de ab. ROCKY_QUICK=1: solo arranque y carga
// (sin las esperas de reposo ni la fase del HEALTHCHECK), con la memoria tras cada prueba.
const concurrency = Number(process.env.ROCKY_CONCURRENCY || 20);
const quick = process.env.ROCKY_QUICK === '1';
const docker = (...args) => execFileSync('docker', args, { encoding: 'utf8' }).trim();
const inContainer = (command) => docker('exec', name, 'sh', '-c', command);

function runContainer({ healthcheck }) {
  spawnSync('docker', ['rm', '-f', name], { stdio: 'ignore' });
  fs.rmSync(dataDir, { recursive: true, force: true });
  fs.mkdirSync(dataDir, { recursive: true, mode: 0o700 });
  const args = [
    'run', '-d', '--name', name,
    '--init', '--read-only',
    '--tmpfs', '/tmp:rw,noexec,nosuid,nodev,size=16m',
    '--cap-drop', 'ALL', '--security-opt', 'no-new-privileges:true', '--pids-limit', '256',
    '--cpus=1', `--memory=${memory}`, `--memory-swap=${memory}`,
    '-p', `127.0.0.1:${port}:3001`,
    '-v', `${dataDir}:/app/.data`,
  ];
  // Lo mínimo que exige NODE_ENV=production, sin Shopify. La clave del almacén es
  // aleatoria y solo vive lo que dura la medida.
  const containerEnv = {
    PUBLIC_ORIGIN: 'https://rocky.test',
    API_ALLOWED_ORIGINS: 'https://rocky.test',
    TRUST_PROXY_HOPS: '1',
    SITE_ACCESS_ENABLED: 'false',
    SHOPIFY_RATE_LIMIT_MAX: '2000',
    APP_ENCRYPTION_KEY: crypto.randomBytes(32).toString('base64'),
  };
  for (const [key, value] of Object.entries(containerEnv)) args.push('-e', `${key}=${value}`);
  if (!healthcheck) args.push('--no-healthcheck');
  args.push(image);
  const startedAt = performance.now();
  docker(...args);
  return startedAt;
}

async function waitUntilServing(startedAt) {
  for (let attempt = 0; attempt < 2000; attempt += 1) {
    try {
      const response = await fetch(`${base}/`);
      if (response.ok) return performance.now() - startedAt;
    } catch {
      /* todavía arrancando */
    }
    await wait(10);
  }
  throw new Error(`El contenedor no sirve /:\n${docker('logs', name)}`);
}

// Tiempo de arranque de Node dentro del contenedor: de State.StartedAt a la línea
// «listening» del log, sin el coste de crear el contenedor.
function bootInsideContainer() {
  const started = Date.parse(docker('inspect', '-f', '{{.State.StartedAt}}', name));
  const line = docker('logs', '--timestamps', name)
    .split('\n')
    .find((entry) => entry.includes('listening'));
  return line ? Date.parse(line.split(' ')[0]) - started : null;
}

function memorySnapshot() {
  const stat = Object.fromEntries(
    inContainer('cat /sys/fs/cgroup/memory.stat')
      .split('\n')
      .map((line) => line.split(' '))
      .map(([key, value]) => [key, Number(value)])
  );
  const node = docker('top', name, '-eo', 'pid,rss,comm')
    .split('\n')
    .map((line) => line.trim().split(/\s+/))
    .filter(([, , comm]) => /node|MainThread/.test(comm || ''))
    .map(([, rss]) => Number(rss));
  const mb = (bytes) => Math.round((bytes / 1024 / 1024) * 10) / 10;
  return {
    cgroupMB: mb(Number(inContainer('cat /sys/fs/cgroup/memory.current'))),
    peakMB: mb(Number(inContainer('cat /sys/fs/cgroup/memory.peak'))),
    anonMB: mb(stat.anon),
    fileMB: mb(stat.file),
    nodeRssMB: Math.round((Math.max(0, ...node) / 1024) * 10) / 10,
  };
}

const cpuUsec = () => Number(inContainer('grep usage_usec /sys/fs/cgroup/cpu.stat').split(' ')[1]);

async function idleCpu(seconds) {
  const before = cpuUsec();
  await wait(seconds * 1000);
  const used = (cpuUsec() - before) / 1000; // ms de CPU
  return { seconds, cpuMs: Math.round(used), percentOfOneCpu: Math.round((used / (seconds * 1000)) * 10000) / 100 };
}

function ab(label, urlPath, { n, c = concurrency, headers = [] }) {
  const args = ['-q', '-k', '-n', String(n), '-c', String(c)];
  for (const header of headers) args.push('-H', header);
  args.push(`${base}${urlPath}`);
  const result = spawnSync('ab', args, { encoding: 'utf8', maxBuffer: 32 * 1024 * 1024 });
  const text = result.stdout || '';
  const number = (pattern) => {
    const match = text.match(pattern);
    return match ? Number(match[1]) : null;
  };
  return {
    label,
    path: urlPath,
    n,
    c,
    bytes: number(/Document Length:\s+(\d+)/),
    failed: number(/Failed requests:\s+(\d+)/),
    non2xx: number(/Non-2xx responses:\s+(\d+)/) ?? 0,
    rps: number(/Requests per second:\s+([\d.]+)/),
    kbPerSecond: number(/Transfer rate:\s+([\d.]+)/),
    p50: number(/\s50%\s+(\d+)/),
    p95: number(/\s95%\s+(\d+)/),
    p99: number(/\s99%\s+(\d+)/),
    max: number(/100%\s+(\d+)/),
  };
}

// Los nombres con hash salen del dist/ del mismo commit que la imagen.
const repo = process.env.ROCKY_REPO || path.join(outDir, '..', 'repo');
const assets = fs.readdirSync(path.join(repo, 'dist', 'assets'));
const asset = (pattern) => `/assets/${assets.find((file) => pattern.test(file))}`;
const report = { image, memory, cpus: 1, date: new Date().toISOString() };

// 1. Reposo sin HEALTHCHECK: arranque, memoria y CPU propias del servidor.
let startedAt = runContainer({ healthcheck: false });
report.startupMs = Math.round(await waitUntilServing(startedAt));
report.bootInsideMs = bootInsideContainer();
await wait(quick ? 3000 : 15_000);
report.idle = memorySnapshot();
if (!quick) report.idleCpuNoHealthcheck = await idleCpu(60);

// 2. Carga: estáticos pequeños y grandes, y dos rutas de la API sin Shopify. Tras cada
// prueba se anota la memoria; si el contenedor muere (OOM), se para ahí.
const isRunning = () => docker('inspect', '-f', '{{.State.Running}}', name) === 'true';
const steps = [
  ['index.html', '/', { n: 5000 }],
  ['JS principal', asset(/^index-.*\.js$/), { n: 2000 }],
  ['JS principal (Accept-Encoding: br, gzip)', asset(/^index-.*\.js$/), { n: 2000, headers: ['Accept-Encoding: br, gzip'] }],
  ['CSS principal', asset(/^index-.*\.css$/), { n: 3000 }],
  ['WebP animado del pie', asset(/^lata-spray-walk-seedance-224-.*\.webp$/), { n: 1000 }],
  ['GET /api/drops', '/api/drops', { n: 5000 }],
  ['GET /api/shopify/status', '/api/shopify/status', { n: 1500 }],
];
report.concurrency = concurrency;
report.load = [];
for (const [label, urlPath, options] of steps) {
  const result = ab(label, urlPath, options);
  if (isRunning()) {
    report.load.push({ ...result, memory: memorySnapshot() });
  } else {
    report.load.push({ ...result, memory: null });
    report.diedDuring = label;
    break;
  }
}
report.oomKilled = docker('inspect', '-f', '{{.State.OOMKilled}}', name);
report.exitCode = docker('inspect', '-f', '{{.State.ExitCode}}', name);
if (isRunning()) {
  report.afterLoad = memorySnapshot();
  await wait(quick ? 5000 : 20_000);
  report.afterLoadSettled = memorySnapshot();
}
docker('rm', '-f', name);

const outFile = path.join(outDir, `contenedor-${memory}${quick ? `-c${concurrency}` : ''}.json`);
if (quick) {
  fs.rmSync(dataDir, { recursive: true, force: true });
  fs.writeFileSync(outFile, `${JSON.stringify(report, null, 2)}\n`);
  console.log(JSON.stringify(report, null, 2));
  process.exit(0);
}

// 3. Reposo con el HEALTHCHECK de la imagen (un `node -e fetch` cada 30 s).
startedAt = runContainer({ healthcheck: true });
await waitUntilServing(startedAt);
await wait(15_000);
report.idleCpuWithHealthcheck = await idleCpu(120);
report.idleWithHealthcheck = memorySnapshot();
report.health = docker('inspect', '-f', '{{.State.Health.Status}}', name);
docker('rm', '-f', name);
fs.rmSync(dataDir, { recursive: true, force: true });

fs.writeFileSync(outFile, `${JSON.stringify(report, null, 2)}\n`);
console.log(JSON.stringify(report, null, 2));
