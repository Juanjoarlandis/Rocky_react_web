const LOGIN = '/api/shopify/account/login?returnPath=%2Fmi-crew';

async function startS(keep = false) {
  S = await start(3413, { ...storeEnv }, { https: true, httpsPort: 3414, keepObs: keep });
  return S;
}

async function seed(context, lines) {
  for (const [alias, quantity] of lines) {
    await context.request.post(`${S.origin}/api/shopify/cart/lines`, { headers: { Origin: S.origin }, data: { variantId: variantId(alias), quantity, operationId: opId() } });
  }
}

async function loginDirect(page) {
  const cbP = page.waitForResponse((r) => r.url().includes('/api/shopify/account/callback'), { timeout: 15_000 });
  await page.goto(`${S.origin}${LOGIN}`);
  const cb = await cbP;
  await page.waitForURL(`${S.origin}/mi-crew`, { timeout: 15_000 });
  await waitStore(page);
  return cb;
}

async function account(context) {
  const r = await context.request.get(`${S.origin}/api/shopify/account`);
  return { status: r.status(), json: await r.json().catch(() => null) };
}

async function crewProfile(context) {
  return (await (await context.request.get(`${S.origin}/api/shopify/account/crew`)).json()).profile;
}

const sessionCookie = async (context) => (await context.cookies()).find((c) => c.name === 'rocky_session')?.value || null;

async function accountEntry(page, vp) {
  if (vp === 'movil') {
    if (!(await page.locator('.navbar-sheet').count())) await page.locator('.navbar-menu').click();
    const el = page.locator('.navbar-sheet-account');
    await el.waitFor();
    return el;
  }
  return page.locator('.navbar-links .navbar-account');
}

async function entryInfo(page, vp) {
  const el = await accountEntry(page, vp);
  const info = await el.evaluate((a) => ({
    tag: a.tagName,
    href: a.getAttribute('href'),
    cls: a.className,
    aria: a.getAttribute('aria-label'),
    title: a.getAttribute('title'),
    text: a.textContent.trim(),
    imgAlt: a.querySelector('img')?.getAttribute('alt') ?? null,
    imgSrc: a.querySelector('img')?.getAttribute('src')?.split('/').pop() || null,
    after: getComputedStyle(a, '::after').content,
    html: a.outerHTML,
  }));
  if (vp === 'movil') await page.locator('.navbar-menu').click();
  return info;
}

async function pageText(page) {
  return nb(await page.locator('body').innerText());
}

async function main() {
  browser = await sim.launchE2Browser(chromium);
  // R-ACC-03: instancia HTTP (PUBLIC_ORIGIN http://localhost:3413)
  if (want('R-ACC-03')) {
    const H = await start(3413);
    for (const vp of VPS) {
      await R.run('R-ACC-03', vp, async () => {
        const status = await api(H, 'GET', '/api/shopify/status');
        const { context, page, issues, routes } = await openPage(browser, H, { viewportName: vp });
        await page.goto(`${H.origin}/`);
        await waitStore(page);
        let entry;
        if (vp === 'movil') {
          await page.locator('.navbar-menu').click();
          entry = await page.locator('.navbar-sheet-account').evaluate((a) => ({ tag: a.tagName, href: a.getAttribute('href'), text: a.textContent.trim() }));
        } else entry = await page.locator('.navbar-links .navbar-account').evaluate((a) => ({ tag: a.tagName, href: a.getAttribute('href'), text: a.textContent.trim() }));
        const login = await api(H, 'GET', LOGIN, { origin: null });
        await context.close();
        const checks = [
          check(status.json.capabilities.customerAccounts === false && status.json.capabilities.cart === true, `status: ${status.text} (PUBLIC_ORIGIN ${H.origin}, client ID y almacén presentes)`),
          check(entry.href === '/mi-crew' && entry.text === 'Mi Crew', `«Mi Crew» → ${entry.href}`),
          check(login.status === 503 && login.text === '{"message":"Las cuentas de cliente no están configuradas."}' && !login.setCookie, `login → ${login.status} ${login.text}`),
        ];
        return verdict(checks, { notes: [sumIssues(issues, routes)] });
      });
    }
    extra.unexpected.acc03 = H.unexpectedCalls();
    await stopWait(H);
  }
  for (const vp of VPS) await pass(vp);
  await browser.close();
}

try {
  await main();
} catch (error) {
  process.stdout.write(`ERROR ${error?.stack || error}\n`);
} finally {
  if (S) {
    extra.unexpected.final = S.unexpectedCalls();
    await S.stop().catch(() => {});
  }
  if (browser) await browser.close().catch(() => {});
  saveJson('cuenta-extra', extra);
  R.save();
  process.stdout.write(`unexpected: ${JSON.stringify(Object.fromEntries(Object.entries(extra.unexpected).map(([k, v]) => [k, v.length])))}\n`);
}
