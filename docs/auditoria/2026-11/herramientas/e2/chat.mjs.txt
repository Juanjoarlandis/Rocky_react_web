async function startM(env = {}, name = 'chat', { https = true } = {}) {
  const store = persistentStore(name);
  M = await start(3415, { SHOPIFY_EXPOSE_QUANTITY: 'true', CHAT_RATE_LIMIT_WINDOW_MS: '600000', ...store, ...env }, https ? { https: true, httpsPort: 3416 } : {});
  M.store = store;
  return M;
}

async function openChat(page) {
  await page.goto(`${M.origin}/rockyIA`);
  await page.getByLabel(BOX).waitFor({ timeout: 15_000 });
  await waitStore(page);
  await sleep(200);
}

async function lastAi(page) {
  return page.evaluate(() => {
    const turns = [...document.querySelectorAll('.chat-turn.ai')];
    const t = turns.at(-1);
    if (!t) return null;
    return {
      text: t.querySelector('.chat-bubble')?.textContent.trim(),
      error: t.querySelector('.chat-message')?.classList.contains('error'),
      cards: [...t.querySelectorAll('.chat-product-card')].map((c) => ({
        title: c.querySelector('.chat-product-title')?.textContent.trim(),
        href: c.querySelector('.chat-product-title')?.getAttribute('href'),
        mediaLabel: c.querySelector('.chat-product-media')?.getAttribute('aria-label'),
        img: c.querySelector('.chat-product-media img')?.getAttribute('src')?.split('?')[0].split('/').pop() || null,
        drop: c.querySelector('.chat-product-drop')?.textContent.trim(),
        price: (c.querySelector('.chat-product-facts strong')?.textContent || '').replace(/ /g, ' ').trim() || null,
        stock: c.querySelector('.chat-product-facts span')?.textContent.trim(),
        select: c.querySelector('select')?.getAttribute('aria-label') || null,
        options: [...c.querySelectorAll('select option')].map((o) => `${o.textContent.trim()}${o.disabled ? '(deshab.)' : ''}`),
        selected: c.querySelector('select')?.selectedOptions[0]?.textContent.trim() || null,
        single: c.querySelector('.chat-product-single-variant')?.textContent.trim() || null,
        action: (c.querySelector('.chat-product-add')?.textContent || '').trim(),
        actionTag: c.querySelector('.chat-product-add')?.tagName,
        disabled: c.querySelector('button.chat-product-add')?.disabled ?? null,
      })),
    };
  });
}

async function send(page, text, { enter = false } = {}) {
  const box = page.getByLabel(BOX);
  const w = waitApi(page, '/api/chat', 'POST', 30_000);
  await box.fill(text);
  if (enter) await box.press('Enter');
  else await page.getByRole('button', { name: 'Enviar mensaje' }).click();
  const res = await w;
  const json = await res.json().catch(() => null);
  await page.waitForFunction(() => !document.querySelector('.chat-typing'), null, { timeout: 30_000 });
  await sleep(150);
  return { status: res.status(), json, headers: res.headers(), sc: await res.headerValue('set-cookie'), ai: await lastAi(page) };
}

const plain = (t) => !/[*_`#>]/.test(t || '') && !/^\s*[-•]\s/m.test(t || '') && (t || '').length <= 600;
const lastObs = () => M.obs('openrouter').at(-1);

async function clickQuick(page, label) {
  const w = waitApi(page, '/api/chat', 'POST', 30_000);
  await page.locator('.chat-quick-prompt', { hasText: label }).click();
  const focus = await focusDesc(page);
  const res = await w;
  await page.waitForFunction(() => !document.querySelector('.chat-typing'), null, { timeout: 30_000 });
  await sleep(150);
  return { status: res.status(), json: await res.json().catch(() => null), focus, ai: await lastAi(page) };
}

async function login(page) {
  const cb = page.waitForResponse((r) => r.url().includes('/api/shopify/account/callback'), { timeout: 15_000 });
  await page.goto(`${M.origin}/api/shopify/account/login?returnPath=%2FrockyIA`);
  await cb;
  await page.waitForURL(`${M.origin}/rockyIA`, { timeout: 15_000 });
  await page.getByLabel(BOX).waitFor();
  await waitStore(page);
}

        return verdict(checks, { notes: ['la plaza del cupo global ya estaba reservada (por código, orchestrator.mjs:270)'], hallazgos: ['C-CHAT-01'] });
      });
    }
  }

  // ------------------------------------------------------------ R-CHAT-17
  if (want('R-CHAT-17')) {
    await startM({ OPENROUTER_MODELS: 'e2sim/down-a:free' }, 'chat-17');
    await R.run('R-CHAT-17', 'escritorio', async () => {
      const p = await openPage(browser, M, { viewportName: 'escritorio' });
      await openChat(p.page);
      const r1 = await send(p.page, 'hola');
      const r2 = await send(p.page, 'Ver camisetas disponibles');
      await p.context.close();
      extra.unexpected.chat17 = M.unexpectedCalls();
      await stopWait(M);
      // Modelo que responde 200 sin texto
      await startM({}, 'chat-17b');
      M.setControl({ openrouter: { mode: 'empty' } });
      const e1 = await api(M, 'POST', '/api/chat', { body: { message: 'hola' } });
      const e2 = await api(M, 'POST', '/api/chat', { body: { message: 'Ver camisetas disponibles' } });
      M.resetControl();
      const invalidLog = /invalid_response/.test(M.logText());
      extra.unexpected.chat17b = M.unexpectedCalls();
      await stopWait(M);
      // Sólo se usan los tres primeros modelos
      await startM({ OPENROUTER_MODELS: 'a/uno:free,b/dos:free,c/tres:free,d/cuatro:free' }, 'chat-17c');
      await api(M, 'POST', '/api/chat', { body: { message: 'hola' } });
      const o = M.obs('openrouter').at(-1);
      extra.unexpected.chat17c = M.unexpectedCalls();
      await stopWait(M);
      M = null;
      // Arranques que deben fallar
      const tryStart = async (env, name) => {
        try {
          const x = await startM(env, name);
          await stopWait(x);
          M = null;
          return 'ARRANCÓ';
        } catch (error) {
          M = null;
          return String(error.message).match(/OPENROUTER_MODELS[^\n"]*/)?.[0] || String(error.message).slice(0, 200);
        }
      };
      const sEmpty = await tryStart({ OPENROUTER_MODELS: '' }, 'chat-17d');
      const sPaid = await tryStart({ OPENROUTER_MODELS: 'openai/gpt-4o' }, 'chat-17e');
      const checks = [
        check(r1.status === 502 && r1.json?.message === 'La IA no está disponible ahora mismo.' && r1.ai.text === SERVER_DOWN, `«hola» con el único modelo caído → ${r1.status} «${r1.json?.message}»; burbuja genérica`),
        check(r2.status === 502, `«Ver camisetas disponibles» → ${r2.status} aunque la frase fija no necesita al modelo`),
        check(e1.status === 502 && invalidLog && e2.status === 200 && e2.json?.message?.startsWith('He encontrado'), `modelo que responde 200 sin texto (openrouter.mode=empty) → «hola» ${e1.status} (log invalid_response=${invalidLog}); con intención de compra → ${e2.status} «${e2.json?.message?.slice(0, 40)}…»`),
        check(o && [o.model, ...(o.models || [])].length === 3, `con 4 modelos se envían ${[o?.model, ...(o?.models || [])].join(', ')}`),
        check(sEmpty.includes('OPENROUTER_MODELS necesita al menos un modelo gratuito.'), `OPENROUTER_MODELS vacío → no arranca: «${sEmpty}»`),
        check(sPaid.includes('OPENROUTER_MODELS sólo puede contener modelos gratuitos con :free u openrouter/free.'), `modelo sin :free → no arranca: «${sPaid}»`),
      ];
      return verdict(checks, { notes: ['«rocky/no-existe:free» se sustituyó por «e2sim/down-a:free»: el doble no conoce el catálogo de OpenRouter y respondería a cualquier nombre :free (limitación del doble); para el servidor cualquier no-2xx acaba en 502'], hallazgos: ['C-CHAT-05', 'C-CHAT-01'] });
    });
  }
}

async function main() {
  browser = await sim.launchE2Browser(chromium);
  for (const vp of VPS) await mainPass(vp);
  await specials();
  await browser.close();
}

try {
  await main();
} catch (error) {
  process.stdout.write(`ERROR ${error?.stack || error}\n`);
} finally {
  if (M) {
    extra.unexpected.final = M.unexpectedCalls();
    await M.stop().catch(() => {});
  }
  if (browser) await browser.close().catch(() => {});
  saveJson('chat-extra', extra);
  R.save();
  process.stdout.write(`unexpected: ${JSON.stringify(Object.fromEntries(Object.entries(extra.unexpected).map(([k, v]) => [k, v.length])))}\n`);
}
