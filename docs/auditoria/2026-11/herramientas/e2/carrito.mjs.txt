async function seed(context, lines) {
  const out = [];
  for (const [alias, quantity] of lines) {
    const res = await context.request.post(`${A.origin}/api/shopify/cart/lines`, {
      headers: { Origin: A.origin },
      data: { variantId: variantId(alias), quantity, operationId: opId() },
    });
    out.push(res.status());
  }
  return out;
}

async function gotoCart(page) {
  const p = waitApi(page, '/api/shopify/cart');
  await page.goto(`${A.origin}/cart`);
  await p;
  await sleep(350);
}

async function rows(page) {
  return page.evaluate(() =>
    [...document.querySelectorAll('.cart-item')].map((li) => ({
      title: li.querySelector('.cart-item-title')?.textContent.trim(),
      href: li.querySelector('.cart-item-title')?.getAttribute('href'),
      mediaHref: li.querySelector('.cart-item-media')?.getAttribute('href'),
      img: Boolean(li.querySelector('img.cart-item-image')),
      variant: li.querySelector('.cart-item-variant')?.textContent.trim() || null,
      price: (li.querySelector('.cart-item-price')?.textContent || '').replace(/ /g, ' ').trim(),
      qty: li.querySelector('.quantity-value')?.textContent.trim(),
      total: (li.querySelector('.cart-item-total')?.textContent || '').replace(/ /g, ' ').trim(),
      soldout: li.querySelector('.cart-item-soldout')?.textContent.trim() || null,
      group: li.querySelector('[role=group]')?.getAttribute('aria-label'),
      btns: [...li.querySelectorAll('button')].map((b) => b.getAttribute('aria-label')),
    }))
  );
}

// Observador del repartidor (montajes del nodo data-testid="cart-runner").
const RUNNER_SPY = () => {
  window.__runs = [];
  const tick = () => {
    const el = document.querySelector('[data-testid="cart-runner"]');
    const now = performance.now();
    if (el && el !== window.__cur) {
      if (window.__cur && !window.__runs.at(-1).end) window.__runs.at(-1).end = now;
      window.__cur = el;
      const mini = document.querySelector('.mini-player')?.getBoundingClientRect();
      window.__runs.push({
        start: now,
        h: el.style.height,
        w: el.style.width,
        aria: el.getAttribute('aria-hidden'),
        src: el.querySelector('img')?.getAttribute('src'),
        correY: el.style.getPropertyValue('--corre-y'),
        hastaX: el.style.getPropertyValue('--hasta-x'),
        mini: mini ? { left: mini.left, right: mini.right, bottom: mini.bottom } : null,
      });
    }
    if (!el && window.__cur) {
      window.__runs.at(-1).end = now;
      window.__cur = null;
    }
  };
  const obs = new MutationObserver(tick);
  document.addEventListener('DOMContentLoaded', () => obs.observe(document.body, { childList: true, subtree: true }));
};

      check(Object.entries(res).filter(([k]) => k.startsWith('q=')).every(([, s]) => s.startsWith('400 INVALID_QUANTITY')), `cantidades: ${Object.entries(res).filter(([k]) => k.startsWith('q=')).map(([k, s]) => `${k}→${s}`).join('; ')}`),
      check(res['variant=123'].startsWith('400 INVALID_VARIANT «La variante no es válida.»') && res['variant=gid://shopify/Product/1'].startsWith('400 INVALID_VARIANT'), `variantes: ${res['variant=123']}; ${res['variant=gid://shopify/Product/1']}`),
      check(res['op=abc'].startsWith('400 INVALID_OPERATION_ID «operationId no válido.»') && res['op=op id con espacios'].startsWith('400 INVALID_OPERATION_ID'), `operationId: ${res['op=abc']}; ${res['op=op id con espacios']}`),
      check(res['PATCH linea-1'].startsWith('400 INVALID_LINE «La línea del carrito no es válida.»') && res['DELETE linea-1'].startsWith('400 INVALID_LINE'), `lineId: ${res['PATCH linea-1']}; ${res['DELETE linea-1']}`),
      check(res['>16kB'] === '413 {"message":"Petición demasiado grande."}', `cuerpo > 16 kB → ${res['>16kB']}`),
      check(!anyCookie && callsAfterErrors === callsBefore, `ningún error trae Set-Cookie (${anyCookie}) ni llama a Shopify (${callsAfterErrors - callsBefore} llamadas)`),
      check(five.status === 200 && five.json.cart.lines[0].quantity === 5, `quantity "5" (texto) → ${five.status}, cantidad ${five.json?.cart?.lines?.[0]?.quantity}`),
      check(extraFields.status === 200 && added.includes('dwag/XLx1') && !added.includes('smoker'), `campos de más (price, merchandiseId) → ${extraFields.status}; líneas: ${added}`),
    ];
    return verdict(checks);
  });

  if (want('R-CART-20')) await R.run('R-CART-20', 'api', async () => {
    const body1 = { variantId: variantId('flora-blue/S'), quantity: 1, operationId: 'audit-replay-0001' };
    const r1 = await api(A, 'POST', '/api/shopify/cart/lines', { body: body1 });
    const cookie = cookieFrom(r1.setCookie);
    const r2 = await api(A, 'POST', '/api/shopify/cart/lines', { cookie, body: body1 });
    const cartAfter2 = await api(A, 'GET', '/api/shopify/cart', { cookie });
    const r3a = await api(A, 'POST', '/api/shopify/cart/lines', { cookie, body: { variantId: 'gid://shopify/ProductVariant/999999999999', quantity: 1, operationId: 'audit-replay-0002' } });
    const r3b = await api(A, 'POST', '/api/shopify/cart/lines', { cookie, body: { variantId: variantId('danger/M'), quantity: 1, operationId: 'audit-replay-0002' } });
    A.setControl({ storefront: { failNext: [{ op: 'RockyCartLinesAdd', mode: 'network', times: 1 }] } });
    const r4a = await api(A, 'POST', '/api/shopify/cart/lines', { cookie, body: { variantId: variantId('danger/L'), quantity: 1, operationId: 'audit-replay-0003' } });
    A.resetControl();
    const r4b = await api(A, 'POST', '/api/shopify/cart/lines', { cookie, body: { variantId: variantId('danger/L'), quantity: 1, operationId: 'audit-replay-0003' } });
    const qty = cartAfter2.json.cart.lines.find((l) => l.variant.product.handle === 'flora-blue')?.quantity;
    const checks = [
      check(r1.status === 200 && r2.status === 200 && r2.text === r1.text && qty === 1, `repetición exacta → ${r2.status}, misma respuesta=${r2.text === r1.text}, cantidad ${qty}`),
      check(r3a.status === 422 && r3b.status === 200, `variante inexistente → ${r3a.status} ${r3a.text}; corregida con el mismo id → ${r3b.status}`),
      check(r4a.status === 502 && r4b.status === 409 && r4b.text === '{"message":"La operación anterior sigue pendiente.","code":"OPERATION_PENDING"}', `fallo de red → ${r4a.status} ${r4a.json?.code}; reintento con el mismo id → ${r4b.status} ${r4b.text}`),
    ];
    return verdict(checks, { notes: ['la web nunca reutiliza un operationId (UUID por llamada, src/api/shopify.js:27-31)'] });
  });

  extra.unexpected = A.unexpectedCalls();
  extra.calls = A.calls().length;
  await browser.close();
  await stopWait(A);
  A = null;

  // ---------------------------------------------------------------- R-CHK-04 paso 2 (arranque)
  if (want('R-CHK-04')) {
    let startErr = '';
    try {
      const bad = await start(3411, { SHOPIFY_CHECKOUT_HOSTS: 'checkout.example/ruta' });
      startErr = 'ARRANCÓ (no esperado)';
      await stopWait(bad);
    } catch (error) {
      startErr = String(error.message).match(/SHOPIFY_CHECKOUT_HOSTS[^\n"]*/)?.[0] || String(error.message).slice(0, 300);
    }
    for (const vp of VPS) {
      await R.run('R-CHK-04', vp, async () => {
        const e = extra[`chk04-${vp}`];
        const checks = [
          check(null, 'paso 1 (dominio propio no permitido): no reproducible con el doble, cuyo checkoutUrl es siempre https://rocky-dev.myshopify.com y ese host se permite siempre; revisado por código: cart-operations.mjs:7-18 lanza ShopifyGraphqlError «El dominio de checkout no está permitido.» (502 SHOPIFY_ERROR)'),
          check(startErr.includes('SHOPIFY_CHECKOUT_HOSTS sólo puede contener hostnames sin ruta ni puerto.'), `paso 2: con SHOPIFY_CHECKOUT_HOSTS=checkout.example/ruta el servidor no arranca: «${startErr}»`),
          check(e?.msgs?.http === 'Shopify ha devuelto un enlace de pago no seguro.' && e?.msgs?.invalida === 'Shopify no ha devuelto un enlace de pago válido.' && e.url.endsWith('/cart'), `cliente (respuesta del BFF sustituida en el navegador): http: → «${e?.msgs?.http}»; URL no válida → «${e?.msgs?.invalida}»; sin navegación`),
        ];
        return verdict(checks, { notes: ['paso 3: tests revisados en el informe (grep)', e?.issues || ''], hallazgos: ['C-CART-27'] });
      });
    }
  }
}

try {
  await main();
} catch (error) {
  process.stdout.write(`ERROR ${error?.stack || error}\n`);
} finally {
  if (A) {
    extra.unexpectedFinal = A.unexpectedCalls().length;
    await A.stop().catch(() => {});
  }
  if (browser) await browser.close().catch(() => {});

}
