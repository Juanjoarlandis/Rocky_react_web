import { startInstance, launchBrowser } from '../harness.mjs';
import { guardedPage, save } from './common.mjs';
const inst = await startInstance(3203, {});
const b = await launchBrowser();
const out = {};
const pos = (page, id) => page.evaluate((cid) => { const r = document.getElementById(`cromo-${cid}`).getBoundingClientRect(); return { offBy: Math.round(Math.abs(r.top + r.height / 2 - innerHeight / 2)), scrollY: Math.round(scrollY) }; }, id);
try {
  for (const vp of ['escritorio', 'movil']) {
    const { page, context } = await guardedPage(b, vp, { origin: inst.origin });
    await page.goto(inst.origin + '/crew#rocky');
    await page.waitForTimeout(2500);
    const o = { rocky: await pos(page, 'rocky') };
    await page.evaluate(() => { location.hash = 'ollie'; });
    await page.waitForTimeout(3500);
    o.ollie = await pos(page, 'ollie');
    await page.goBack();
    o.backAt1200 = (await page.waitForTimeout(1200), await pos(page, 'rocky'));
    await page.waitForTimeout(3000);
    o.backAt4200 = await pos(page, 'rocky');
    await page.goForward();
    await page.waitForTimeout(4200);
    o.fwdAt4200 = await pos(page, 'ollie');
    o.flipped = await page.evaluate(() => [...document.querySelectorAll('.crew-card.is-flipped')].map((c) => c.id));
    out[vp] = o;
    await context.close();
  }
} finally { await b.close(); inst.stop(); }
save('probe-crew04', out);
console.log(JSON.stringify(out));
