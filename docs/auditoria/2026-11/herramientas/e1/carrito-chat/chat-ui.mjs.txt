function trackChat(page) {
  const posts = [];
  page.on('request', (r) => {
    if (r.url().endsWith('/api/chat') && r.method() === 'POST') posts.push(r.postData());
  });
  const responses = [];
  page.on('response', async (r) => {
    if (r.url().endsWith('/api/chat')) {
      responses.push({ status: r.status(), body: await r.text().catch(() => ''), setCookie: (await r.headerValue('set-cookie')) || null });
    }
  });
  return { posts, responses };
}

async function delayChat(page, ms) {
  await page.route('**/api/chat', async (route) => {
    await new Promise((r) => setTimeout(r, ms));
    await route.continue();
  });
}

async function chat03(vp) {
  const o = {};
  const { page, issues, context } = await guardedPage(browser, vp, { origin: W });
  await context.grantPermissions(['clipboard-read', 'clipboard-write'], { origin: W });
  const t = trackChat(page);
  await page.goto(`${W}/rockyIA`);
  await page.waitForSelector('.chat-welcome');
  const ta = page.getByRole('textbox', { name: 'Mensaje para Rocky IA' });
  const send = page.locator('.chat-send');
  // 1. Tab hasta el campo
  const seq = [];
  for (let i = 0; i < 30; i += 1) {
    await page.keyboard.press('Tab');
    const f = await focusInfo(page);
    seq.push(f.label || f.tag);
    if (f.tag === 'TEXTAREA') { o.textareaFocus = f; break; }
  }
  o.tabsToTextarea = seq.length;
  o.tabSeq = seq;
  // 2. vacío + Enter
  await page.keyboard.press('Enter');
  await page.waitForTimeout(300);
  o.emptyEnter = { posts: t.posts.length, sendDisabled: await send.isDisabled() };
  // 3. espacios + Enter
  await page.keyboard.type('    ');
  o.spacesSendDisabled = await send.isDisabled();
  await page.keyboard.press('Enter');
  await page.waitForTimeout(300);
  o.spacesEnter = { posts: t.posts.length, value: JSON.stringify(await ta.inputValue()) };
  await ta.fill('');
  // 4. línea, Shift+Enter, línea, Enter
  await page.keyboard.type('primera línea');
  await page.keyboard.press('Shift+Enter');
  await page.keyboard.type('segunda línea  ');
  o.multilineValue = JSON.stringify(await ta.inputValue());
  await page.keyboard.press('Enter');
  await page.waitForSelector('.chat-message.ai.error', { timeout: 8000 });
  o.multiline = {
    posts: t.posts.slice(-1),
    bubble: JSON.stringify(await page.locator('.chat-message.user .chat-bubble').last().innerText()),
    whiteSpace: await page.locator('.chat-message.user .chat-bubble').last().evaluate((e) => getComputedStyle(e).whiteSpace),
    focusAfter: (await focusInfo(page)).tag,
    valueAfter: await ta.inputValue(),
  };
  // 5. pegar 1.200 caracteres
  const long = 'x'.repeat(1200);
  await page.evaluate((s) => navigator.clipboard.writeText(s), long);
  await ta.focus();
  await page.keyboard.press('ControlOrMeta+V');
  await page.waitForTimeout(200);
  o.pasteLength = (await ta.inputValue()).length;
  o.maxLength = await ta.getAttribute('maxlength');
  await ta.fill('');
  // 6. Enter durante composición IME (evento sintético con isComposing)
  await ta.fill('こんにちは');
  const before = t.posts.length;
  await ta.evaluate((el) => el.dispatchEvent(new KeyboardEvent('keydown', { key: 'Enter', code: 'Enter', keyCode: 229, isComposing: true, bubbles: true, cancelable: true })));
  await page.waitForTimeout(300);
  o.imeEnter = { sent: t.posts.length - before, value: await ta.inputValue() };
  await ta.fill('');
  // 7. con el indicador visible, escribir y Enter
  await delayChat(page, 2500);
  await ta.fill('mensaje lento');
  await page.keyboard.press('Enter');
  await page.waitForTimeout(300);
  o.typingVisible = await page.locator('.chat-typing').count();
  const beforeBusy = t.posts.length;
  await page.keyboard.type('otro texto');
  await page.keyboard.press('Enter');
  await page.waitForTimeout(300);
  o.busyEnter = {
    sentWhileBusy: t.posts.length - beforeBusy,
    valueKept: await ta.inputValue(),
    textareaEditable: await ta.isEditable(),
    sendDisabled: await send.isDisabled(),
  };
  await page.waitForSelector('.chat-typing', { state: 'detached', timeout: 8000 });
  await page.unroute('**/api/chat');
  // 8. móvil: Enter sin Shift envía
  await ta.fill('desde el teclado del móvil');
  const b8 = t.posts.length;
  await page.keyboard.press('Enter');
  await page.waitForTimeout(500);
  o.enterSends = t.posts.length - b8;
  o.posts = t.posts.length;
  o.issues = summarize(issues);
  await context.close();
  return o;
}

async function chat04(vp) {
  const o = {};
  const { page, issues, context } = await guardedPage(browser, vp, { origin: W });
  await delayChat(page, 1200);
  await page.goto(`${W}/rockyIA`);
  await page.waitForSelector('.chat-welcome');
  o.region = await page.locator('section.chat-card').evaluate((e) => ({ tag: e.tagName, label: e.getAttribute('aria-label') }));
  o.headings = await page.locator('h1, h2').evaluateAll((hs) => hs.map((h) => `${h.tagName}:${h.textContent.trim()}`));
  o.log = await page.locator('.chat-messages').evaluate((e) => ({ role: e.getAttribute('role'), live: e.getAttribute('aria-live'), busy: e.getAttribute('aria-busy') }));
  o.sendName = await page.locator('.chat-send').getAttribute('aria-label');
  o.ariaMain = await page.locator('section.chat-card').ariaSnapshot();
  const ta = page.getByRole('textbox', { name: 'Mensaje para Rocky IA' });
  await ta.fill('hola');
  await page.keyboard.press('Enter');
  await page.waitForTimeout(300);
  o.busyDuring = await page.locator('.chat-messages').getAttribute('aria-busy');
  o.typing = await page.locator('.chat-typing').evaluate((e) => ({ tag: e.tagName, role: e.getAttribute('role'), label: e.getAttribute('aria-label') }));
  o.searching = await page.locator('.chat-searching').textContent();
  o.ariaLogDuring = await page.locator('.chat-messages').ariaSnapshot();
  await page.waitForSelector('.chat-message.ai.error', { timeout: 8000 });
  o.busyAfter = await page.locator('.chat-messages').getAttribute('aria-busy');
  o.avatars = await page.locator('.chat-messages img.chat-avatar').evaluateAll((is) => is.map((i) => i.getAttribute('alt')));
  o.hiddenSpeaker = await page.locator('.chat-messages .sr-only, .chat-messages .visually-hidden').count();
  o.ariaLogAfter = await page.locator('.chat-messages').ariaSnapshot();
  // 3. recargar, Tab a la pregunta rápida y Enter
  await page.unroute('**/api/chat');
  await page.reload();
  await page.waitForSelector('.chat-welcome');
  let reached = false;
  for (let i = 0; i < 30; i += 1) {
    await page.keyboard.press('Tab');
    const f = await focusInfo(page);
    if ((f.label || '').includes('Ver camisetas disponibles')) { reached = true; o.quickFocus = f; break; }
  }
  o.quickReached = reached;
  await page.keyboard.press('Enter');
  await page.waitForSelector('.chat-message.ai.error', { timeout: 8000 });
  await page.waitForTimeout(300);
  o.focusAfterQuick = await page.evaluate(() => ({ tag: document.activeElement?.tagName, isBody: document.activeElement === document.body }));
  // ¿Adónde va el siguiente Tab?
  await page.keyboard.press('Tab');
  o.nextTabAfterQuick = await focusInfo(page);
  o.issues = summarize(issues);
  await context.close();
  return o;
}
