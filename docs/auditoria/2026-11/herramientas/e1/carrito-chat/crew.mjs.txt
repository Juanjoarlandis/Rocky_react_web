const albumState = (page) =>

async function goCrewFromMenu(page, vp, label = 'Crew') {
  await page.goto(`${W}/`);
  await page.waitForTimeout(1000);
  if (vp === 'escritorio') {
    await page.locator('.navbar-links a', { hasText: new RegExp(`^${label}$`) }).click();
  } else {
    await page.getByRole('button', { name: 'Abrir el menú' }).tap();
    await page.waitForTimeout(300);
    await page.locator('.navbar-sheet a', { hasText: new RegExp(`^${label}$`) }).tap();
  }
  await page.waitForTimeout(1200);
}

async function crew04(vp) {
  const o = {};
  const centered = (page, id) =>

      const r = document.getElementById(`cromo-${cid}`).getBoundingClientRect();
      return { center: Math.round(r.top + r.height / 2), vh: innerHeight, offBy: Math.round(Math.abs(r.top + r.height / 2 - innerHeight / 2)), scrollY: Math.round(scrollY) };
    }, id);

    await page.waitForTimeout(250);
    o.rockyAt250 = await centered(page, 'rocky');
    await page.waitForTimeout(1500);
    o.rocky = { ...(await albumState(page)), pos: await centered(page, 'rocky') };
    // 4. cambiar el hash estando en /crew
    await page.evaluate(() => { window.location.hash = 'ollie'; });
    await page.waitForTimeout(1600);
    o.hashChange = { ...(await albumState(page)), pos: await centered(page, 'ollie'), url: page.url() };
    await page.goBack();
    await page.waitForTimeout(1200);
    o.back = { url: page.url(), ...(await albumState(page)), pos: await centered(page, 'rocky') };
    await page.goForward();
    await page.waitForTimeout(1200);
    o.forward = { url: page.url(), ...(await albumState(page)), pos: await centered(page, 'ollie') };
    o.issues = summarize(issues);
    await context.close();
  }
  for (const h of ['no-existe', '%E0%A4%A']) {

    await page.goto(`${W}/crew#${h}`);

  }
  {
    const { page, issues, context } = await guardedPage(browser, vp, { origin: W, reducedMotion: 'reduce' });
    await page.goto(`${W}/crew#ollie`);
    await page.waitForTimeout(420);

    o.reducedFlipped = await albumState(page);
