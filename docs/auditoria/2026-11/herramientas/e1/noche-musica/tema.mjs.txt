function analyseFlash(samples) {
  // samples: [t, opacidadCapa, opacidadRótulo]; el rótulo va dentro de la capa: su opacidad efectiva es el producto.
  const pres = samples.filter((x) => x[2] !== null);
  if (!pres.length) return { visible: false };
  const t0 = pres[0][0];
  const series = pres.map(([t, o1, o2]) => ({ t: t - t0, o: o1 * o2, capa: o1 }));
  const ext = [series[0]];
  let trend = 0;
  let cand = series[0];
  for (const p of series) {
    if (trend === 0) {
      if (p.o - ext[0].o >= 0.1) { trend = 1; cand = p; } else if (ext[0].o - p.o >= 0.1) { trend = -1; cand = p; }
    } else if (trend === 1) {
      if (p.o > cand.o) cand = p; else if (cand.o - p.o >= 0.1) { ext.push(cand); trend = -1; cand = p; }
    } else if (p.o < cand.o) cand = p; else if (p.o - cand.o >= 0.1) { ext.push(cand); trend = 1; cand = p; }
  }
  ext.push(cand);
  const changes = [];
  for (let i = 1; i < ext.length; i += 1) if (Math.abs(ext[i].o - ext[i - 1].o) >= 0.1) changes.push({ t: ext[i].t, from: ext[i - 1].o, to: ext[i].o });
  let maxFlashes = 0;
  for (const c of changes) {
    const inWin = changes.filter((x) => x.t >= c.t && x.t < c.t + 1000).length;
    maxFlashes = Math.max(maxFlashes, Math.floor(inWin / 2));
  }
  const capaFin = series.find((x) => x.capa < 0.99);
  return {
    visible: true,
    lastMs: series[series.length - 1].t,
    extremos: ext.map((e) => `${e.t}ms:${e.o.toFixed(2)}`).join(' '),
    cambios: changes.length,
    maxDestellosPorSegundo: maxFlashes,
    capaEmpiezaAFundir: capaFin?.t,
  };
}

async function installFlashSampler(page) {
  await page.evaluate(() => {
    window.__flash = [];
    window.__flashSeen = 0;
    const t0 = performance.now();
    const loop = () => {
      const f = document.querySelector('.descontrol-flash');
      const s = f?.querySelector('span');
      if (f) window.__flashSeen += 1;
      window.__flash.push([Math.round(performance.now() - t0), f ? Number(getComputedStyle(f).opacity) : null, s ? Number(getComputedStyle(s).opacity) : null]);
      if (performance.now() - t0 < 2600) requestAnimationFrame(loop);
    };
    requestAnimationFrame(loop);
    window.__flashAdded = [];
    window.__flashObs?.disconnect();
    window.__flashObs = new MutationObserver((muts) => {
      for (const m of muts) for (const n of m.addedNodes) if (n.classList?.contains('descontrol-flash')) {
        const cs = getComputedStyle(n);
        window.__flashAdded.push({ t: Math.round(performance.now() - t0), z: cs.zIndex, pe: cs.pointerEvents, ariaHidden: n.getAttribute('aria-hidden'), text: n.textContent, bg: cs.backgroundColor });
      }
    });
    window.__flashObs.observe(document.body, { childList: true, subtree: true });
  });
}

      evidencia: `con reduce: apagón añadido ${flashAdded} veces, escena pared=${st.wall} bote=${st.fab}; chorretón en ${spotA} (borde inferior en px de página a +70 ms y a +1070 ms del disparo): ${reducedDrip.a} → ${reducedDrip.b} (px ${reducedDrip.nA} → ${reducedDrip.nB}); radio ${audio.currentSrc} paused=${audio.paused} groove=${anim.groove}; animación .al-ritmo=${anim.alRitmo}, disco=${anim.disco}; transición color spray=${anim.fabTrans}, bote=${fabTrans}; radiocasete ${JSON.stringify(bb)}; quitando la emulación con la escena puesta: chorretón en ${spotB} ${liveDrip.a} → ${liveDrip.b} (crece), apagón al reencender=${flashAfter}; ${sum(issues)}`,
      hallazgos: 'C-NOCHE-25',
    });
    if (!ok) await snap(page, `R-NIGHT-04-${vp}`);
    await context.close();
  }

  // ---------------- R-NIGHT-05 (ambos) ----------------
  for (const vp of ['escritorio', 'movil']) {
    if (!want('R-NIGHT-05')) break;
    const { page, issues, context } = await openPage(browser, vp, { origin: inst.origin });
    await page.goto(`${inst.origin}/crew`);
    await page.waitForTimeout(1500);
    await nightOn(page, vp);
    if (vp === 'escritorio') {
      await page.locator('.spray-fab').click();
      await mouseStroke(page, [[40, 300], [60, 420], [50, 520]]);
    } else {
      await page.locator('.spray-fab').tap();
      await touchDrag(page, line(12, 300, 20, 480, 14));
    }
    await page.waitForTimeout(500);
    await page.locator('.spray-tool--close').evaluate((b) => b.click());
    const audioBefore = await audioState(page);
    await page.evaluate(() => localStorage.setItem('rocky-theme', 'neon'));
    await page.reload();
    await page.waitForTimeout(1500);
    const st = await themeState(page);
    const audio = await audioState(page);
    const legacy = await page.evaluate(() => localStorage.getItem('rocky-theme'));
    const url = page.url();
    await installFlashSampler(page);
    await toggleNight(page, vp);
    await page.waitForTimeout(200);
    const flash = await page.evaluate(() => window.__flashAdded.length);
    await page.waitForTimeout(1500);
    const ink = await canvasInk(page, { x: 0, y: 280, w: 90, h: 260 });
    // Pestaña nueva.
    const p2 = await context.newPage();
    await p2.goto(`${inst.origin}/crew`);
    await p2.waitForTimeout(1500);
    const st2 = await themeState(p2);
    // data-theme desde la consola (borde documentado).
    await p2.evaluate(() => { document.documentElement.dataset.theme = 'neon'; });
    await p2.waitForTimeout(400);
    const forced = await themeState(p2);
    const forcedAudio = await audioState(p2);
    const ok = st.theme === null && !st.fab && audio.src === null && audio.paused === true && audio.pillBtn === 'Reproducir BARRO' && legacy === null && flash === 1 && ink?.n > 0 && st2.theme === null;
    C.add({
      id: 'R-NIGHT-05', viewport: vp, resultado: ok ? 'OK' : 'FALLO',
      evidencia: `antes de recargar: radio ${audioBefore.currentSrc} paused=${audioBefore.paused}; tras F5 (${url}): theme=${st.theme} meta=${st.meta} bote=${st.fab}; radio src=${audio.src} paused=${audio.paused} pill=${audio.pillBtn}; localStorage['rocky-theme'] (puesto a 'neon' antes de recargar)=${legacy}; reactivar: apagón=${flash}, pintada repintada px=${ink?.n}; pestaña nueva: theme=${st2.theme}; con data-theme forzado desde la consola: escena pared=${forced.wall} bote=${forced.fab}, bombilla ${forced.bulbs[0].label}/aria-pressed=${forced.bulbs[0].pressed}, radio src=${forcedAudio.src} pill=${forcedAudio.pillBtn} (desincronizada); ${sum(issues)}`,
      hallazgos: '',
    });
    if (!ok) await snap(page, `R-NIGHT-05-${vp}`);
    await context.close();
  }

  // ---------------- R-NIGHT-12 (ambos; visibilidad simulada) ----------------
  for (const vp of ['escritorio', 'movil']) {
    if (!want('R-NIGHT-12')) break;
    const { page, issues, context } = await openPage(browser, vp, { origin: inst.origin });
    await page.goto(`${inst.origin}/crew`);
    await page.waitForTimeout(1500);
    const setHidden = (hidden) =>
      page.evaluate((h) => {
        Object.defineProperty(document, 'hidden', { configurable: true, get: () => h });
        Object.defineProperty(document, 'visibilityState', { configurable: true, get: () => (h ? 'hidden' : 'visible') });
        document.dispatchEvent(new Event('visibilitychange'));
      }, hidden);
    await nightOn(page, vp);
    const t0 = await page.title();
    await setHidden(true);
    const tHidden = await page.title();
    await setHidden(false);
    const tBack = await page.title();
    await setHidden(true);
    const tHidden2 = await page.title();
    await toggleNight(page, vp); // apagar con la pestaña «oculta»
    await page.waitForTimeout(200);
    const tOff = await page.title();
    await setHidden(false);
    await setHidden(true);
    const tDay = await page.title();
    await setHidden(false);
    const ok = tHidden === 'Te estamos viendo…' && tBack === t0 && tHidden2 === 'Te estamos viendo…' && tOff === t0 && tDay === t0;
    C.add({
      id: 'R-NIGHT-12', viewport: vp, resultado: ok ? 'OK' : 'FALLO',
      evidencia: `visibilidad simulada (document.hidden + visibilitychange; Chrome headless no oculta pestañas con bringToFront): título «${t0}» → oculto «${tHidden}» → visible «${tBack}»; oculto otra vez «${tHidden2}» y apagar el descontrol → «${tOff}»; de día oculto → «${tDay}»; ${sum(issues)}`,
      hallazgos: '',
    });
    if (!ok) await snap(page, `R-NIGHT-12-${vp}`);
    await context.close();
  }

  // ---------------- Capturas de día y noche ----------------
  if (want('capturas')) {
    for (const vp of ['escritorio', 'movil']) {
      const { page, context } = await openPage(browser, vp, { origin: inst.origin });
      await page.goto(`${inst.origin}/`);
      await page.waitForTimeout(2500);
      await snap(page, `tema-portada-dia-${vp}`);
      await nightOn(page, vp);
      await page.mouse.move(vp === 'escritorio' ? 720 : 187, 380);
      await page.waitForTimeout(800);
      await snap(page, `tema-portada-noche-${vp}`);
      await navTo(page, vp, 'Estudio');
      await page.waitForTimeout(1500);
      await snap(page, `tema-estudio-noche-${vp}`);
      await page.goto(`${inst.origin}/estudio`);
      await page.waitForTimeout(2500);
      await snap(page, `tema-estudio-dia-${vp}`);
      await context.close();
    }
    console.log('capturas hechas');
  }
} finally {
  await browser.close();
  inst.stop();
}
