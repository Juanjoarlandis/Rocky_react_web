const strokesOf = async (page, route) => (await sprayStorage(page))?.pages?.[route]?.strokes ?? [];
const pick = async (page, vp) => {
  if (vp === 'escritorio') await page.locator('.spray-fab').click();

const tapOrClick = async (page, vp, locator) => {
  if (vp === 'escritorio') await locator.click();

};
// Un trazo de ~60 px en un punto libre del margen.
async function paint(page, vp, spot, { dy = 60, dx = 6 } = {}) {
  if (vp === 'escritorio') await mouseStroke(page, [spot, [spot[0] + dx, spot[1] + dy / 2], [spot[0], spot[1] + dy]]);
  else {
    await touchDrag(page, line(spot[0], spot[1], spot[0] + dx, spot[1] + dy, 12));
    await page.waitForTimeout(650); // tras un trazo táctil sintético, el siguiente toque necesita respiro
  }
}

const spotNear = (page, vp, ys) => emptySpot(page, margins(vp).flatMap((x) => ys.map((y) => [x, y])));

function pngSize(file) {
  const b = fs.readFileSync(file);
  return { w: b.readUInt32BE(16), h: b.readUInt32BE(20), bytes: b.length };
}
