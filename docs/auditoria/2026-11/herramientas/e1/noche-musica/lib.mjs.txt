// Colector de resultados por script: se vuelcan a out/<nombre>.json y el
// script consolidar.mjs los escribe en el TSV del área con record().
export function collector(name) {
  const file = path.join(OUT, `${name}.json`);
  const rows = fs.existsSync(file) ? JSON.parse(fs.readFileSync(file, 'utf8')) : [];
  return {
    rows,
    add(row) {
      const i = rows.findIndex((r) => r.id === row.id && r.viewport === row.viewport);
      if (i >= 0) rows[i] = row;
      else rows.push(row);
      console.log(`[${row.id} · ${row.viewport}] ${row.resultado} — ${String(row.evidencia).slice(0, 400)}`);
      fs.writeFileSync(file, JSON.stringify(rows, null, 2));
    },
  };
}

// --- Tema -------------------------------------------------------------------

export async function themeState(page) {
  return page.evaluate(() => {
    const bulbs = [...document.querySelectorAll('button.navbar-theme')].map((b) => ({
      sheet: b.classList.contains('navbar-theme--sheet'),
      label: b.getAttribute('aria-label'),
      title: b.getAttribute('title'),
      pressed: b.getAttribute('aria-pressed'),
      on: b.classList.contains('navbar-theme--on'),
      visible: !!(b.offsetWidth || b.offsetHeight),
      color: getComputedStyle(b).color,
    }));
    return {
      theme: document.documentElement.dataset.theme || null,
      meta: document.querySelector('meta[name="theme-color"]')?.getAttribute('content'),
      flash: !!document.querySelector('.descontrol-flash'),
      wall: !!document.querySelector('.uv-wall'),
      dark: !!document.querySelector('.uv-oscuridad'),
      canvas: !!document.querySelector('canvas.spray-wall'),
      fab: !!document.querySelector('.spray-fab'),
      tools: !!document.querySelector('.spray-tools'),
      spraying: document.documentElement.classList.contains('is-spraying'),
      uvFondo: document.body.classList.contains('uv-fondo'),
      bulbs,
    };
  });
}

// Enciende o apaga el descontrol como lo haría una persona en ese viewport.
export async function toggleNight(page, vp) {
  if (vp === 'escritorio') {
    await page.locator('header button.navbar-theme').click();
    return;
  }
  const menu = page.locator('button.navbar-menu');
  if ((await menu.getAttribute('aria-expanded')) !== 'true') await menu.tap();
  await page.locator('.navbar-sheet button.navbar-theme--sheet').tap();
  await page.locator('button.navbar-menu').tap(); // cerrar la hoja
}

export async function nightOn(page, vp, { waitFlash = true } = {}) {
  const st = await themeState(page);
  if (st.theme !== 'neon') await toggleNight(page, vp);
  if (waitFlash) await page.waitForFunction(() => !document.querySelector('.descontrol-flash'), null, { timeout: 4000 }).catch(() => {});
}

// Navegación SPA (sin recargar: recargar devuelve el día).
export async function spaGo(page, pathname) {
  await page.evaluate((p) => {
    window.history.pushState({}, '', p);
    window.dispatchEvent(new PopStateEvent('popstate', { state: {} }));
  }, pathname);
  await page.waitForTimeout(400);
  await page.waitForFunction(() => !document.querySelector('.page-loading'), null, { timeout: 8000 }).catch(() => {});
  await page.waitForTimeout(300);
}

// Pulsa un destino de la barra (en móvil, desde la hoja).
export async function navTo(page, vp, label) {
  if (vp === 'escritorio') {
    await page.locator('header .navbar-links a', { hasText: label }).first().click();
  } else {
    const menu = page.locator('button.navbar-menu');
    if ((await menu.getAttribute('aria-expanded')) !== 'true') await menu.tap();
    await page.locator('.navbar-sheet a', { hasText: label }).first().tap();
  }
  await page.waitForTimeout(400);
  await page.waitForFunction(() => !document.querySelector('.page-loading'), null, { timeout: 8000 }).catch(() => {});
  await page.waitForTimeout(300);
}

      bpm: document.body.style.getPropertyValue('--bpm') || null,

    };
  });
}

export function trackMusic(page) {
  const reqs = [];
  page.on('request', (r) => {
    if (r.url().includes('/music/')) reqs.push({ t: Date.now(), url: r.url().replace(/^https?:\/\/[^/]+/, ''), range: r.headers().range || '' });
  });
  const resps = [];
  page.on('response', async (r) => {
    if (r.url().includes('/music/')) {
      const h = r.headers();
      resps.push({ url: r.url().replace(/^https?:\/\/[^/]+/, ''), status: r.status(), type: h['content-type'], length: h['content-length'], range: h['content-range'] || '', cache: h['cache-control'] || '', accept: h['accept-ranges'] || '', enc: h['content-encoding'] || '' });
    }
  });
  return { reqs, resps };
}

// --- Táctil (CDP) ----------------------------------------------------------

const sessions = new WeakMap();
export async function cdp(page) {
  if (!sessions.has(page)) sessions.set(page, await page.context().newCDPSession(page));
  return sessions.get(page);
}

// Arrastre con uno o dos dedos. `points` son [x, y] del primer dedo; con
// `second` se añade un segundo dedo fijo en esa posición desde `secondFrom`.
export async function touchDrag(page, points, { second = null, secondFrom = 1, cancel = false, stepDelay = 16 } = {}) {
  const s = await cdp(page);
  const tp = (i) => {
    const list = [{ x: points[i][0], y: points[i][1], id: 1 }];
    if (second && i >= secondFrom) list.push({ x: second[0], y: second[1], id: 2 });
    return list;
  };
  await s.send('Input.dispatchTouchEvent', { type: 'touchStart', touchPoints: tp(0) });
  for (let i = 1; i < points.length; i += 1) {
    const type = second && i === secondFrom ? 'touchStart' : 'touchMove';
    await s.send('Input.dispatchTouchEvent', { type, touchPoints: tp(i) });
    await page.waitForTimeout(stepDelay);
  }
  if (cancel) await s.send('Input.dispatchTouchEvent', { type: 'touchCancel', touchPoints: [] });
  else await s.send('Input.dispatchTouchEvent', { type: 'touchEnd', touchPoints: [] });
}

export function line(x0, y0, x1, y1, n = 20) {
  return Array.from({ length: n + 1 }, (_, i) => [x0 + ((x1 - x0) * i) / n, y0 + ((y1 - y0) * i) / n]);
}

// Scroll táctil: arrastre de un dedo hacia arriba (dy > 0 baja la página).
export async function touchScroll(page, x, y, dy) {
  await touchDrag(page, line(x, y, x, y - dy, 20));
  await page.waitForTimeout(400);
}

// Píxeles pintados (alfa > 0) del lienzo en una caja en coordenadas CSS de página.
export async function canvasInk(page, box = null) {
  return page.evaluate((b) => {
    const c = document.querySelector('canvas.spray-wall');
    if (!c) return null;
    const sx = c.width / parseFloat(c.style.width || c.width);
    const sy = c.height / parseFloat(c.style.height || c.height);
    const r = b ? { x: Math.max(0, Math.floor(b.x * sx)), y: Math.max(0, Math.floor(b.y * sy)), w: Math.max(1, Math.floor(b.w * sx)), h: Math.max(1, Math.floor(b.h * sy)) } : { x: 0, y: 0, w: c.width, h: c.height };
    const ctx = c.getContext('2d');
    const data = ctx.getImageData(r.x, r.y, Math.min(r.w, c.width - r.x), Math.min(r.h, c.height - r.y)).data;
    let n = 0;
    let hash = 0;
    let minY = Infinity;
    let maxY = -Infinity;
    const w = Math.min(r.w, c.width - r.x);
    for (let i = 3; i < data.length; i += 4) {
      if (data[i] > 0) {
        n += 1;
        const row = Math.floor((i - 3) / 4 / w);
        if (row < minY) minY = row;
        if (row > maxY) maxY = row;
      }
      hash = (hash * 31 + data[i]) | 0;
    }
    return { n, hash, minY: n ? Math.round((minY + r.y) / sy) : null, maxY: n ? Math.round((maxY + r.y) / sy) : null, canvas: [c.width, c.height], css: [c.style.width, c.style.height] };
  }, box);
}

// Trazo con ratón entre puntos (coordenadas de viewport).
export async function mouseStroke(page, pts, { holdMs = 0, release = true, steps = 4 } = {}) {
  await page.mouse.move(pts[0][0], pts[0][1]);
  await page.mouse.down();
  for (let i = 1; i < pts.length; i += 1) await page.mouse.move(pts[i][0], pts[i][1], { steps });
  if (holdMs) await page.waitForTimeout(holdMs);
  if (release) await page.mouse.up();
}

// --- Contraste WCAG --------------------------------------------------------

export function parseRgb(s) {
  const m = String(s).match(/rgba?\(([^)]+)\)/);
  if (m) {
    const [r, g, b, a = 1] = m[1].split(/[ ,/]+/).filter(Boolean).map(Number);
    return { r, g, b, a };
  }
  const h = String(s).trim().replace('#', '');
  if (/^[0-9a-f]{6}$/i.test(h)) return { r: parseInt(h.slice(0, 2), 16), g: parseInt(h.slice(2, 4), 16), b: parseInt(h.slice(4, 6), 16), a: 1 };
  throw new Error(`Color no reconocido: ${s}`);
}
function lin(c) {
  const v = c / 255;
  return v <= 0.03928 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4;
}
export function lum({ r, g, b }) {
  return 0.2126 * lin(r) + 0.7152 * lin(g) + 0.0722 * lin(b);
}
export function ratio(a, b) {
  const [l1, l2] = [lum(a), lum(b)].sort((x, y) => y - x);
  return (l1 + 0.05) / (l2 + 0.05);
}
// Mezcla `top` (con alfa) sobre `bottom` (opaco).
export function over(top, bottom) {
  const a = top.a ?? 1;
  return { r: top.r * a + bottom.r * (1 - a), g: top.g * a + bottom.g * (1 - a), b: top.b * a + bottom.b * (1 - a), a: 1 };
}

// Busca un punto del viewport donde el spray pinta (nada interactivo encima).
export async function emptySpot(page, candidates) {
  return page.evaluate((cands) => {
    const INTERACTIVE = 'a, button, input, textarea, select, label, summary, iframe, [role="button"], [role="slider"], [contenteditable="true"], .navbar, .spray-tools, .spray-fab, .mini-player';
    for (const [x, y] of cands) {
      let ok = true;
      for (let dy = 0; dy <= 120 && ok; dy += 10) {
        const el = document.elementFromPoint(x, y + dy);
        if (!el || el.closest(INTERACTIVE)) ok = false;
      }
      if (ok) return [x, y];
    }
    return null;
  }, candidates);
}
