// Bytes de audio recibidos por petición (CDP).
async function netBytes(page) {
  const s = await cdp(page);
  await s.send('Network.enable');
  const reqs = new Map();
  s.on('Network.requestWillBeSent', (e) => { if (e.request.url.includes('/music/')) reqs.set(e.requestId, { url: e.request.url.replace(/^https?:\/\/[^/]+/, ''), range: e.request.headers.Range || e.request.headers.range || '', bytes: 0, status: null, state: 'pendiente' }); });
  s.on('Network.responseReceived', (e) => { const r = reqs.get(e.requestId); if (r) r.status = e.response.status; });
  s.on('Network.dataReceived', (e) => { const r = reqs.get(e.requestId); if (r) r.bytes += e.encodedDataLength || e.dataLength; });
  s.on('Network.loadingFinished', (e) => { const r = reqs.get(e.requestId); if (r) { r.state = 'fin'; r.total = e.encodedDataLength; } });
  s.on('Network.loadingFailed', (e) => { const r = reqs.get(e.requestId); if (r) r.state = `fallo:${e.errorText}${e.canceled ? '(cancelada)' : ''}`; });
  return {
    list: () => [...reqs.values()].map((r) => `${r.url} ${r.range} → ${r.status} ${r.state} ${r.total ?? r.bytes} B`),
    total: () => [...reqs.values()].reduce((a, r) => a + (r.total ?? r.bytes), 0),
  };
}
