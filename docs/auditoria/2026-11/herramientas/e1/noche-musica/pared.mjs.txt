'mover el ratón a 300,500' : `tocar ${tapsMoved.join(' ; ')} (el primer toque da el foco a main#main-content, tabIndex -1, y la linterna salta a su centro; los siguientes, con main ya enfocado, sí la mueven)`}: ${moved}; de día sin recargar: «${dayView.h1}» enlace «${dayView.link?.text}»; al reencender misma posición=${b.link.sx === a.link.sx && b.link.sy === a.link.sy}; ${vp === 'escritorio' ? `Tab: ${JSON.stringify(tabbed)}; ` : ''}posiciones en 7 montajes: ${positions.join(' ')}; enlace fuera de la pantalla en ${overflows.length} de 7 montajes (${overflows.join('; ')}); ancho del enlace ${w}px → se sale si --sx > ${r2(maxSx)} % (≈ ${Math.round(outFrac * 100)} % de las posiciones posibles); «Salir de aquí» → ${after}; ${sum(issues)}`,
      hallazgos: [overflows.length || outFrac > 0 ? 'NUEVO-3' : '', vp === 'movil' && tapsMoved[0] !== '30,300→30px,300px' ? 'NUEVO-4' : ''].filter(Boolean).join(', '),
    });
    if (!ok && !overflows.length) await snap(page, `R-NIGHT-10-${vp}`);
    await context.close();
  }

  // ---------------- R-NIGHT-11 ----------------
  for (const vp of ['escritorio', 'movil']) {
    if (!want('R-NIGHT-11')) break;
    const { page, issues, context } = await openPage(browser, vp, { origin: inst.origin });
    await page.goto(`${inst.origin}/`);
    await page.waitForTimeout(1300);
    await nightOn(page, vp);
    if (vp === 'escritorio') {
      await page.locator('.spray-fab').click();
      const spot = await emptySpot(page, [[40, 300], [60, 400], [1400, 300]]);
      await mouseStroke(page, [spot, [spot[0] + 10, spot[1] + 60]]);
    } else {
      await page.locator('.spray-fab').tap();
      const spot = await emptySpot(page, [[6, 300], [10, 400], [366, 300]]);
      await touchDrag(page, line(spot[0], spot[1], spot[0] + 4, spot[1] + 60, 10));
      await page.waitForTimeout(650);
    }
    await navTo(page, vp, 'Rocky IA');
    await page.waitForTimeout(500);
    const chat = await page.evaluate(() => ({
      path: location.pathname, fab: !!document.querySelector('.spray-fab'), tools: !!document.querySelector('.spray-tools'), canvas: !!document.querySelector('canvas.spray-wall'),
      spraying: document.documentElement.classList.contains('is-spraying'), wall: !!document.querySelector('.uv-wall'), dark: !!document.querySelector('.uv-oscuridad'),
      depth: document.querySelector('.uv-oscuridad')?.style.getPropertyValue('--depth'), pill: !!document.querySelector('.mini-player--chat'), pillContent: !!document.querySelector('.mini-player--content'),
    }));
    const store = await sprayStorage(page);
    if (vp === 'escritorio') await page.mouse.move(400, 400);
    await navTo(page, vp, 'Tienda');
    await page.waitForTimeout(500);
    const back = await themeState(page);
    const ok = chat.path === '/rockyIA' && !chat.fab && !chat.tools && !chat.canvas && !chat.spraying && chat.wall && chat.dark && chat.depth === '0.000' && chat.pill && back.fab && !back.tools && (store?.pages?.['/']?.strokes?.length ?? 0) >= 1;
    C.add({
      id: 'R-NIGHT-11', viewport: vp, resultado: ok ? 'OK' : 'FALLO',
      evidencia: `en /rockyIA con el spray cogido antes: ${JSON.stringify(chat)}; rocky-spray['/'] trazos=${store?.pages?.['/']?.strokes?.length ?? 0} (guardado al salir); al volver a Tienda: bote=${back.fab} caja=${back.tools} (spray suelto); la pared y la oscuridad sí se montan en el chat aunque el comentario de Descontrol.jsx:165 diga «ahí no hay pared»; ${sum(issues)}`,
      hallazgos: 'C-NOCHE-07',
    });
    if (!ok) await snap(page, `R-NIGHT-11-${vp}`);
    await context.close();
  }

  // ---------------- R-NIGHT-23 (contraste) ----------------
  const contrastData = {};
  for (const vp of ['escritorio', 'movil']) {
    if (!want('R-NIGHT-23')) break;
    const { page, issues, context } = await openPage(browser, vp, { origin: inst.origin });
    await page.goto(`${inst.origin}/`);
    await page.waitForTimeout(1300);
    await nightOn(page, vp);
    const tok = await page.evaluate(() => {
      const cs = getComputedStyle(document.documentElement);
      const g = (n) => cs.getPropertyValue(n).trim();
      return { ink: g('--ink'), muted: g('--muted'), paper: g('--paper'), card: g('--card'), lime: g('--uv-lime'), violetSoft: g('--uv-violet-soft'), danger: g('--danger'), night: g('--neon-night'), dark: getComputedStyle(document.querySelector('.uv-oscuridad')).backgroundImage };
    });
    const black = { r: 0, g: 0, b: 0 };
    const dark = (c, alpha) => over({ ...black, a: alpha }, c);
    const P = (s) => parseRgb(s);
    const alphaAt = (d) => 0.08 + 0.34 * d;
    const pair = (fg, bg, d) => r2(ratio(dark(P(fg), alphaAt(d)), dark(P(bg), alphaAt(d))));
    let crossCard = null;
    let crossPaper = null;
    for (let d = 0; d <= 1.0001; d += 0.01) {
      if (crossCard === null && ratio(dark(P(tok.muted), alphaAt(d)), dark(P(tok.card), alphaAt(d))) < 4.5) crossCard = r2(d);
      if (crossPaper === null && ratio(dark(P(tok.muted), alphaAt(d)), dark(P(tok.paper), alphaAt(d))) < 4.5) crossPaper = r2(d);
    }
    const table = {
      'muted/card sin oscuridad': r2(ratio(P(tok.muted), P(tok.card))),
      'muted/paper sin oscuridad': r2(ratio(P(tok.muted), P(tok.paper))),
      'muted/card d=0 (α 0,08)': pair(tok.muted, tok.card, 0),
      'muted/card d=0,5 (α 0,25)': pair(tok.muted, tok.card, 0.5),
      'muted/card d=1 (α 0,42)': pair(tok.muted, tok.card, 1),
      'muted/paper d=1 (α 0,42)': pair(tok.muted, tok.paper, 1),
      'ink/card d=0': pair(tok.ink, tok.card, 0),
      'ink/card d=1 (α 0,42)': pair(tok.ink, tok.card, 1),
      'ink/paper d=1': pair(tok.ink, tok.paper, 1),
      'muted cae de 4,5 sobre card en depth': crossCard,
      'muted cae de 4,5 sobre paper en depth': crossPaper,
    };
    // Elementos reales con texto en --muted y su fondo opaco más cercano.
    const realMuted = async () => page.evaluate((mutedHex) => {
      const hex = mutedHex.replace('#', '');
      const target = `rgb(${parseInt(hex.slice(0, 2), 16)}, ${parseInt(hex.slice(2, 4), 16)}, ${parseInt(hex.slice(4, 6), 16)})`;
      const out = [];
      for (const el of document.querySelectorAll('main *, footer *')) {
        const cs = getComputedStyle(el);
        if (cs.color !== target || !el.childNodes.length || ![...el.childNodes].some((n) => n.nodeType === 3 && n.textContent.trim())) continue;
        if (!(el.offsetWidth || el.offsetHeight)) continue;
        let p = el;
        let bg = null;
        while (p && p !== document.documentElement) {
          const b = getComputedStyle(p).backgroundColor;
          if (b && b !== 'rgba(0, 0, 0, 0)' && b !== 'transparent') { bg = b; break; }
          p = p.parentElement;
        }
        out.push({ sel: `${el.tagName.toLowerCase()}.${[...el.classList].slice(0, 2).join('.')}`, text: el.textContent.trim().slice(0, 30), fs: cs.fontSize, fw: cs.fontWeight, bg: bg ?? 'body' });
        if (out.length > 60) break;
      }
      return out;
    }, tok.muted);
    const samples = {};
    for (const r of ['/', '/crew', '/estudio', '/no-existe']) {
      if (r !== '/') await spaGo(page, r);
      await page.waitForTimeout(500);
      const list = await realMuted();
      const bodyBg = await page.evaluate(() => getComputedStyle(document.body).backgroundColor);
      const uniq = {};
      for (const e of list) {
        const key = `${e.sel}|${e.bg}`;
        if (uniq[key]) continue;
        const bgc = P(e.bg === 'body' ? bodyBg : e.bg);
        const bgOpaque = bgc.a < 1 ? over(bgc, P(tok.paper)) : bgc;
        uniq[key] = `${e.sel} «${e.text}» ${e.fs}/${e.fw} sobre ${e.bg === 'body' ? `body ${bodyBg}` : e.bg}: d0 ${r2(ratio(dark(P(tok.muted), 0.08), dark(bgOpaque, 0.08)))} · d1 ${r2(ratio(dark(P(tok.muted), 0.42), dark(bgOpaque, 0.42)))}`;
      }
      samples[r] = { count: list.length, ejemplos: Object.values(uniq).slice(0, 6) };
    }
    // Profundidad y alfa real abajo del todo en '/'.
    await spaGo(page, '/');
    await page.evaluate(() => window.scrollTo(0, document.documentElement.scrollHeight));
    await page.waitForTimeout(400);
    const bottomDark = await page.evaluate(() => ({ depth: document.querySelector('.uv-oscuridad').style.getPropertyValue('--depth'), bg: getComputedStyle(document.querySelector('.uv-oscuridad')).backgroundImage.match(/rgba\([^)]*\)\s*100%/)?.[0], torchNow: getComputedStyle(document.querySelector('.uv-oscuridad')).getPropertyValue('--torch-now').trim(), vw: innerWidth, vh: innerHeight, tx: document.querySelector('.uv-oscuridad').style.getPropertyValue('--tx'), ty: document.querySelector('.uv-oscuridad').style.getPropertyValue('--ty') }));
    // Fracción del viewport a oscuridad máxima (fuera de 1,8 × haz) con la linterna donde está.
    const torchR = (vp === 'escritorio' ? 300 : 190) * (1 - 0.35 * Number(bottomDark.depth)) * 1.8;
    const tx = parseFloat(bottomDark.tx);
    const ty = parseFloat(bottomDark.ty);
    let outside = 0;
    let total = 0;
    for (let x = 0; x < bottomDark.vw; x += 5) for (let y = 0; y < bottomDark.vh; y += 5) { total += 1; if (Math.hypot(x - tx, y - ty) >= torchR) outside += 1; }
    // Caja del spray, píldora, pista del 404 y rótulo del apagón.
    const tb = P(tok.paper);
    const toolsBg = over({ ...P(tok.paper), a: 0.92 }, dark(P(tok.paper), 0.42));
    const pill = await page.evaluate(() => {
      window.scrollTo(0, 0);
      const t = document.querySelector('.mini-player-title');
      const p = document.querySelector('.mini-player');
      return t ? { color: getComputedStyle(t).color, bg: getComputedStyle(p).backgroundColor, btnColor: getComputedStyle(document.querySelector('.mini-player-btn')).color, btnBg: getComputedStyle(document.querySelector('.mini-player-btn')).backgroundColor } : null;
    });
    const extra = {
      'estado spray muted/caja': r2(ratio(P(tok.muted), toolsBg)),
      'Foto neon-night/lima': r2(ratio(P(tok.night), P(tok.lime))),
      '¿Borrar todo? danger/caja': r2(ratio(P(tok.danger), toolsBg)),
      'pista 404 violet-soft/negro .97': r2(ratio(P(tok.violetSoft), over({ ...black, a: 0.97 }, tb))),
      'apagón lima/negro .97': r2(ratio(P(tok.lime), over({ ...black, a: 0.97 }, tb))),
      'píldora título': pill ? `${pill.color} sobre ${pill.bg}: d0 ${r2(ratio(dark(P(pill.color), 0.08), dark(P(pill.bg).a < 1 ? over(P(pill.bg), tb) : P(pill.bg), 0.08)))}` : 'sin píldora',
    };
    contrastData[vp] = { tok, table, samples, bottomDark, fraccionOscuridadMax: r2(outside / total), extra };
    const worstMuted = table['muted/card d=1 (α 0,42)'];
    // El CSS promete > 4,5:1 abajo del todo: con --muted no se cumple.
    const ok = worstMuted >= 4.5;
    C.add({
      id: 'R-NIGHT-23', viewport: vp, resultado: ok ? 'OK' : 'FALLO',
      evidencia: `tokens ${JSON.stringify({ ink: tok.ink, muted: tok.muted, paper: tok.paper, card: tok.card })}; relaciones WCAG (texto y fondo oscurecidos por negro α=0,08+0,34·depth fuera del haz): ${JSON.stringify(table)}; abajo del todo en '/': ${JSON.stringify(bottomDark)}, ${Math.round((outside / total) * 100)} % del viewport a oscuridad máxima con la linterna en ${bottomDark.tx},${bottomDark.ty}; textos reales en --muted: ${Object.entries(samples).map(([r, s]) => `${r} (${s.count}): ${s.ejemplos.slice(0, 3).join(' / ')}`).join(' ; ')}; por encima de la oscuridad: ${JSON.stringify(extra)}; ${sum(issues)}`,
      hallazgos: 'C-NOCHE-08',
    });
    if (!ok) {
      await page.evaluate(() => window.scrollTo(0, document.documentElement.scrollHeight));
      await page.waitForTimeout(400);
      await snap(page, `R-NIGHT-23-${vp}`);
    }
    await context.close();
  }
  if (Object.keys(contrastData).length) fs.writeFileSync(path.join(OUT, 'contraste.json'), JSON.stringify(contrastData, null, 2));
} finally {
  await browser.close();
  inst.stop();
}
