function contrast(fg, bg) {
  const lum = (c) => {
    const [r, g, b2] = c.match(/[\d.]+/g).slice(0, 3).map(Number).map((v) => {
      const s = v / 255;
      return s <= 0.03928 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
    });
    return 0.2126 * r + 0.7152 * g + 0.0722 * b2;
  };
  const [l1, l2] = [lum(fg), lum(bg)].sort((a, c) => c - a);
  return ((l1 + 0.05) / (l2 + 0.05)).toFixed(2);
}

          for (const e of l.getEntries()) if (!e.hadRecentInput) window.__cls += e.value;
        }).observe({ type: 'layout-shift', buffered: true });
      });
      await o.page.goto(ORIGIN + '/loquesea');
      await o.page.waitForSelector('main h1');
      await sleep(300);
      const y0 = await o.page.evaluate(() => Math.round(document.querySelector('main h1').getBoundingClientRect().top));
      await sleep(2200);
      const y1 = await o.page.evaluate(() => ({ y: Math.round(document.querySelector('main h1').getBoundingClientRect().top), cls: window.__cls.toFixed(3), dog: document.querySelector('.notfound-dog').getBoundingClientRect().height }));
      o.page.__c13 = { y0, ...y1 };
      R.add({ id: 'R-404-01', viewport: vp, resultado: 'OK', evidencia: `(comprobación de C-EST-13, perro retrasado 1,5 s) h1 en y=${y0} antes de llegar el perro y y=${y1.y} después (perro ${Math.round(y1.dog)} px de alto); CLS acumulado ${y1.cls}`, hallazgos: y0 !== y1.y ? 'C-EST-13' : '' });
      await o.context.close();
    }
  }

  // ---------- R-404-02 + TECLADO-404 ----------
  if (want('R-404-02')) {
    for (const vp of ['escritorio', 'movil']) {
      const parts = [];
      let ok = true;
      const exits = [];
      for (let rep = 0; rep < (vp === 'movil' ? 6 : 3); rep += 1) {
        const o = await open(b, vp);
        const { page } = o;
        await page.goto(ORIGIN + '/loquesea');
        await sleep(700);
        await turnOn(page, vp);
        await sleep(300);
        const s = await page.evaluate(() => {
          const dark = document.querySelector('.notfound-oscuridad');
          const exit = document.querySelector('.notfound-salida');
          const r = exit?.getBoundingClientRect();
          const nav = document.querySelector('header.navbar');
          const pista = document.querySelector('.notfound-pista');
          return {
            h1: document.querySelector('main h1')?.textContent,
            text: document.querySelector('.notfound-text')?.textContent,
            darkZ: dark ? getComputedStyle(dark).zIndex : null,
            darkPE: dark ? getComputedStyle(dark).pointerEvents : null,
            mask: dark ? getComputedStyle(dark).maskImage.slice(0, 60) : null,
            navZ: getComputedStyle(nav).zIndex,
            navHit: document.elementFromPoint(innerWidth / 2, 20)?.closest('header') !== null,
            exit: exit ? { text: exit.textContent, href: exit.getAttribute('href'), sx: exit.style.getPropertyValue('--sx'), sy: exit.style.getPropertyValue('--sy'), left: Math.round(r.left), right: Math.round(r.right), top: Math.round(r.top), w: Math.round(r.width), vw: innerWidth, pos: getComputedStyle(exit).position } : null,
            pista: pista?.textContent,
            pistaBottom: pista ? Math.round(innerHeight - pista.getBoundingClientRect().bottom) : null,
          };
        });
        exits.push(s.exit);
        if (rep === 0) {
          // Linterna: puntero y foco
          await page.mouse.move(300, 300);
          await sleep(150);
          const t1 = await page.evaluate(() => document.querySelector('.notfound-oscuridad').style.cssText);
          if (vp === 'movil') {
            await page.touchscreen.tap(120, 500);
            await sleep(150);
          }
          const t2 = await page.evaluate(() => document.querySelector('.notfound-oscuridad').style.cssText);
          // Alumbrar la salida y pulsarla
          const ex = s.exit;
          const inView = ex.left >= 0 && ex.right <= ex.vw;
          await page.mouse.move(Math.min(Math.max(ex.left + ex.w / 2, 5), ex.vw - 5), ex.top + 20);
          await sleep(200);
          if (vp === 'movil') await page.touchscreen.tap(Math.min(Math.max(ex.left + ex.w / 2, 5), ex.vw - 5), ex.top + 20);
          else await page.mouse.click(ex.left + ex.w / 2, ex.top + 20);
          await sleep(500);
          const after = await page.evaluate(() => location.pathname);
          // 4. Atrás: el 404 vuelve todavía de noche; llegar con Tab
          await page.goBack();
          await sleep(700);
          const back = await page.evaluate(() => ({ path: location.pathname, theme: document.documentElement.dataset.theme, h1: document.querySelector('main h1')?.textContent }));
          const seq = [];
          let reached = null;
          for (let i = 0; i < 25; i += 1) {
            await page.keyboard.press('Tab');
            const f = await focusInfo(page);
            seq.push(f);
            if (f.name === 'Salir de aquí') {
              reached = f;
              break;
            }
          }
          const torch = await page.evaluate(() => document.querySelector('.notfound-oscuridad')?.style.cssText);
          parts.push(`textos «${s.h1}» / «${s.text}»; oscuridad z=${s.darkZ} pointer-events=${s.darkPE}, barra z=${s.navZ} visible y pulsable=${s.navHit}; linterna tras mover el puntero: «${t1}»${vp === 'movil' ? `, tras tocar: «${t2}»` : ''}; salida «${ex.text}» → ${ex.href} en --sx=${ex.sx} --sy=${ex.sy}; clic sobre ella → ${after}; Atrás → ${back.path} data-theme=${back.theme} «${back.h1}»; con Tab llega a «${reached?.name}» tras ${seq.length} pulsaciones (${fmtFocus(reached || seq[seq.length - 1])}), linterna al foco: «${torch}»; pista «${s.pista}» a ${s.pistaBottom}px del borde inferior`);
          ok = ok && s.h1 === 'No deberías estar aquí' && s.text === 'Se ha ido la luz. Busca la salida.' && s.darkZ === '950' && s.navHit && after === '/' && back.theme === 'neon' && Boolean(reached) && /--tx/.test(torch);
          o.inView = inView;
        }
        ok = ok && clean(o);
        if (rep === 0) parts.push(sum(o));
        await o.context.close();
      }
      const cut = exits.filter((e) => e.left < 0 || e.right > e.vw);
      if (vp === 'movil') parts.push(`posiciones de la salida en ${exits.length} cargas (ancho ${exits[0].vw}px, botón ${exits[0].w}px): ${exits.map((e) => `x=${e.sx}→${e.left}..${e.right}`).join(', ')}; cortadas por la derecha: ${cut.length}`);
      R.add({ id: 'R-404-02', viewport: vp, resultado: ok && !cut.length ? 'OK' : 'FALLO', evidencia: parts.join('; '), hallazgos: vp === 'movil' ? 'C-EST-14' : '' });
    }
    // C-EST-14 con posiciones forzadas (Math.random fijo) en móvil
    const forced = [];
    for (const rnd of [0.3, 0.5, 0.7, 0.99]) {
      const o = await open(b, 'movil');
      await o.context.addInitScript((r) => {
        const orig = Math.random;
        window.__forceExit = true;
        Math.random = function () {
          return window.__forceExit ? r : orig();
        };
      }, rnd);
      await o.page.goto(ORIGIN + '/loquesea');
      await sleep(600);
      await o.page.evaluate(() => { window.__forceExit = true; });
      await turnOn(o.page, 'movil');
      await sleep(300);
      // El 404 se monta al cargar: x = 10 + rnd*66. Forzar requiere que hiddenExit corra con Math.random fijo
      const e = await o.page.evaluate(() => {
        const ex = document.querySelector('.notfound-salida');
        const r = ex.getBoundingClientRect();
        return { sx: ex.style.getPropertyValue('--sx'), left: Math.round(r.left), right: Math.round(r.right), vw: innerWidth };
      });
      forced.push(`x=${e.sx}: ${e.left}..${e.right} de ${e.vw}px${e.right > e.vw ? ' (CORTADO)' : ''}`);
      if (e.right > e.vw && rnd === 0.99) await shot(o.page, 'estructura', 'R-404-02-movil-salida-cortada');
      await o.context.close();
    }
    R.add({ id: 'R-404-02', viewport: 'movil', resultado: forced.some((f) => f.includes('CORTADO')) ? 'FALLO' : 'OK', evidencia: `(C-EST-14 con Math.random fijado al cargar) ${forced.join('; ')}`, hallazgos: 'C-EST-14' });
  }

  // ---------- TECLADO-404 ----------
  if (want('TECLADO-404')) {
    const o = await open(b, 'escritorio');
    const { page } = o;
    const walk = async () => {
      await page.evaluate(() => { document.body.setAttribute('tabindex', '-1'); document.body.focus(); document.body.removeAttribute('tabindex'); });
      const seq = [];
      for (let i = 0; i < 30; i += 1) {
        await page.keyboard.press('Tab');
        const f = await focusInfo(page);
        seq.push(f);
        if (/Volver a la tienda|Salir de aquí/.test(f.name)) break;
      }
      return seq;
    };
    await page.goto(ORIGIN + '/loquesea');
    await sleep(800);
    const day = await walk();
    await page.keyboard.press('Shift+Tab');
    await turnOn(page, 'escritorio');
    const night = await walk();
    await sleep(300);
    const lit = await page.evaluate(() => {
      const ex = document.querySelector('.notfound-salida').getBoundingClientRect();
      const d = document.querySelector('.notfound-oscuridad').style;
      return { tx: parseFloat(d.getPropertyValue('--tx')), ty: parseFloat(d.getPropertyValue('--ty')), cx: Math.round(ex.left + ex.width / 2), cy: Math.round(ex.top + ex.height / 2) };
    });
    await page.keyboard.press('Enter');
    await sleep(500);
    const after = await page.evaluate(() => location.pathname);
    const dLast = day[day.length - 1];
    const nLast = night[night.length - 1];
    const ok = dLast.name === 'Volver a la tienda' && dLast.ring && !dLast.covered && nLast.name === 'Salir de aquí' && nLast.ring && Math.abs(lit.tx - lit.cx) <= 2 && Math.abs(lit.ty - lit.cy) <= 2 && after === '/';
    R.add({ id: 'TECLADO-404', viewport: 'escritorio', resultado: ok ? 'OK' : 'FALLO', evidencia: `día, ${day.length} Tab hasta ${fmtFocus(dLast)} (${dLast.outline}); noche, ${night.length} Tab hasta ${fmtFocus(nLast)} (${nLast.outline}); la linterna apunta al foco (${lit.tx},${lit.ty}) = centro del botón (${lit.cx},${lit.cy}); Enter → ${after}; la barra sigue usable con teclado por encima de la oscuridad; ${sum(o)}` });
    await o.context.close();
  }

  // ---------- R-404-03 ----------
  if (want('R-404-03')) {
    for (const vp of ['escritorio', 'movil']) {
      const o = await open(b, vp);
      const { page } = o;
      await page.goto(ORIGIN + '/');
      await sleep(600);
      await page.goto(ORIGIN + '/loquesea');
      await sleep(600);
      const parts = [];
      let ok = true;
      if (vp === 'escritorio') {
        await page.locator('header nav').getByRole('link', { name: 'Drops' }).click();
      } else {
        await page.locator('header .navbar-menu').click();
        await page.getByRole('dialog', { name: 'Menú' }).getByRole('link', { name: 'Drops' }).click();
      }
      await page.waitForFunction(() => location.pathname === '/menudrop');
      await page.goBack();
      await sleep(600);
      const back = await page.evaluate(() => ({ path: location.pathname, h1: document.querySelector('main h1')?.textContent }));
      parts.push(`Drops → /menudrop, Atrás → ${back.path} «${back.h1}»`);
      ok = ok && back.path === '/loquesea' && back.h1 === 'Te has salido del mapa';
      if (vp === 'movil') {
        await page.locator('header .navbar-menu').click();
        await page.getByRole('dialog', { name: 'Menú' }).getByRole('link', { name: 'Tienda' }).click();
        await sleep(500);
        const p = await page.evaluate(() => location.pathname);
        parts.push(`menú → Tienda → ${p}`);
        ok = ok && p === '/';
        await page.goBack();
        await sleep(600);
      }
      // 4. De noche la barra queda por encima de la oscuridad y se usa
      await turnOn(page, vp);
      const n = await page.evaluate(() => {
        const pts = [[40, 28], [innerWidth / 2, 28], [innerWidth - 30, 28]].map(([x, y]) => document.elementFromPoint(x, y)?.closest('header') !== null);
        return { pts, dark: Boolean(document.querySelector('.notfound-oscuridad')) };
      });
      if (vp === 'escritorio') await page.locator('header nav').getByRole('link', { name: 'Crew', exact: true }).click();
      else {
        await page.locator('header .navbar-menu').click();
        await page.getByRole('dialog', { name: 'Menú' }).getByRole('link', { name: 'Crew', exact: true }).click();
      }
      await page.waitForFunction(() => location.pathname === '/crew', null, { timeout: 5000 }).catch(() => {});
      const p2 = await page.evaluate(() => location.pathname);
      parts.push(`de noche (oscuridad=${n.dark}) la barra recibe los puntos [${n.pts}] y «Crew» lleva a ${p2}`);
      ok = ok && n.pts.every(Boolean) && p2 === '/crew' && clean(o);
      R.add({ id: 'R-404-03', viewport: vp, resultado: ok ? 'OK' : 'FALLO', evidencia: `${parts.join('; ')}; ${sum(o)}` });
      await o.context.close();
    }
  }
} finally {
  await b.close();
}
