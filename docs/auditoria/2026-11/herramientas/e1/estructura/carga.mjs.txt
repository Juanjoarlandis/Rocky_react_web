const loadingState = (page) =>
  page.evaluate(() => {
    const l = document.querySelector('.page-loading');
    const sp = l?.querySelector('.page-loading-spinner');
    return {
      loading: Boolean(l),
      text: l?.textContent.trim(),
      role: l?.getAttribute('role') || null,
      live: l?.getAttribute('aria-live') || null,
      spinnerHidden: sp?.getAttribute('aria-hidden'),
      spinnerAnim: sp ? getComputedStyle(sp).animationName : null,
      nav: Boolean(document.querySelector('header.navbar')),
      footer: Boolean(document.querySelector('footer.footer')),
      path: location.pathname,
      h1: document.querySelector('main h1')?.textContent?.trim().slice(0, 40) || null,
      title: document.title,
    };
  });

      hallazgos: metaFixed ? 'C-EST-11, C-EST-12' : '',
    });
    await o.context.close();
  }

  // ---------- R-404-04 ----------
  if (want('R-404-04')) {
    const parts = [];
    const head = async (url, method = 'HEAD') => {
      const res = await fetch(url, { method, redirect: 'manual' });
      const body = method === 'GET' ? await res.text() : '';
      return { status: res.status, cc: res.headers.get('cache-control'), cf: res.headers.get('cloudflare-cdn-cache-control'), type: res.headers.get('content-type'), body };
    };
    const v = await head(`${VITE}/loquesea`, 'GET');
    parts.push(`Vite /loquesea: ${v.status} ${v.type} SPA=${/<div id="root">/.test(v.body)} noindex=${/noindex/.test(v.body)}`);
    const a = await head(`${ORIGIN}/loquesea`, 'GET');
    parts.push(`build /loquesea: ${a.status} ${a.type} Cache-Control «${a.cc}» Cloudflare-CDN-Cache-Control «${a.cf}» noindex=${/noindex/.test(a.body)} canonical portada=${/rel="canonical" href="https:\/\/rocky035.com\/"/.test(a.body)}`);
    const l = await head(`${ORIGIN}/legal/loquesea`, 'GET');
    parts.push(`build /legal/loquesea: ${l.status} «${l.cc}» noindex=${/noindex/.test(l.body)}`);
    const hd = await head(`${ORIGIN}/loquesea`, 'HEAD');
    parts.push(`HEAD /loquesea: ${hd.status}`);
    const as = await head(`${ORIGIN}/assets/noexiste.js`, 'GET');
    parts.push(`/assets/noexiste.js: ${as.status} «${as.body.trim()}» Cache-Control «${as.cc}»`);
    const api = await head(`${ORIGIN}/api/loquesea`, 'GET');
    parts.push(`/api/loquesea: ${api.status} ${api.type} ${api.body.trim().slice(0, 120)}`);
    const ok = v.status === 200 && a.status === 200 && l.status === 200 && a.cc === 'public, max-age=0, must-revalidate' && a.cf === 'no-store' && !/noindex/.test(a.body) && as.status === 404 && /Asset not found/.test(as.body) && api.status === 404 && /json/.test(api.type);
    R.add({ id: 'R-404-04', viewport: 'no aplica (HTTP)', resultado: ok ? 'OK' : 'FALLO', evidencia: parts.join('; '), hallazgos: 'C-EST-12' });
  }
} finally {
  await b.close();
}
