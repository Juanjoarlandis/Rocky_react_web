// Marca cuándo aparece y desaparece la capa de la intro (ms desde el inicio de la navegación).
const splashTimer = () => {
  window.__splash = {};
  new MutationObserver(() => {
    const el = document.querySelector('.splash-intro');
    if (el && !window.__splash.added) window.__splash.added = performance.now();
    if (!el && window.__splash.added && !window.__splash.removed) window.__splash.removed = performance.now();
  }).observe(document, { childList: true, subtree: true });
};

// Deslizamiento táctil real (touchStart/touchMove/touchEnd por CDP).
export async function swipe(page, dy, x = 200, y = 650) {
  const cdp = await page.context().newCDPSession(page);
  await cdp.send('Input.dispatchTouchEvent', { type: 'touchStart', touchPoints: [{ x, y }] });
  for (let i = 1; i <= 12; i += 1) {
    await cdp.send('Input.dispatchTouchEvent', { type: 'touchMove', touchPoints: [{ x, y: y - (dy * i) / 12 }] });
    await sleep(16);
  }
  await cdp.send('Input.dispatchTouchEvent', { type: 'touchEnd', touchPoints: [] });
  await cdp.detach();
}

async function scrollAttempt(page, vp) {
  if (vp === 'escritorio') {
    await page.mouse.move(700, 450);
    await page.mouse.wheel(0, 900);
  } else {
    await swipe(page, 400);
  }
  await sleep(500);
  return page.evaluate(() => Math.round(window.scrollY));
}

async function introRun(vp, route, reducedMotion = 'no-preference') {
  const o = await open(b, vp, { skipIntro: false, reducedMotion });
  await o.context.addInitScript(splashTimer);
  const { page } = o;
  await page.goto(ORIGIN + route);
  await page.waitForSelector('.splash-intro', { timeout: 3000 });
  await sleep(150);
  const info = await page.evaluate(() => {
    const s = document.querySelector('.splash-intro');
    const cs = getComputedStyle(s);
    const r = s.getBoundingClientRect();
    const delays = (sel) => [...document.querySelectorAll(sel)].map((e) => parseFloat(e.style.getPropertyValue('--delay')));
    const anim = (sel) => {
      const e = document.querySelector(sel);
      if (!e) return null;
      const c = getComputedStyle(e);
      return `${c.animationName} ${c.animationDuration} ${c.animationDelay}`;
    };
    const sq = document.querySelector('.splash-squiggle path');
    return {
      role: s.getAttribute('role'),
      z: cs.zIndex,
      pos: cs.position,
      covers: r.width >= innerWidth - 1 && r.height >= innerHeight - 1,
      logoAlt: s.querySelector('.splash-logo')?.alt,
      lema: s.querySelector('.splash-lema')?.textContent,
      munecos: s.querySelectorAll('.splash-muneco').length,
      heads: s.querySelectorAll('.splash-head').length,
      dMun: delays('.splash-muneco'),
      dHead: delays('.splash-head'),
      squiggle: Boolean(sq),
      sqDash: sq ? getComputedStyle(sq).strokeDashoffset : null,
      bodyOverflow: document.body.style.overflow,
      animMun: anim('.splash-muneco'),
      animHead: anim('.splash-head'),
      animLogo: anim('.splash-logo'),
      animLema: anim('.splash-lema'),
      animSq: sq ? `${getComputedStyle(sq).animationName}` : null,
      animIntro: `${cs.animationName} ${cs.animationDuration} ${cs.animationDelay}`,
      prioDuring: document.querySelectorAll('main [fetchpriority="high"]').length,
      peeker: document.querySelectorAll('[data-testid="curious-peeker"]').length,
    };
  });
  const scrollY = await scrollAttempt(page, vp);
  // Estampida desactivada durante la intro
  await page.keyboard.type('035');
  await sleep(1700);
  const estampidaDuring = await page.locator('[data-testid="estampida"]').count();
  await page.waitForSelector('.splash-intro', { state: 'detached', timeout: 5000 });
  await sleep(300);
  const after = await page.evaluate(() => ({
    t: window.__splash,
    seen: sessionStorage.getItem('rocky-splash-seen'),
    bodyOverflow: document.body.style.overflow,
    h1: document.querySelector('main h1, main h2')?.textContent?.trim().slice(0, 40),
    path: location.pathname,
    prioAfter: document.querySelectorAll('main [fetchpriority="high"]').length,
    notice: document.querySelector('.commerce-notice')?.textContent,
  }));
  return { o, info, scrollY, estampidaDuring, after };
}
