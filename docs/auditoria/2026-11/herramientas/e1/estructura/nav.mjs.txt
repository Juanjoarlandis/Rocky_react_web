async function navState(page) {
  return page.evaluate(() => {
    const links = [...document.querySelectorAll('header nav[aria-label="Navegación principal"] a')];
    const cart = document.querySelector('.navbar-cart');
    return {
      path: location.pathname,
      title: document.title,
      current: [...links, cart].filter((a) => a.getAttribute('aria-current') === 'page').map((a) => a.getAttribute('aria-label') || a.textContent.trim()),
      active: links.filter((a) => a.classList.contains('is-active')).map((a) => a.textContent.trim()),
      footer: Boolean(document.querySelector('footer.footer')),
      radio: Boolean(document.querySelector('.mini-player-slot--content')),
      marker: window.__noReload === true,
    };
  });
}
