// R-EGG-04, R-EGG-05 y R-EGG-06: la Estampida del 035.
import { launchBrowser, open, sum, clean, shot, ORIGIN, sleep, results } from './common.mjs';

const R = results('estampida');
const b = await launchBrowser();
const only = process.argv[2] ? process.argv[2].split(',') : null;
const want = (id) => !only || only.includes(id);

const SEL = '[data-testid="estampida"]';

async function blankClick(page) {
  // Un hueco de la página: el margen izquierdo a media altura, comprobando que no es interactivo
  const pt = await page.evaluate(() => {
    for (const [x, y] of [[6, 500], [6, 700], [innerWidth - 6, 600], [innerWidth / 2, innerHeight - 30]]) {
      const el = document.elementFromPoint(x, y);
      if (el && !el.closest('a, button, input, textarea, select, label, [role="button"], [tabindex]:not([tabindex="-1"])')) return [x, y];
    }
    return null;
  });
  if (pt) await page.mouse.click(pt[0], pt[1]);
  else await page.evaluate(() => document.activeElement?.blur());
  return pt;
}

// Teclado numérico con Bloq Num activado: Playwright sólo emula Bloq Num desactivado
// (Numpad0 → key «Insert»), así que se envía por CDP lo que manda un teclado real: key «0», code «Numpad0».
async function numpad(page, digits) {
  const cdp = await page.context().newCDPSession(page);
  for (const d of digits) {
    const base = { key: d, code: `Numpad${d}`, windowsVirtualKeyCode: 96 + Number(d), location: 3 };
    await cdp.send('Input.dispatchKeyEvent', { type: 'keyDown', ...base, text: d });
    await cdp.send('Input.dispatchKeyEvent', { type: 'keyUp', ...base });
    await sleep(60);
  }
  await cdp.detach();
}

async function typeSeq(page, keys) {
  if (keys[0] === 'NUMPAD') return numpad(page, keys.slice(1));
  for (const k of keys) {
    await page.keyboard.press(k);
    await sleep(60);
  }
}

const runInfo = (page) =>
  page.evaluate((sel) => {
    const e = document.querySelector(sel);
    if (!e) return null;
    const rs = [...e.querySelectorAll('.estampida-corredor')];
    const num = (v) => parseFloat(v);
    const altos = rs.map((r) => num(r.style.getPropertyValue('--alto')));
    const duras = rs.map((r) => num(r.style.getPropertyValue('--dura')));
    const sales = rs.map((r) => num(r.style.getPropertyValue('--sale')));
    const cs = getComputedStyle(e);
    const first = rs[0] ? getComputedStyle(rs[0]) : null;
    return {
      n: rs.length,
      alto: [Math.min(...altos), Math.max(...altos)],
      dura: [Math.min(...duras), Math.max(...duras)],
      life: Math.ceil(Math.max(...rs.map((r, i) => sales[i] + duras[i])) * 1000) + 300,
      jumps: rs.filter((r) => r.classList.contains('salta')).length,
      z: cs.zIndex,
      pe: cs.pointerEvents,
      hidden: e.getAttribute('aria-hidden'),
      grito: e.querySelector('.estampida-grito')?.textContent,
      quieta: e.classList.contains('is-quieta'),
      srcs: rs.map((r) => r.querySelector('img').getAttribute('src').split('/').pop()),
      firstAnim: first?.animationName,
      firstBottom: first?.bottom,
      firstHeight: first?.height,
      vh: innerHeight,
    };
  }, SEL);

async function launchAndMeasure(page, keys) {
  const t0 = Date.now();
  await typeSeq(page, keys);
  const appeared = await page.waitForSelector(SEL, { timeout: 3500 }).then(() => true).catch(() => false);
  if (!appeared) return { appeared: false };
  const tA = Date.now();
  const info = await runInfo(page);
  await page.waitForSelector(SEL, { state: 'detached', timeout: 8000 }).catch(() => {});
  const tD = Date.now();
  return { appeared: true, wait: tA - t0, lived: tD - tA, info };
}

try {
  // ---------- R-EGG-04 ----------
  if (want('R-EGG-04')) {
    const o = await open(b, 'escritorio');
    const { page } = o;
    const bytes = {};
    page.on('response', async (r) => {
      if (/\/assets\/.*(-hd-|tumbado-800).*\.webp/.test(r.url())) bytes[r.url()] = Number(r.headers()['content-length'] || 0);
    });
    await page.goto(ORIGIN + '/');
    await sleep(1200);
    const before = Object.keys(bytes).length;
    const pt = await blankClick(page);
    const r1 = await launchAndMeasure(page, ['0', '3', '5']);
    const hdCount = Object.keys(bytes).length - before;
    const hdKiB = Math.round(Object.values(bytes).reduce((a, c) => a + c, 0) / 1024);
    // 3. 035 mientras corren
    const t3 = await (async () => {
      await typeSeq(page, ['0', '3', '5']);
      await page.waitForSelector(SEL, { timeout: 3000 }).catch(() => {});
      await sleep(300);
      await typeSeq(page, ['0', '3', '5']);
      await page.waitForSelector(SEL, { state: 'detached', timeout: 8000 }).catch(() => {});
      await sleep(2000);
      return page.locator(SEL).count();
    })();
    // 4. 1035 y 0 · clic · 35
    const r4a = await launchAndMeasure(page, ['1', '0', '3', '5']);
    await typeSeq(page, ['0']);
    await blankClick(page);
    const r4b = await launchAndMeasure(page, ['3', '5']);
    // 6. Teclado numérico
    const r6 = await launchAndMeasure(page, ['NUMPAD', '0', '3', '5']);
    // Con Bloq Num desactivado (lo que emula Playwright) las teclas son Insert/PageDown/Clear
    const r6off = await launchAndMeasure(page, ['Numpad0', 'Numpad3', 'Numpad5']);
    // 5. De noche
    await page.locator('header .navbar-tools > .navbar-theme').click();
    await sleep(300);
    await blankClick(page);
    const r5 = await launchAndMeasure(page, ['0', '3', '5']);
    const nightDiff = r5.info && r1.info ? r5.info.srcs.filter((s) => !r1.info.srcs.includes(s)).length : null;
    const i = r1.info;
    const ok = r1.appeared && r1.wait <= 1900 && i.n === 24 && i.alto[0] >= 70 && i.alto[1] <= 140 && i.dura[0] >= 2.3 && i.dura[1] <= 3.8 && Number(i.z) > 1000 && i.pe === 'none' && i.grito === '¡Estampida!' && Math.abs(r1.lived - i.life) < 700 && t3 === 0 && r4a.appeared && r4b.appeared && r6.appeared && r5.appeared && nightDiff > 0 && clean(o);
    R.add({
      id: 'R-EGG-04',
      viewport: 'escritorio',
      resultado: ok ? 'OK' : 'FALLO',
      evidencia: `clic en hueco ${JSON.stringify(pt)} y 0·3·5 → aparece a ${r1.wait} ms (precarga ≤ 1,5 s); ${i.n} corredores, alto ${i.alto.join('–')} px, duración ${i.dura.join('–')} s, ${i.jumps} saltan, z-index ${i.z} (barra 1000), pointer-events ${i.pe}, aria-hidden ${i.hidden}, «${i.grito}»; vive ${r1.lived} ms (calculado ${i.life} ms); primera vez descarga ${hdCount} ilustraciones -hd/tumbado (${hdKiB} KiB); 035 mientras corren → otra estampida después=${t3}; 1035 → ${r4a.appeared}; 0·clic·35 → ${r4b.appeared}; teclado numérico con Bloq Num (key «0»…, code «Numpad0»…, por CDP) → ${r6.appeared}; con Bloq Num desactivado (key Insert/PageDown/Clear) → ${r6off.appeared}, como es lógico; de noche → ${r5.appeared}, ${nightDiff}/24 imágenes distintas de las de día; ${sum(o)}`,
      hallazgos: 'C-EST-19',
    });
    if (!ok) await shot(page, 'estructura', 'R-EGG-04-escritorio');
    await o.context.close();
  }

  // ---------- R-EGG-05 ----------
  if (want('R-EGG-05')) {
    for (const vp of ['escritorio', 'movil']) {
      const o = await open(b, vp);
      const { page } = o;
      const parts = [];
      const none = async () => {
        await sleep(1900);
        return (await page.locator(SEL).count()) === 0;
      };
      // 1. Chat
      await page.goto(ORIGIN + '/rockyIA');
      const ta = page.locator('main textarea, main input[type="text"]').first();
      await ta.waitFor({ timeout: 6000 });
      await ta.click();
      await typeSeq(page, ['0', '3', '5']);
      const c1 = await none();
      const chatValue = await ta.inputValue();
      parts.push(`(1) 035 en el redactor del chat (valor «${chatValue}», sin enviar) → sin estampida=${c1}`);
      // 2. Otro campo de texto y un select
      await page.goto(ORIGIN + '/');
      await sleep(900);
      const email = page.locator('input.drop-aviso-email');
      await email.scrollIntoViewIfNeeded();
      await email.click();
      await typeSeq(page, ['0', '3', '5']);
      const c2 = await none();
      await email.fill('');
      await page.goto(ORIGIN + '/estudio');
      const sel = page.locator('main select').first();
      await sel.waitFor({ timeout: 6000 });
      await sel.focus();
      await typeSeq(page, ['0', '3', '5']);
      const c2b = await none();
      parts.push(`(2) 035 en el email del aviso → sin estampida=${c2}; en un select de /estudio → sin estampida=${c2b}`);
      // 3. Con Ctrl y con Alt fuera de campos
      await page.goto(ORIGIN + '/');
      await sleep(800);
      await blankClick(page);
      await typeSeq(page, ['Control+0', 'Control+3', 'Control+5']);
      const c3a = await none();
      await typeSeq(page, ['Alt+0', 'Alt+3', 'Alt+5']);
      const c3b = await none();
      parts.push(`(3) Ctrl+0/3/5 → sin estampida=${c3a}; Alt+0/3/5 → sin estampida=${c3b}`);
      // 5. Móvil: ningún disparador táctil (toques en logo, hueco y título)
      if (vp === 'movil') {
        for (let i = 0; i < 3; i += 1) await page.touchscreen.tap(30, 28);
        await page.goto(ORIGIN + '/');
        await sleep(600);
        for (let i = 0; i < 3; i += 1) await page.touchscreen.tap(6, 500);
        const c5 = await none();
        await blankClick(page);
        const r = await launchAndMeasure(page, ['0', '3', '5']);
        parts.push(`(5) toques en logo y hueco → sin estampida=${c5} (Estampida sólo escucha keydown); con teclado físico (simulado) sí sale=${r.appeared}`);
      }
      // 6. Foco en la mesa de beats, fuera de campos
      await page.goto(ORIGIN + '/estudio');
      const pad1 = page.locator('button[aria-label^="Pad 1:"]');
      await pad1.waitFor({ timeout: 6000 });
      await pad1.focus();
      await page.evaluate(() => {
        window.__golpes = [];
        new MutationObserver(() => {
          const g = document.querySelector('.pad--golpe');
          if (g) window.__golpes.push(g.getAttribute('aria-label'));
        }).observe(document.querySelector('.pads-rejilla'), { subtree: true, attributes: true, attributeFilter: ['class'] });

      await typeSeq(page, ['0', '3', '5']);

      const golpes = await page.evaluate(() => [...new Set(window.__golpes)]);
      parts.push(`(6) foco en «Pad 1» y 0·3·5 → pads golpeados [${golpes.join(', ')}] y estampida=${appeared6}`);
      parts.push('(4) durante la intro: ver R-NAV-01/R-NAV-04 (035 no la lanza)');
      const ok = c1 && c2 && c2b && c3a && c3b && clean(o);
      R.add({ id: 'R-EGG-05', viewport: vp, resultado: ok ? 'OK' : 'FALLO', evidencia: `${parts.join('; ')}; ${sum(o)}`, hallazgos: appeared6 && golpes.length ? 'C-EST-18' : '' });
      await o.context.close();
    }
  }

  // ---------- R-EGG-06 ----------
  if (want('R-EGG-06')) {
    const o = await open(b, 'escritorio', { reducedMotion: 'reduce' });
    const { page } = o;
    await page.goto(ORIGIN + '/');
    await sleep(1000);
    await blankClick(page);
    const r1 = await launchAndMeasure(page, ['0', '3', '5']);
    await page.emulateMedia({ reducedMotion: 'no-preference' });
    await blankClick(page);
    const r2 = await launchAndMeasure(page, ['0', '3', '5']);
    const i = r1.info;
    const ok = r1.appeared && i.quieta && i.n === 24 && i.firstHeight === '90px' && Math.abs(parseFloat(i.firstBottom) - 0.06 * i.vh) < 2 && i.firstAnim === 'estampida-aparece' && Math.abs(r1.lived - 2600) < 500 && r2.appeared && !r2.info.quieta && clean(o);
    R.add({
      id: 'R-EGG-06',
      viewport: 'escritorio',
      resultado: ok ? 'OK' : 'FALLO',
      evidencia: `movimiento reducido: is-quieta=${i?.quieta}, ${i?.n} en fila, alto ${i?.firstHeight}, bottom ${i?.firstBottom} (6 vh = ${(0.06 * i?.vh).toFixed(1)} px), animación ${i?.firstAnim}; se desmonta a los ${r1.lived} ms (esperado 2600); quitada la emulación sin recargar → corren (is-quieta=${r2.info?.quieta}, animación ${r2.info?.firstAnim}); ${sum(o)}`,
    });
    await o.context.close();
  }
} finally {
  await b.close();
}
