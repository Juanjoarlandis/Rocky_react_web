// Recorridos de la mesa (sin servidor de por medio): R-STU-01…05, 07…12, 14…17.
import { run, MESA, codeOf, presetCode, encodePattern, emptyPattern } from './runner.mjs';

const BOTH = ['escritorio', 'movil'];
const press = (vp, loc) => (vp === 'movil' ? loc.tap() : loc.click());
const lcdText = (page) => page.locator('.lcd-rec').textContent();
const rowsOf = (id) => {
  const p = MESA.PRESETS.find((x) => x.id === id);
  return MESA.TRACK_IDS.map((t) => (p.rows[t] || '').replace(/-/g, '.')).join('|');
};
const cell = (page, track, step) => page.locator(`button.celda[aria-label="${track}, paso ${step}"]`);
const verdict = (fails) => (fails.length ? 'FALLO' : 'OK');
const NOAUDIO = () => {
  delete window.AudioContext;
  delete window.webkitAudioContext;
};

async function headSteps(page, ms = 1500) {
  const seen = new Set();
  const t0 = Date.now();
  while (Date.now() - t0 < ms) {
    const s = await page.evaluate(() => [...document.querySelectorAll('.lcd-pasos i')].findIndex((i) => i.classList.contains('is-on')));
    seen.add(s);
    await sleep(40);
  }
  return [...seen].sort((a, b) => a - b);
}

function checkRandom(g) {
  const errs = [];
  const [bombo, caja, hat, diana, palmas, hatab, tom, sub] = g;
  if (bombo[0] !== 3) errs.push('bombo1');
  bombo.forEach((v, s) => { if (s && v && ![3, 6, 7, 10, 11, 14].includes(s)) errs.push(`bombo${s + 1}`); });
  if (caja[4] !== 3 || caja[12] !== 3) errs.push('caja5/13');
  caja.forEach((v, s) => { if (v && ![4, 12, 7, 14].includes(s)) errs.push(`caja${s + 1}`); if ((s === 7 || s === 14) && v && v !== 1) errs.push('caja fantasma'); });
  const stride = hat[1] ? 1 : 2;
  hat.forEach((v, s) => { const should = s % stride === 0; if (should !== Boolean(v) || (v && v > 2)) errs.push(`hat${s + 1}`); });
  if (diana.some(Boolean)) errs.push('diana');
  palmas.forEach((v, s) => { if (v && s !== 4) errs.push('palmas'); });
  hatab.forEach((v, s) => { if (v && s !== 10) errs.push('hatab'); });
  tom.forEach((v, s) => { if (v && s !== 15) errs.push('tom'); });
  sub.forEach((v, s) => { if (v && ![0, 8, 10].includes(s)) errs.push('808'); });
  return { errs, stride };
}

    await page.selectOption('.pads-tono select', 'mi');
    await press(vp, page.locator('button.pista-nombre', { hasText: 'Caja' }));
    if (vp === 'escritorio') { await page.locator('input[aria-label="Volumen de Tom"]').focus(); await page.keyboard.press('Home'); }
    await press(vp, page.locator('button.pista-solo[aria-label="Solo de Hi-hat"]'));
    await page.locator('input[aria-label="Volumen general"]').focus(); await page.keyboard.press('Home');
    await press(vp, page.locator('.pads-bancos .chip', { hasText: 'Efectos' }));
    await press(vp, page.locator('.mesa-rec'));
    const snapState = async () => ({
      grid: gridToRows(await readGrid(page)),
      info: await mesaInfo(page),
      mudas: await page.locator('button.pista-nombre').evaluateAll((b) => b.map((x) => x.getAttribute('aria-pressed')).join(',')),
      solos: await page.locator('button.pista-solo').evaluateAll((b) => b.map((x) => x.getAttribute('aria-pressed')).join(',')),
      niveles: await page.locator('.pista-nivel').evaluateAll((b) => b.map((x) => x.value).join(',')),
      bank: await page.locator('.pads-bancos .chip.is-on').textContent(),
      rec: await page.locator('.mesa-rec').getAttribute('aria-pressed'),
    });
    const edited = await snapState();
    // 2. Otra ruta y volver (navegación de la SPA)
    await press(vp, page.locator('a.navbar-brand'));
    await page.waitForURL(`${origin}/`);
    await sleep(300);
    await page.evaluate(() => { history.pushState({}, '', '/estudio'); dispatchEvent(new PopStateEvent('popstate')); });
    await page.waitForSelector('.mesa-play');
    const back = await snapState();
    await page.reload(); await page.waitForSelector('.mesa-play');
    const reloaded = await snapState();
    const same = (a) => a.grid === edited.grid && a.info.ranges[0] === edited.info.ranges[0] && a.info.ranges[1] === edited.info.ranges[1] && a.info.tono === 'mi' && a.mudas === edited.mudas && a.niveles === edited.niveles;
    const notKept = (a) => !a.solos.includes('true') && a.info.ranges[2] === '0.85' && a.bank === 'Batería' && a.rec === 'false';
    // 4. JSON corrupto
    await page.evaluate(() => localStorage.setItem('rocky-mesa-beats', '{no es json'));
    await page.reload(); await page.waitForSelector('.mesa-play');
    const corrupt = await mesaInfo(page);
    const corruptGrid = gridToRows(await readGrid(page));
    const errsCorrupt = issues.pageErrors.length;
    // 6. Enlace, editar y recargar (C-STU-02)
    await gotoStudio(page, origin, `?beat=${presetCode('house')}&bpm=124&swing=0&tono=do`);
    await press(vp, page.locator('button', { hasText: 'Sorpréndeme' }));
    await page.locator('input[aria-label="Tempo en pulsos por minuto"]').focus(); await page.keyboard.press('Home');
    const editedLink = { grid: gridToRows(await readGrid(page)), saved: await readSaved(page) };
    await page.reload(); await page.waitForSelector('.mesa-play');
    const afterLink = { grid: gridToRows(await readGrid(page)), info: await mesaInfo(page) };
    ev.push(`editado: tempo ${edited.info.ranges[0]}, swing ${edited.info.ranges[1]}, tono ${edited.info.tono}, silencios ${edited.mudas}, niveles ${edited.niveles}`);
    ev.push(`vuelta por la SPA: igual=${same(back)}, no vuelven solos/volumen/banco/REC=${notKept(back)}; tras recargar: igual=${same(reloaded)}, no vuelven=${notKept(reloaded)}`);
    ev.push(`JSON corrupto → ${corrupt.lcd.slice(0, 3).join(' / ')}, rejilla Boom bap=${corruptGrid === rowsOf('boombap')}, errores de página ${errsCorrupt}`);
    ev.push(`enlace + edición (guardado en localStorage bpm=${editedLink.saved?.bpm}) y recarga: la URL sigue con ?beat=… y la mesa vuelve a House 124 (rejilla House=${afterLink.grid === rowsOf('house')}, tempo ${afterLink.info.ranges[0]}); edición perdida=${afterLink.grid !== editedLink.grid}`);
    if (!same(back) || !notKept(back) || !same(reloaded) || !notKept(reloaded)) fails.push('recuperación');
    if (corrupt.lcd.slice(0, 3).join('|') !== '92 bpm|24% swing|Do m' || corruptGrid !== rowsOf('boombap') || errsCorrupt) fails.push('JSON corrupto');
    const c02 = afterLink.grid === rowsOf('house') && afterLink.info.ranges[0] === '124' && editedLink.saved?.bpm === 60;
    if (c02) { fails.push('tras abrir un enlace, recargar descarta lo editado porque ?beat=… sigue en la URL (C-STU-02)'); hallazgos.push('C-STU-02'); }
    ev.push(sum(issues, blocked));
    if (fails.length) await snap(page, `R-STU-17-${vp}`);
    await context.close();
    // 5. Almacenamiento bloqueado
    const bl = await open(browser, vp, origin, { initScripts: [() => { Object.defineProperty(window, 'localStorage', { get() { throw new DOMException('bloqueado', 'SecurityError'); } }); }] });
    await gotoStudio(bl.page, origin);
    await press(vp, bl.page.locator('.mesa-presets .chip', { hasText: 'Trap' }));
    await press(vp, bl.page.locator('.mesa-play'));
    await sleep(500);
    const blLcd = await lcdText(bl.page);
    await bl.page.reload(); await bl.page.waitForSelector('.mesa-play');
    const blAfter = await mesaInfo(bl.page);
    ev.push(`localStorage bloqueado: la mesa suena («${blLcd}»), tras recargar vuelve a ${blAfter.lcd[0]} (no guarda); ${sum(bl.issues, bl.blocked)}`);
    if (blLcd.trim() !== '▶ PLAY' || blAfter.lcd[0] !== '92 bpm' || bl.issues.pageErrors.length) fails.push('almacenamiento bloqueado');
    await bl.context.close();
    return { resultado: verdict(fails), evidencia: (fails.length ? `falla: ${fails.join(', ')}. ` : '') + ev.join('; '), hallazgos: hallazgos.join(', ') };
  },
});
