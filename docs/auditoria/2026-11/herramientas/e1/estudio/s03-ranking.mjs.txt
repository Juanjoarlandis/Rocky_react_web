const BOTH = ['escritorio', 'movil'];
const press = (vp, loc) => (vp === 'movil' ? loc.tap() : loc.click());
const verdict = (f) => (f.length ? 'FALLO' : 'OK');
const rowsOf = (id) => { const p = MESA.PRESETS.find((x) => x.id === id); return MESA.TRACK_IDS.map((t) => p.rows[t] || '').join('|'); };
const inPage = (page, method, p, body) => page.evaluate(async ({ method, p, body }) => {
  const r = await fetch(p, { method, headers: body ? { 'Content-Type': 'application/json' } : {}, body: body ? JSON.stringify(body) : undefined, credentials: 'same-origin' });
  let data = null; try { data = await r.json(); } catch { /* sin cuerpo */ }
  return { status: r.status, data };
}, { method, p, body });
const rowNames = (page) => page.locator('.ranking-beat .ranking-name').allTextContents();
const row = (page, name) => page.locator('.ranking-beat', { has: page.locator('.ranking-name', { hasText: new RegExp(`^${name}$`) }) });
const note = (page) => page.locator('.ranking-note').textContent();
async function reportAnon(origin, id, n) {
  const out = [];
  for (let i = 0; i < n; i += 1) out.push((await api(origin, 'POST', `/api/beats/${id}/report`)).data);
  return out;
}

async function publishUi(page, vp, { name, alias }, { open: doOpen = true } = {}) {
  if (doOpen && !(await page.locator('form.mesa-publicar').count())) await press(vp, page.locator('button', { hasText: 'Publicar en el ranking' }));
  await page.locator('input[name="nombre"]').fill(name);
  await page.locator('input[name="apodo"]').fill(alias);
  const respP = page.waitForResponse((r) => r.url().endsWith('/api/beats') && r.request().method() === 'POST', { timeout: 4000 }).catch(() => null);
  await press(vp, page.locator('form.mesa-publicar button[type="submit"]'));
  const resp = await respP;
  await sleep(250);
  const n = page.locator('.mesa-publicar-nota');
  return {
    status: resp?.status() ?? null,
    body: resp ? await resp.json().catch(() => null) : null,
    req: resp ? resp.request().postDataJSON() : null,
    setCookie: resp ? await resp.headerValue('set-cookie') : null,
    note: (await n.textContent()).trim(),
    error: await n.evaluate((e) => e.classList.contains('is-error')),
    color: await n.evaluate((e) => getComputedStyle(e).color),
    name: await page.locator('input[name="nombre"]').inputValue(),
  };
}
