import { startInstance, launchBrowser } from '../harness.mjs';
import { open, audioState, meterReset, gotoStudio, sum, sleep } from './common.mjs';
const inst = await startInstance(3205, { BEATS_PUBLISH_RATE_LIMIT_MAX: '100' });
const b = await launchBrowser();
try {
  const { page, issues, blocked, context } = await open(b, 'escritorio', inst.origin);
  await gotoStudio(page, inst.origin);
  console.log('title', await page.title());
  console.log('before', JSON.stringify(await audioState(page)));
  await page.click('.mesa-play');
  await sleep(300);
  await meterReset(page);
  await sleep(1500);
  console.log('playing', JSON.stringify(await audioState(page)));
  console.log('lcd', await page.locator('.lcd-rec').textContent(), await page.locator('.mesa-play').getAttribute('class'));
  await page.click('.mesa-play');
  await sleep(500);
  await meterReset(page);
  await sleep(500);
  console.log('stopped', JSON.stringify(await audioState(page)));
  console.log(sum(issues, blocked));
  await context.close();
} finally { await b.close(); inst.stop(); }
