// Ejecuta casos (ficha × viewport) con una instancia propia por caso y guarda
// los resultados en resultados-e1/estudio-parts/<script>.json (se compilan al
// TSV del área con compile.mjs, así las repeticiones no duplican líneas).
import fs from 'node:fs';
import path from 'node:path';
import { once } from 'node:events';
import { SCRATCH, startInstance, launchBrowser } from '../harness.mjs';
import { encodePattern, patternFromRows, emptyPattern } from '../../../../../../../../RUTA/AL/REPO/shared/beat-codec.mjs';

export { encodePattern, patternFromRows, emptyPattern };
const MESA_SRC = fs.readFileSync('/RUTA/AL/REPO/src/data/mesa.js', 'utf8');
export const MESA = await import('data:text/javascript;base64,' + Buffer.from(MESA_SRC).toString('base64'));
export const codeOf = (rows) => encodePattern(patternFromRows(rows, MESA.TRACK_IDS));
export const presetCode = (id) => codeOf(MESA.PRESETS.find((p) => p.id === id).rows);

const PARTS = path.join(SCRATCH, 'resultados-e1', 'estudio-parts');

export const BASE_ENV = {
  BEATS_PUBLISH_RATE_LIMIT_MAX: '100',
  BEATS_VOTE_RATE_LIMIT_MAX: '1000',
  CHAT_RATE_LIMIT_MAX: '50',
  SHOPIFY_RATE_LIMIT_MAX: '2000',
  AVISOS_RATE_LIMIT_MAX: '100',
};

export async function start(port = 3205, env = BASE_ENV) {
  const inst = await startInstance(port, env);
  return {
    ...inst,
    async stop() {
      if (inst.child.exitCode !== null) return;
      inst.child.kill('SIGTERM');
      await Promise.race([once(inst.child, 'exit'), new Promise((r) => setTimeout(r, 4000))]);
    },
  };
}

export async function run(script, cases) {
  const only = process.argv.slice(2);
  fs.mkdirSync(PARTS, { recursive: true });
  const file = path.join(PARTS, `${script}.json`);
  const store = fs.existsSync(file) ? JSON.parse(fs.readFileSync(file, 'utf8')) : {};
  const browser = await launchBrowser();
  try {
    for (const c of cases) {
      for (const vp of c.vps) {
        const key = `${c.id}|${vp}`;
        if (only.length && !only.some((o) => key === o || key.startsWith(o + '|') || (o.includes('|') && key === o))) continue;
        let inst = null;
        try {
          if (c.instance !== false) inst = await start(c.port ?? 3205, { ...BASE_ENV, ...(c.env || {}) });
          const out = await c.fn({ browser, vp, inst, origin: inst?.origin });
          store[key] = { id: c.id, viewport: vp, resultado: out.resultado, evidencia: out.evidencia, hallazgos: out.hallazgos || '' };
          fs.writeFileSync(file, JSON.stringify(store, null, 1));
          console.log(`\n== ${key} → ${out.resultado}\n   ${out.evidencia}\n   [${out.hallazgos || ''}]`);
        } catch (e) {
          console.error(`\n!! ${key} ERROR`, e.stack);
        } finally {
          await inst?.stop();
        }
      }
    }
  } finally {
    await browser.close();
  }
}

// Llamadas a la API desde Node (como curl): Origin opcional, sin cookie salvo que se pase.
export async function api(origin, method, p, body, { cookie, originHeader = origin, headers: extra = {} } = {}) {
  const headers = { Accept: 'application/json', ...extra };
  if (originHeader) headers.Origin = originHeader;
  if (body !== undefined) headers['Content-Type'] = 'application/json';
  if (cookie) headers.Cookie = cookie;
  const res = await fetch(origin + p, { method, headers, body: body !== undefined ? JSON.stringify(body) : undefined });
  const text = await res.text();
  let data = null;
  try { data = JSON.parse(text); } catch { data = text; }
  return { status: res.status, data, headers: Object.fromEntries(res.headers), setCookie: res.headers.getSetCookie() };
}

let seedN = 0;
export async function seed(origin, { code, rows, preset, bpm = 100, swing = 0, tono = 'do', name, alias = 'semilla' } = {}) {
  seedN += 1;
  const c = code || (rows ? codeOf(rows) : presetCode(preset || 'trap'));
  const r = await api(origin, 'POST', '/api/beats', { code: c, bpm, swing, tono, name: name || `Semilla ${seedN}`, alias });
  if (r.status !== 201 && r.status !== 200) throw new Error(`seed falló ${r.status} ${JSON.stringify(r.data)}`);
  return r.data.beat;
}
