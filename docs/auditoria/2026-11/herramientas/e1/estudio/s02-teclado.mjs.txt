import { run, seed } from './runner.mjs';
import { open, sum, snap, sleep, gotoStudio, audioState, appendTsv } from './common.mjs';

const verdict = (f) => (f.length ? 'FALLO' : 'OK');
const ESC = ['escritorio'];

const describe = () => {
  const el = document.activeElement;
  if (!el || el === document.body) return { d: '<body>' };
  const cs = getComputedStyle(el);
  return {
    d: (el.tagName === 'MAIN' ? `main#${el.id}` : '') || el.getAttribute('aria-label') || (el.tagName === 'INPUT' ? `input[${el.name || el.type}]` : el.tagName === 'SELECT' ? 'select' : el.textContent.trim().replace(/\s+/g, ' ').slice(0, 30)),
    tag: el.tagName,
    fv: el.matches(':focus-visible'),
    outline: `${cs.outlineStyle} ${cs.outlineWidth} ${cs.outlineColor}`,
    shadow: cs.boxShadow !== 'none',
    visible: el.getBoundingClientRect().width > 0 && cs.visibility !== 'hidden',
  };
};
const focusOk = (f) => f.fv && ((f.outline.startsWith('solid') && parseFloat(f.outline.split(' ')[1]) >= 2) || f.shadow);

async function tabs(page, n) {
  const out = [];
  for (let i = 0; i < n; i += 1) {
    await page.keyboard.press('Tab');
    out.push(await page.evaluate(describe));
  }
  return out;
}
