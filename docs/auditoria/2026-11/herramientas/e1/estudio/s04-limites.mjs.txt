const press = (vp, loc) => (vp === 'movil' ? loc.tap() : loc.click());
const verdict = (f) => (f.length ? 'FALLO' : 'OK');
const inPage = (page, method, p, body, headers = {}) => page.evaluate(async ({ method, p, body, headers }) => {
  const r = await fetch(p, { method, headers: { ...(body ? { 'Content-Type': 'application/json' } : {}), ...headers }, body: body ? JSON.stringify(body) : undefined, credentials: 'same-origin' });
  let data = null; try { data = await r.json(); } catch { /* sin cuerpo */ }
  return { status: r.status, data, retry: r.headers.get('retry-after') };
}, { method, p, body, headers });
const row = (page, name) => page.locator('.ranking-beat', { has: page.locator('.ranking-name', { hasText: new RegExp(`^${name}$`) }) });
const ranNote = async (page) => (await page.locator('.ranking-note').textContent()).trim();
const pubNote = (page) => page.locator('.mesa-publicar-nota').evaluate((e) => ({ text: e.textContent.trim(), error: e.classList.contains('is-error'), color: getComputedStyle(e).color }));
const beatBody = (bpm, extra = {}) => ({ code: presetCode('funk'), bpm, swing: 0.16, tono: 'do', name: `Ritmo ${bpm}`, alias: 'limites', ...extra });

async function uiPublish(page, vp, name = 'Desde la web', alias = 'limites') {
  if (!(await page.locator('form.mesa-publicar').count())) await press(vp, page.locator('button', { hasText: 'Publicar en el ranking' }));
  await page.locator('input[name="nombre"]').fill(name);
  await page.locator('input[name="apodo"]').fill(alias);
  const rp = page.waitForResponse((r) => r.url().endsWith('/api/beats') && r.request().method() === 'POST');
  await press(vp, page.locator('form.mesa-publicar button[type="submit"]'));
  const r = await rp;
  await sleep(300);
  return { status: r.status(), body: await r.json().catch(() => null), retry: r.headers()['retry-after'], note: await pubNote(page) };
}

    return { resultado: verdict(fails), evidencia: (fails.length ? `falla: ${fails.join(', ')}. ` : '') + ev.join('; '), hallazgos: hallazgos.join(', ') };
  },
});

await run('s04-limites', cases);
