// R-CAT-08 (lightbox con ratón/dedo), R-CAT-09 (lightbox con teclado), pasada de teclado (TECLADO-*)
// y tiempos de descarga y decodificación de las fotos grandes.
import { startInstance, launchBrowser, openPage } from '../harness.mjs';
import { Checks, rec, waitStore, activeDesc, focusStyle, pixelFocusDiff } from './util.mjs';

const ONLY = process.argv[2] ? new Set(process.argv[2].split(',')) : null;
const want = (id) => !ONLY || ONLY.has(id);
const VPS = (process.env.VPS || 'escritorio,movil').split(',');

const inst = await startInstance(3202, { AVISOS_RATE_LIMIT_MAX: '100', SHOPIFY_RATE_LIMIT_MAX: '2000' });
const browser = await launchBrowser();
const O = inst.origin;
inst.child.on('exit', (code, sig) => console.error(`!! la instancia ${O} terminó (code ${code}, señal ${sig}). Últimas líneas:\n${inst.log.join('').slice(-800)}`));

const dialogState = (page) =>
  page.evaluate(() => {
    const d = document.querySelector('dialog.lightbox');
    if (!d) return { open: false };
    const im = d.querySelector('img');
    const r = im.getBoundingClientRect();
    const dr = d.getBoundingClientRect();
    return {
      open: d.open,
      modal: d.matches(':modal'),
      label: d.getAttribute('aria-label'),
      bg: getComputedStyle(d).backgroundColor,
      dialogRect: [Math.round(dr.left), Math.round(dr.top), Math.round(dr.width), Math.round(dr.height)],
      img: { w: im.offsetWidth, h: im.offsetHeight, bw: Math.round(r.width), left: Math.round(r.left), top: Math.round(r.top), src: im.getAttribute('src'), nat: im.naturalWidth },
      vh: innerHeight,
      vw: innerWidth,
      closeName: d.querySelector('button')?.getAttribute('aria-label'),
      bodyOverflow: document.body.style.overflow,
      imgs: d.querySelectorAll('img').length,
    };
  });

async function press(page, vp, locator) {
  if (vp === 'movil') await locator.tap();
  else await locator.click();
}
async function pressAt(page, vp, x, y) {
  if (vp === 'movil') await page.touchscreen.tap(x, y);
  else await page.mouse.click(x, y);
}

try {
  // ---------------- R-CAT-08 ----------------
  if (want('R-CAT-08')) {
    for (const vp of VPS) {
      const c = new Checks();
      const { page, issues, context } = await openPage(browser, vp, { origin: O });
      const scenario = async (label, url, opener, expectedLabel, full) => {
        await page.goto(O + url);
        await waitStore(page);
        const btn = page.locator(opener).first();
        await btn.scrollIntoViewIfNeeded();
        await press(page, vp, btn);
        await page.waitForSelector('dialog.lightbox[open]');
        await page.waitForTimeout(300);
        const s = await dialogState(page);
        c.ok(`${label}: diálogo modal a pantalla completa`, s.open && s.modal && s.dialogRect[2] === s.vw && s.dialogRect[3] === s.vh, `${JSON.stringify(s.dialogRect)} vw ${s.vw} vh ${s.vh}`);
        c.ok(`${label}: velo oscuro`, /rgba\(\d+, \d+, \d+, 0\.85\)/.test(s.bg), s.bg);
        c.ok(`${label}: foto ≤ 860 px y ≤ 84vh`, s.img.w <= 860 && s.img.h <= Math.ceil(s.vh * 0.84), `caja ${s.img.w}×${s.img.h} (84vh=${Math.round(s.vh * 0.84)}; ${s.img.bw}px girada), natural ${s.img.nat}px`);
        c.eq(`${label}: nombre del diálogo`, s.label, expectedLabel);
        c.eq(`${label}: botón Cerrar`, s.closeName, 'Cerrar');
        if (!full) {
          await page.locator('dialog.lightbox [aria-label="Cerrar"]').click();
          await page.waitForTimeout(200);
          c.ok(`${label}: × cierra`, !(await dialogState(page)).open);
          return;
        }
        // Pulsar la foto no cierra
        const ib = await page.locator('dialog.lightbox img').boundingBox();
        await pressAt(page, vp, ib.x + ib.width / 2, ib.y + ib.height / 2);
        await page.waitForTimeout(250);
        c.ok(`${label}: pulsar la foto no cierra`, (await dialogState(page)).open);
        // Pulsar fuera
        await pressAt(page, vp, Math.max(4, ib.x / 2), Math.min(s.vh - 10, ib.y + ib.height + (s.vh - ib.y - ib.height) / 2));
        await page.waitForTimeout(250);
        c.ok(`${label}: pulsar fuera cierra`, !(await dialogState(page)).open);
        // Abrir y cerrar con ×
        await press(page, vp, btn);
        await page.waitForSelector('dialog.lightbox[open]');
        // Bloqueo del desplazamiento
        const before = await page.evaluate(() => Math.round(scrollY));
        const st = await dialogState(page);
        await page.mouse.move(st.vw / 2, 30);
        await page.mouse.wheel(0, 700);
        await page.waitForTimeout(400);
        const afterWheel = await page.evaluate(() => Math.round(scrollY));
        let afterTouch = afterWheel;
        if (vp === 'movil') {
          // Arrastre del dedo por el velo (CDP), como al desplazar en un móvil
          const cdp = await context.newCDPSession(page);
          await cdp.send('Input.dispatchTouchEvent', { type: 'touchStart', touchPoints: [{ x: 16, y: st.vh - 100 }] });
          for (let y = st.vh - 120; y >= 200; y -= 40) {
            await cdp.send('Input.dispatchTouchEvent', { type: 'touchMove', touchPoints: [{ x: 16, y }] });
            await page.waitForTimeout(16);
          }
          await cdp.send('Input.dispatchTouchEvent', { type: 'touchEnd', touchPoints: [] });
          await page.waitForTimeout(700);
          afterTouch = await page.evaluate(() => Math.round(scrollY));
          await cdp.detach();
        }
        const htmlOv = await page.evaluate(() => { const h = getComputedStyle(document.documentElement); return `${h.overflowX}/${h.overflowY}`; });
        c.ok(`${label}: el fondo no se desplaza`, afterWheel === before && afterTouch === afterWheel, `body.style.overflow=${st.bodyOverflow}, html overflow ${htmlOv}; scrollY ${before}→${afterWheel} con la rueda${vp === 'movil' ? `, →${afterTouch} con el dedo` : ''}; diálogo abierto ${(await dialogState(page)).open}`);
        await press(page, vp, page.locator('dialog.lightbox [aria-label="Cerrar"]'));
        await page.waitForTimeout(250);
        const closed = await dialogState(page);
        const ov = await page.evaluate(() => document.body.style.overflow);
        c.ok(`${label}: × cierra y restaura el overflow`, !closed.open && ov === '', `overflow=«${ov}»`);
        await page.mouse.move(200, 400);
        await page.mouse.wheel(0, 500);
        await page.waitForTimeout(400);
        const after2 = await page.evaluate(() => Math.round(scrollY));
        c.ok(`${label}: se vuelve a desplazar tras cerrar`, after2 !== before, `scrollY ${before}→${after2}`);
      };
      await scenario('portada Night Runner', '/', 'button[aria-label="Ver Night Runner en grande"]', 'Camiseta oversize negra Night Runner con mapa urbano nocturno', true);
      await scenario('costa-035 Solar Club', '/products/costa-035', 'button[aria-label="Ver Solar Club en grande"]', 'Camiseta oversize color arena Solar Club con sol skater', true);
      await scenario('ficha smoker-edition', '/product/smoker-edition', 'button.detail-media', 'Smoker Edition', true);
      await scenario('portada Rockydz boyz', '/', 'button[aria-label="Ver Rockydz boyz en grande"]', 'Rockydz boyz', false);

      // Tiempos de las fotos grandes (contexto nuevo, sin caché): descarga y decodificación
      const t = await context.newPage();
      const cdp = await context.newCDPSession(t);
      await t.goto(O + '/robots.txt');
      const measure = () => t.evaluate(async () => {
        const out = {};
        for (const src of ['/products/rocky35-camel.webp', '/products/rockydz-boyz.webp', '/products/rocky-night-runner.webp', '/products/rocky-racing.webp']) {
          const times = [];
          for (let i = 0; i < 3; i += 1) {
            const url = `${src}?m=${Math.random()}`;
            const t0 = performance.now();
            const im = new Image();
            im.src = url;
            await im.decode();
            const t2 = performance.now();
            const e = performance.getEntriesByName(new URL(url, location.href).href)[0];
            const dl = e ? e.responseEnd - e.startTime : 0;
            times.push([Math.round(dl), Math.round(t2 - t0 - dl), im.naturalWidth]);
          }
          out[src] = times;
        }
        return out;
      });
      const m1 = await measure();
      await cdp.send('Emulation.setCPUThrottlingRate', { rate: 4 });
      const m4 = await measure();
      await cdp.send('Emulation.setCPUThrottlingRate', { rate: 1 });
      const fmt = (m) => Object.entries(m).map(([k, v]) => `${k.split('/').pop()} ${v[0][2]}px: descarga local ${v[0][0]} ms, primera decodificación ${v[0][1]} ms`).join('; ');
      c.note(`tiempos (servidor local; las decodificaciones repetidas salen de caché y no cuentan) CPU x1: ${fmt(m1)}`);
      c.note(`tiempos con CPU x4: ${fmt(m4)}`);
      // Tamaño pintado frente al natural en la portada y la ficha
      await t.close();
      await page.goto(O + '/');
      await waitStore(page);
      await page.locator('button[aria-label="Ver Smoker Edition en grande"]').scrollIntoViewIfNeeded();
      await page.waitForTimeout(800);
      const sizes = await page.evaluate(() => [...document.querySelectorAll('img')].filter((i) => /rocky35-camel|rockydz-boyz/.test(i.src)).map((i) => `${i.getAttribute('src')} pintada ${Math.round(i.getBoundingClientRect().width)}px de ${i.naturalWidth}px (srcset ${i.getAttribute('srcset') ? 'sí' : 'no'}, sizes ${i.getAttribute('sizes') ? 'sí' : 'no'}, loading ${i.getAttribute('loading')})`));
      c.note(`portada: ${sizes.join('; ')}`);
      await rec({ id: 'R-CAT-08', viewport: vp, checks: c, issues, page, hallazgos: 'C-CAT-04, C-CAT-11, H-003' });
      await context.close();
    }
  }

  // ---------------- R-CAT-09 ----------------
  if (want('R-CAT-09')) {
    const vp = 'escritorio';
    const c = new Checks();
    const { page, issues, context } = await openPage(browser, vp, { origin: O });
    const tabTo = async (label, max = 80) => {
      for (let i = 1; i <= max; i += 1) {
        await page.keyboard.press('Tab');
        const l = await page.evaluate(() => document.activeElement?.getAttribute('aria-label'));
        if (l === label) return i;
      }
      return -1;
    };
    const kbScenario = async (url, label, name) => {
      await page.goto(O + url);
      await waitStore(page);
      const n = await tabTo(label);
      c.ok(`${name}: se llega con Tab a «${label}»`, n > 0, `${n} tabulaciones`);
      const fs = await focusStyle(page);
      const px = await pixelFocusDiff(page);
      c.ok(`${name}: foco visible en la foto`, fs?.focusVisible && (px.arriba + px.abajo + px.izquierda + px.derecha + px.dentro) > 30, `outline ${fs?.outline}; recortado por ${fs?.clippedBy.join(' ') || 'nada'}; píxeles cambiados ${JSON.stringify(px)}`);
      await page.keyboard.press('Enter');
      await page.waitForSelector('dialog.lightbox[open]');
      c.eq(`${name}: Enter abre y el foco va a Cerrar`, await page.evaluate(() => document.activeElement?.getAttribute('aria-label')), 'Cerrar');
      const s0 = await dialogState(page);
      await page.keyboard.press('Tab');
      const afterTab = await page.evaluate(() => ({ inDialog: !!document.activeElement?.closest('dialog'), desc: document.activeElement === document.body ? 'body' : document.activeElement?.getAttribute('aria-label') || document.activeElement?.tagName }));
      await page.keyboard.press('Shift+Tab');
      const afterShift = await page.evaluate(() => ({ inDialog: !!document.activeElement?.closest('dialog'), desc: document.activeElement === document.body ? 'body' : document.activeElement?.getAttribute('aria-label') || document.activeElement?.tagName }));
      c.ok(`${name}: Tab y Mayús+Tab no salen a la página`, (afterTab.inDialog || afterTab.desc === 'body') && (afterShift.inDialog || afterShift.desc === 'body'), `Tab→${afterTab.desc}, Mayús+Tab→${afterShift.desc}`);
      // vuelve a Cerrar si el foco se fue a la interfaz del navegador
      await page.evaluate(() => document.querySelector('dialog.lightbox button')?.focus());
      await page.keyboard.press('ArrowRight');
      await page.keyboard.press('ArrowLeft');
      await page.waitForTimeout(200);
      const s1 = await dialogState(page);
      c.ok(`${name}: las flechas no hacen nada`, s1.open && s1.img.src === s0.img.src && s1.imgs === 1, `abierto ${s1.open}, ${s1.imgs} imagen`);
      await page.keyboard.press('Escape');
      await page.waitForTimeout(250);
      const back = await page.evaluate(() => document.activeElement?.getAttribute('aria-label'));
      c.ok(`${name}: Escape cierra y el foco vuelve a la foto`, !(await dialogState(page)).open && back === label, `foco en «${back}»`);
      await page.keyboard.press('Space');
      await page.waitForTimeout(250);
      c.ok(`${name}: Espacio abre`, (await dialogState(page)).open);
      await page.keyboard.press('Escape');
      await page.waitForTimeout(250);
      c.ok(`${name}: Escape tras Espacio cierra`, !(await dialogState(page)).open);
    };
    await kbScenario('/', 'Ver RockyRacing en grande', 'portada');
    await kbScenario('/product/rocky-airwave', 'Ver Airwave en grande', 'ficha rocky-airwave');
    await page.goto(O + '/product/35-red');
    await waitStore(page);
    let reached = false;
    for (let i = 0; i < 70; i += 1) {
      await page.keyboard.press('Tab');
      if (await page.evaluate(() => document.activeElement?.classList.contains('detail-media'))) reached = true;
    }
    c.ok('35-red: Tab se salta la foto deshabilitada', !reached);
    await rec({ id: 'R-CAT-09', viewport: vp, checks: c, issues, page, hallazgos: '' });
    await context.close();
  }

  // ---------------- Pasada de teclado ----------------
  if (want('TECLADO')) {
    const screens = [
      ['listado', '/', null],
      ['ficha', '/product/rockyracing', null],
      ['menudrop', '/menudrop', null],
      ['avisos', '/', '[data-testid="locked-designs"]'],
    ];
    for (const [name, url, scope] of screens) {
      const c = new Checks();
      const { page, issues, context } = await openPage(browser, 'escritorio', { origin: O, reducedMotion: 'reduce' });
      await page.goto(O + url);
      await waitStore(page);
      await page.waitForTimeout(500);
      if (scope) {
        // Empieza justo antes del bloque: foco en el último control de la tarjeta anterior
        await page.evaluate((s) => {
          const prev = document.querySelector(s).previousElementSibling;
          const f = [...prev.querySelectorAll('a,button:not([disabled])')].pop();
          f.focus();
        }, scope);
      }
      const order = [];
      const problems = [];
      const partial = [];
      for (let i = 0; i < 90; i += 1) {
        await page.keyboard.press('Tab');
        const info = await page.evaluate((s) => {
          const el = document.activeElement;
          const inMain = !!el?.closest('#main-content');
          const inScope = s ? !!el?.closest(s) : inMain;
          const inFooter = !!el?.closest('footer');
          const r = el.getBoundingClientRect();
          const hidden = r.width === 0 || r.height === 0 || getComputedStyle(el).visibility === 'hidden' || !!el.closest('[aria-hidden="true"]');
          return { inMain, inScope, inFooter, hidden };
        }, scope);
        const desc = await activeDesc(page);
        if (scope && !info.inScope && order.length) break;
        if (!scope && info.inFooter) {
          order.push(`[pie] ${desc}`);
          break;
        }
        if (!info.inScope) {
          order.push(`(${desc})`);
          continue;
        }
        const fs = await focusStyle(page);
        const px = await pixelFocusDiff(page);
        const band = px ? px.arriba + px.abajo + px.izquierda + px.derecha : 0;
        const sides = px ? ['arriba', 'abajo', 'izquierda', 'derecha'].filter((k) => px[k] > 3) : [];
        order.push(desc);
        if (info.hidden) problems.push(`${desc} oculto pero enfocable`);
        if (!fs?.focusVisible || (band < 20 && (px?.dentro ?? 0) < 30)) problems.push(`${desc} sin foco visible (outline ${fs?.outline}, píxeles ${JSON.stringify(px)})`);
        else if (sides.length < 4) partial.push(`${desc}: anillo solo en ${sides.join('+') || 'ningún lado'}${fs?.clippedBy?.length ? `, recortado por ${fs.clippedBy.join(' ')}` : ''}${px?.dentro ? `, cambio interior ${px.dentro}px` : ''}`);
      }
      if (name === 'avisos') {
        const trap = await page.evaluate(() => {
          const t = document.querySelector('[data-testid="locked-designs"] input[name="apodo"]');
          const r = t.getBoundingClientRect();
          return { tabIndex: t.tabIndex, ariaHidden: t.getAttribute('aria-hidden'), visible: r.width > 1 && r.height > 1 && getComputedStyle(t).opacity !== '0' };
        });
        c.ok('campo trampa fuera del tabulador y oculto', trap.tabIndex === -1 && trap.ariaHidden === 'true' && !trap.visible, JSON.stringify(trap));
        c.eq('orden en el bloque', order.filter((x) => !x.startsWith('(')), ['input[email]«Tu email para el aviso del drop»', 'button«Avísame»', 'input[checkbox]«consentimiento»', 'a«Más en privacidad»→/legal/privacidad']);
      }
      c.ok('todos los controles con foco visible', problems.length === 0, problems.join('; '));
      if (partial.length) c.note(`anillo incompleto: ${partial.join(' · ')}`);
      c.note(`orden de tabulación: ${order.join(' → ')}`);
      await rec({ id: `TECLADO-${name}`, viewport: 'escritorio', checks: c, issues, page, hallazgos: '' });
      await context.close();
    }
    // Lightbox
    {
      const c = new Checks();
      const { page, issues, context } = await openPage(browser, 'escritorio', { origin: O, reducedMotion: 'reduce' });
      await page.goto(O + '/');
      await waitStore(page);
      let n = 0;
      for (n = 1; n < 60; n += 1) {
        await page.keyboard.press('Tab');
        if ((await page.evaluate(() => document.activeElement?.getAttribute('aria-label'))) === 'Ver Smoker Edition en grande') break;
      }
      await page.keyboard.press('Enter');
      await page.waitForSelector('dialog.lightbox[open]');
      await page.waitForTimeout(400);
      const fs = await focusStyle(page);
      const px = await pixelFocusDiff(page);
      c.eq('foco inicial en Cerrar', await page.evaluate(() => document.activeElement?.getAttribute('aria-label')), 'Cerrar');
      c.ok('foco visible en Cerrar', fs?.focusVisible && px && px.arriba + px.abajo + px.izquierda + px.derecha > 20, `outline ${fs?.outline}; píxeles ${JSON.stringify(px)}`);
      await page.keyboard.press('Tab');
      const t1 = await activeDesc(page);
      await page.keyboard.press('Tab');
      const t2 = await activeDesc(page);
      c.note(`Tab dentro del diálogo: ${t1} → ${t2}`);
      await page.evaluate(() => document.querySelector('dialog.lightbox button')?.focus());
      await page.keyboard.press('ArrowLeft');
      await page.keyboard.press('ArrowRight');
      c.ok('flechas: sin efecto (una sola imagen)', await page.evaluate(() => document.querySelector('dialog.lightbox')?.open && document.querySelectorAll('dialog.lightbox img').length === 1));
      await page.keyboard.press('Escape');
      await page.waitForTimeout(250);
      c.eq('Escape cierra y devuelve el foco', await page.evaluate(() => [!!document.querySelector('dialog.lightbox'), document.activeElement?.getAttribute('aria-label')]), [false, 'Ver Smoker Edition en grande']);
      await rec({ id: 'TECLADO-lightbox', viewport: 'escritorio', checks: c, issues, page, hallazgos: '' });
      await context.close();
    }
  }
} finally {
  await browser.close();
  inst.stop();
}
