// R-DROP-04…09 y R-DROP-11: altas de avisos en el navegador (instancia con límites altos).
// Una instancia nueva por viewport para empezar con el almacén vacío.
import { startInstance, launchBrowser, openPage } from '../harness.mjs';
import { Checks, rec, waitStore, text } from './util.mjs';

const VPS = (process.env.VPS || 'escritorio,movil').split(',');
const EMAIL = 'auditoria+drop@rocky.test';
const CONSENT = 'Usaremos tu email sólo para avisarte de este drop; después, la lista se borra.';

const browser = await launchBrowser();

// Registra peticiones y respuestas de /api/avisos
function spyAvisos(page) {
  const log = [];
  page.on('request', (r) => {
    if (r.url().endsWith('/api/avisos')) log.push({ kind: 'req', body: r.postData() });
  });
  page.on('response', async (r) => {
    if (!r.url().endsWith('/api/avisos')) return;
    let body = null;
    try {
      body = await r.json();
    } catch {
      /* sin json */
    }
    log.push({ kind: 'res', status: r.status(), body, headers: r.headers() });
  });
  return log;
}
const resetIssues = (i) => Object.values(i).forEach((a) => a.splice(0));
const lastRes = (log) => [...log].reverse().find((x) => x.kind === 'res');
const lastReq = (log) => {
  const r = [...log].reverse().find((x) => x.kind === 'req');
  return r ? JSON.parse(r.body) : null;
};

const listo = (page, scope) =>
  page.evaluate((s) => {
    const root = document.querySelector(s);
    const l = root?.querySelector('[data-testid="drop-aviso-listo"]');
    return l
      ? { listo: true, sello: l.querySelector('.drop-aviso-sello')?.textContent.trim(), nota: l.querySelector('.drop-aviso-nota')?.textContent.trim(), img: /corriendo-bolsa/.test(l.querySelector('img')?.getAttribute('src') || ''), form: !!root.querySelector('form.drop-aviso') }
      : { listo: false, form: !!root?.querySelector('form.drop-aviso') };
  }, scope);

      c.ok('el log no lleva el email', !inst.log.join('').includes('rocky.test'));

    }

    // ---------------- R-DROP-05 ----------------
    resetIssues(issues);
    {
      const c = new Checks();
      await page.goto(O + '/');
      await waitStore(page);
      const link = page.locator('article', { hasText: 'RockyRacing' }).getByRole('link', { name: 'Avísame' });
      await link.scrollIntoViewIfNeeded();
      if (vp === 'movil') await link.tap();
      else await link.click();
      await page.waitForURL('**/product/rockyracing#aviso');
      await page.waitForSelector('#aviso input[type=email]');
      await page.waitForTimeout(1200);
      const f = await page.evaluate(() => {

        const r = el.getBoundingClientRect();
        return { email: el.matches('#aviso input[type=email]'), center: Math.round(r.top + r.height / 2), vh: innerHeight };

      c.ok('llega a /product/rockyracing#aviso con el foco en el email', f.email, JSON.stringify(f));

      c.eq('recado', await listo(page, '#aviso'), { listo: true, sello: 'Recado apuntado', nota: 'El Recadero te lo trae el día del drop.', img: true, form: false });

      await page.locator('#aviso button[type=submit]').click();
      await page.waitForSelector('#aviso [data-testid="drop-aviso-listo"]');
      c.eq('alta 35-red', [lastReq(log)?.producto, lastRes(log)?.status, lastRes(log)?.body], ['35-red', 200, { ok: true, duplicate: false, product: '35-red' }]);

    }

    // ---------------- R-DROP-06 ----------------
    resetIssues(issues);
    {
      const c = new Checks();
      const before = log.length;
      await page.goto(O + '/product/rockyracing');
      await waitStore(page);
      await page.waitForSelector('#aviso');
      c.eq('ficha con recibo', await listo(page, '#aviso'), { listo: true, sello: 'Recado apuntado', nota: 'El Recadero te lo trae el día del drop.', img: true, form: false });
      c.eq('no se pregunta al servidor', log.length - before, 0);
      await page.evaluate(() => window.scrollTo(0, 0));
      const btn = page.locator('.detail-buy > button', { hasText: 'Avísame' });
      await btn.scrollIntoViewIfNeeded();
      await page.waitForTimeout(300);
      const y0 = await page.evaluate(() => Math.round(scrollY));

      await page.waitForTimeout(800);
      const s2 = await page.evaluate(() => ({ y: Math.round(scrollY), active: document.activeElement?.textContent?.trim().slice(0, 20) || document.activeElement?.tagName }));
      c.ok('paso 2: «Avísame» de compra no hace nada visible', s2.y === y0, `scrollY ${y0}→${s2.y}, foco en «${s2.active}»`);

      await link.scrollIntoViewIfNeeded();
      await link.click();
      await page.waitForURL('**/product/rockyracing#aviso');
      await page.waitForTimeout(1000);
      const s3 = await page.evaluate(() => ({ y: Math.round(scrollY), active: document.activeElement === document.body ? 'body' : document.activeElement?.tagName, avisoTop: Math.round(document.querySelector('#aviso').getBoundingClientRect().top), vh: innerHeight }));

      c.ok('paso 4: sin recibo vuelve el formulario', (await listo(page, '#aviso')).form);
      await rec({ id: 'R-DROP-06', viewport: vp, checks: c, issues, page, hallazgos: 'C-CAT-08' });

    // ---------------- R-DROP-07 ----------------
    resetIssues(issues);
    {
      const c = new Checks();
      await page.goto(O + '/product/dwag');
      await waitStore(page);
      const f = page.locator('#aviso form.drop-aviso');
      const before = log.filter((x) => x.kind === 'req').length;
      const email = f.locator('input[type=email]');
      const submit = f.locator('button[type=submit]');

      c.eq('ninguna petición a /api/avisos', sent, 0);

      await email.focus();
      const long = `${'a'.repeat(280)}@rocky.test`;
      await page.keyboard.insertText(long);

      await rec({ id: 'R-DROP-07', viewport: vp, checks: c, issues, page, hallazgos: '' });

      await f.locator('button[type=submit]').click();

      const st = await page.evaluate(() => {

      const addedAfter = inst.log.join('').split('Aviso de drop apuntado').length - 1;

    resetIssues(issues);

      await f.locator('button[type=submit]').click();
      await page.waitForSelector('#aviso .drop-aviso-fallo');

        const f2 = document.querySelector('#aviso form.drop-aviso');
        return { role: a?.getAttribute('role'), msg: a?.textContent.trim(), value: f2?.querySelector('input[type=email]').value, enabled: f2 ? !f2.querySelector('button[type=submit]').disabled : null, checked: f2?.querySelector('input[name=consentimiento]').checked };

      await page.waitForTimeout(800);
      c.eq('Express parado (sin Vite: fallo de red)', (await st()).msg, 'No hemos podido apuntarte. Prueba otra vez.');

  } finally {
    await context.close();
    inst.stop();
    await new Promise((r) => setTimeout(r, 800));
  }
}
await browser.close();
