async function cards(page) {
  return page.evaluate(() =>
    [...document.querySelectorAll('.product-grid > article')].map((a) => ({
      locked: a.classList.contains('product-card--locked'),
      title: a.querySelector('.product-title')?.textContent.trim(),
      badge: a.querySelector('.badge')?.textContent.trim() ?? null,
      photo: a.querySelector('button.product-media')?.getAttribute('aria-label') ?? null,
      photoDisabled: a.querySelector('button.product-media')?.disabled ?? null,
      actions: [...a.querySelectorAll('.product-actions a, .product-actions button')].map((b) => `${b.textContent.trim()}${b.disabled ? '(deshabilitado)' : ''}`),
    }))
  );
}

async function activeNav(page, vp) {
  if (vp === 'movil') {
    await page.getByRole('button', { name: 'Abrir el menú' }).click();
    await page.waitForSelector('.navbar-sheet');
    const act = await texts(page, '.navbar-sheet-link.is-active');
    await page.getByRole('button', { name: 'Cerrar el menú' }).click();
    await page.waitForTimeout(200);
    return act;
  }
  return texts(page, '.navbar-link.is-active');
}
