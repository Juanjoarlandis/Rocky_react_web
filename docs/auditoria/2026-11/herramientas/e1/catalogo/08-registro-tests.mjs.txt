{
  const c = new Checks();

  c.ok('test «corta el grifo al pasarse del límite por ventana»', has('corta el grifo al pasarse del límite por ventana'));
  c.note('código: server/http/routes/avisos.mjs:26-31 (503 AVISOS_UNAVAILABLE en producción sin almacén persistente) y :55-59 (lista o índice llenos → 429 «La lista está a reventar; inténtalo más tarde.» con Retry-After 3600); server/services/avisos.mjs:15-16 (MAX_POR_PRODUCTO 5000, MAX_PRODUCTOS_INDICE 200) y :75-82');
  c.note('cobertura de los tests: el de 429 comprueba el estado y Retry-After 3600 de la lista llena, pero no el texto del mensaje ni el Retry-After del índice lleno; el de producción comprueba solo code AVISOS_UNAVAILABLE, no el mensaje');
  await rec({ id: 'R-DROP-13', viewport: 'código y tests', checks: c });
