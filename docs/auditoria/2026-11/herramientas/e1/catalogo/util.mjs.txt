// Utilidades comunes de los recorridos de catálogo y drops.
import fs from 'node:fs';
import path from 'node:path';
import { record, shot, summarize, SCRATCH } from '../harness.mjs';

export const DRY = process.env.DRY === '1';
export const RES_DIR = path.join(SCRATCH, 'resultados-e1');
fs.mkdirSync(RES_DIR, { recursive: true });

// Acumula comprobaciones de una ficha y devuelve OK/FALLO con la evidencia.
export class Checks {
  constructor() {
    this.items = [];
    this.notes = [];
  }
  ok(name, cond, detail = '') {
    this.items.push({ name, ok: Boolean(cond), detail: String(detail) });
    return Boolean(cond);
  }
  eq(name, actual, expected) {
    const ok = JSON.stringify(actual) === JSON.stringify(expected);
    this.items.push({ name, ok, detail: ok ? JSON.stringify(actual) : `obtenido ${JSON.stringify(actual)} ≠ esperado ${JSON.stringify(expected)}` });
    return ok;
  }
  note(text) {
    this.notes.push(String(text));
  }
  get passed() {
    return this.items.every((i) => i.ok);
  }
  failures() {
    return this.items.filter((i) => !i.ok);
  }
  evidence() {
    const failed = this.failures().map((i) => `NO CUMPLE ${i.name}: ${i.detail}`);
    const passed = this.items.filter((i) => i.ok).map((i) => `${i.name}${i.detail ? `=${i.detail.length > 110 ? `${i.detail.slice(0, 110)}…` : i.detail}` : ''}`);
    return [...failed, ...this.notes, `cumple: ${passed.join('; ')}`].join(' | ');
  }
}

// Registra la ficha (o la imprime en modo DRY). En FALLO captura la pantalla.
export async function rec({ id, viewport, checks, issues, page, hallazgos = '', resultado }) {
  const res = resultado ?? (checks.passed ? 'OK' : 'FALLO');
  let evidencia = checks.evidence();
  if (issues) evidencia += ` | ${summarize(issues)}`;
  if (res === 'FALLO' && page) {
    try {
      const file = await shot(page, 'catalogo', `${id}-${viewport}`, { fullPage: false });
      evidencia += ` | captura ${file}`;
    } catch (e) {
      evidencia += ` | captura imposible: ${e.message}`;
    }
  }
  const line = { id, viewport, resultado: res, evidencia, hallazgos };
  if (DRY) {
    console.log(`\n### ${id} [${viewport}] → ${res}\n${evidencia}\nhallazgos: ${hallazgos}`);
  } else {
    record('catalogo', line);
    console.log(`${id} [${viewport}] → ${res}`);
    if (res !== 'OK') console.log('   ', evidencia.slice(0, 1500));
  }
  return res;
}

export function appendTsv(name, cols) {
  const clean = (s) => String(s).replace(/[\t\n\r]+/g, ' ').trim();
  const file = path.join(RES_DIR, name);
  if (DRY) {
    console.log(`[${name}]`, cols.map(clean).join(' | '));
    return;
  }
  fs.appendFileSync(file, cols.map(clean).join('\t') + '\n');
}

// Espera a que termine la carga del estado de la tienda (aviso demo o de error).
export async function waitStore(page, timeout = 8000) {
  await page.waitForSelector('.commerce-notice', { timeout }).catch(() => {});
  await page.waitForTimeout(150);
}

export const text = async (page, sel) =>
  page.evaluate((s) => document.querySelector(s)?.textContent?.trim() ?? null, sel);
export const texts = async (page, sel) =>
  page.evaluate((s) => [...document.querySelectorAll(s)].map((e) => e.textContent.trim()), sel);

// Descriptor legible del elemento con foco.
export async function activeDesc(page) {
  return page.evaluate(() => {
    const el = document.activeElement;
    if (!el || el === document.body) return 'body';
    const name = el.getAttribute('aria-label') || el.textContent?.trim().replace(/\s+/g, ' ').slice(0, 40) || el.getAttribute('placeholder') || el.getAttribute('name') || '';
    const href = el.getAttribute('href');
    return `${el.tagName.toLowerCase()}${el.type && el.tagName === 'INPUT' ? `[${el.type}]` : ''}«${name}»${href ? `→${href}` : ''}`;
  });
}

// Estilo de foco del elemento activo y si algún ancestro con overflow lo recorta.
export async function focusStyle(page) {
  return page.evaluate(() => {
    const el = document.activeElement;
    if (!el || el === document.body) return null;
    const cs = getComputedStyle(el);
    const ow = parseFloat(cs.outlineWidth) || 0;
    const oo = parseFloat(cs.outlineOffset) || 0;
    const outline = cs.outlineStyle !== 'none' && ow > 0 && !/rgba\(0, 0, 0, 0\)|transparent/.test(cs.outlineColor);
    const shadow = cs.boxShadow && cs.boxShadow !== 'none';
    const r = el.getBoundingClientRect();
    const ext = outline ? ow + oo : 0;
    const ring = { top: r.top - ext, left: r.left - ext, right: r.right + ext, bottom: r.bottom + ext };
    const clippedBy = [];
    for (let p = el.parentElement; p && p !== document.documentElement; p = p.parentElement) {
      const pcs = getComputedStyle(p);
      if (/(hidden|clip|auto|scroll)/.test(pcs.overflow + pcs.overflowX + pcs.overflowY)) {
        const pr = p.getBoundingClientRect();
        const sides = [];
        if (ring.top < pr.top - 0.5) sides.push('arriba');
        if (ring.left < pr.left - 0.5) sides.push('izquierda');
        if (ring.right > pr.right + 0.5) sides.push('derecha');
        if (ring.bottom > pr.bottom + 0.5) sides.push('abajo');
        if (sides.length) clippedBy.push(`${p.className.toString().split(' ')[0] || p.tagName}(${sides.join(',')})`);
      }
    }
    return {
      outline: outline ? `${cs.outlineStyle} ${cs.outlineWidth} ${cs.outlineColor} off ${cs.outlineOffset}` : 'ninguno',
      shadow: shadow ? cs.boxShadow.slice(0, 60) : 'ninguna',
      focusVisible: el.matches(':focus-visible'),
      clippedBy,
      rect: { x: Math.round(r.left), y: Math.round(r.top), w: Math.round(r.width), h: Math.round(r.height) },
    };
  });
}

// Compara en píxeles la banda alrededor del elemento con foco y sin foco.
// Devuelve cuántos píxeles cambian en cada lado de la banda exterior (8 px)
// y dentro del propio elemento.
export async function pixelFocusDiff(page, pad = 8) {
  const handle = await page.evaluateHandle(() => document.activeElement);
  const box = await handle.evaluate((el) => {
    el.scrollIntoView({ block: 'center', inline: 'nearest' });
    const r = el.getBoundingClientRect();
    return { x: r.left, y: r.top, w: r.width, h: r.height };
  });
  await page.waitForTimeout(350);
  const vp = page.viewportSize();
  const b2 = await handle.evaluate((el) => {
    const r = el.getBoundingClientRect();
    return { x: r.left, y: r.top, w: r.width, h: r.height };
  });
  const clip = {
    x: Math.max(0, Math.floor(b2.x - pad)),
    y: Math.max(0, Math.floor(b2.y - pad)),
  };
  clip.width = Math.min(vp.width - clip.x, Math.ceil(b2.w + pad * 2));
  clip.height = Math.min(vp.height - clip.y, Math.ceil(b2.h + pad * 2));
  if (clip.width <= 0 || clip.height <= 0) return null;
  const focused = (await page.screenshot({ clip })).toString('base64');
  await handle.evaluate((el) => el.blur());
  await page.waitForTimeout(350);
  const blurred = (await page.screenshot({ clip })).toString('base64');
  await handle.evaluate((el) => el.focus());
  await page.waitForTimeout(50);
  const inner = { x: b2.x - clip.x, y: b2.y - clip.y, w: b2.w, h: b2.h };
  const diff = await page.evaluate(
    async ({ a, b, inner: inn }) => {
      const load = (src) =>
        new Promise((res, rej) => {
          const im = new Image();
          im.onload = () => res(im);
          im.onerror = rej;
          im.src = `data:image/png;base64,${src}`;
        });
      const [ia, ib] = await Promise.all([load(a), load(b)]);
      const w = ia.naturalWidth;
      const h = ia.naturalHeight;
      const ctx = (im) => {
        const c = document.createElement('canvas');
        c.width = w;
        c.height = h;
        const x = c.getContext('2d');
        x.drawImage(im, 0, 0);
        return x.getImageData(0, 0, w, h).data;
      };
      const da = ctx(ia);
      const db = ctx(ib);
      const dpr = window.devicePixelRatio || 1;
      const ix0 = inn.x * dpr;
      const iy0 = inn.y * dpr;
      const ix1 = (inn.x + inn.w) * dpr;
      const iy1 = (inn.y + inn.h) * dpr;
      const out = { arriba: 0, abajo: 0, izquierda: 0, derecha: 0, dentro: 0 };
      for (let y = 0; y < h; y += 1) {
        for (let x = 0; x < w; x += 1) {
          const i = (y * w + x) * 4;
          const d = Math.abs(da[i] - db[i]) + Math.abs(da[i + 1] - db[i + 1]) + Math.abs(da[i + 2] - db[i + 2]);
          if (d < 60) continue;
          if (x >= ix0 && x < ix1 && y >= iy0 && y < iy1) out.dentro += 1;
          else if (y < iy0) out.arriba += 1;
          else if (y >= iy1) out.abajo += 1;
          else if (x < ix0) out.izquierda += 1;
          else out.derecha += 1;
        }
      }
      return out;
    },
    { a: focused, b: blurred, inner }
  );
  await handle.dispose();
  return diff;
}

// Recoge las imágenes que descarga una página (URL y bytes).
export function collectImages(page) {
  const list = [];
  page.on('response', async (r) => {
    try {
      if (r.request().resourceType() !== 'image') return;
      const url = new URL(r.url());
      if (!/^localhost$/.test(url.hostname) && !url.protocol.startsWith('data')) return;
      let bytes = Number(r.headers()['content-length'] || 0);
      try {
        const body = await r.body();
        bytes = body.length;
      } catch {
        /* sin cuerpo */
      }
      list.push({ url: url.pathname, bytes, status: r.status() });
    } catch {
      /* respuesta descartada */
    }
  });
  return list;
}

// Desplaza la página hasta el final en pasos para disparar la carga diferida.
export async function scrollThrough(page) {
  await page.evaluate(async () => {
    const step = Math.round(window.innerHeight * 0.8);
    for (let y = 0; y < document.documentElement.scrollHeight; y += step) {
      window.scrollTo(0, y);
      await new Promise((r) => setTimeout(r, 150));
    }
    window.scrollTo(0, document.documentElement.scrollHeight);
    await new Promise((r) => setTimeout(r, 400));
  });
}
