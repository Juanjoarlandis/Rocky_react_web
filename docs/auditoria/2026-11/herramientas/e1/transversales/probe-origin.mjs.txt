import { startInstance, launchBrowser, openPage, summarize } from '../harness.mjs';
const inst = await startInstance(3207, {});
const b = await launchBrowser();
try {
  const { page, issues, context } = await openPage(b, 'escritorio', {});
  const reqs = [];
  page.on('request', (r) => { const o = r.headers()['origin']; if (o) reqs.push(`${r.method()} ${r.url()} origin=${o}`); });
  await page.goto('http://127.0.0.1:3207/', { waitUntil: 'networkidle' });
  console.log('title', await page.title());
  console.log('links', await page.locator('a[href^="/product/"]').count());
  console.log('origin-bearing requests:', reqs.slice(0, 20));
  console.log(summarize(issues));
  await context.close();
} finally { await b.close(); inst.stop(); }
