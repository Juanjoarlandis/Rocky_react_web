async function stopGate() {
  if (!gate) return;
  const child = gate.child;
  logs.push(gate.log.join(''));
  const exited = new Promise((r) => child.once('exit', r));
  gate.stop();
  await Promise.race([exited, sleep(1500)]);
  if (child.exitCode === null && child.signalCode === null) {
    child.kill('SIGKILL');
    await Promise.race([exited, sleep(1500)]);
  }
  gate = null;
}
async function restartGate() {
  await stopGate();
  gate = await startInstance(PORT, ENV);
}

    page.getByRole('button', { name: 'Entrar' }).click(),
  ]);
  await page.waitForLoadState('domcontentloaded');
  return resp;
}

    const paths = [`/assets/${jsAsset}`, '/manifest.json', '/favicon.ico', '/robots.txt', '/products/rocky-racing.webp', '/music/barro.m4a', '/api/shopify/status', '/api/loquesea', '/products', '/menudrop', '/product/x', '/legal/privacidad', '/crew'];
    for (const p of paths) {
      const r = await raw('GET', `${O}${p}`, { headers: { Accept: '*/*' } });
      c.check(`${p} (Accept */*) → 404 «Not found.» no-store + CDN + X-Robots`, r.status === 404 && r.body === 'Not found.' && /text\/plain/.test(r.headers['content-type']) && r.headers['cache-control'] === 'no-store' && r.headers['cloudflare-cdn-cache-control'] === 'no-store' && r.headers['x-robots-tag'] === XROBOTS, brief(r));
    }
    const head = await raw('HEAD', `${O}/`);
    c.check('HEAD / → 404', head.status === 404, `${head.status} ${head.headers['content-type']}`);
    const root = await raw('GET', `${O}/`, { headers: { Accept: '*/*' } });
    c.check('GET / (Accept */*) → 200 puerta', root.status === 200 && root.body.includes(GATE_TITLE), `${root.status}`);
    const crewHtml = await raw('GET', `${O}/crew`, { headers: { Accept: 'text/html' } });
    c.check('GET /crew con Accept text/html → 200 puerta', crewHtml.status === 200 && crewHtml.body.includes(GATE_TITLE), `${crewHtml.status}`);
    const { page, context } = await openPage(browser, 'escritorio', {});
    await page.goto(`${O}/`, { waitUntil: 'networkidle' });
    const fs1 = await page.evaluate(() => fetch('/crew').then((r) => r.status));
    c.check('consola de la puerta: fetch(\'/crew\') → 404', fs1 === 404, String(fs1));
    const bar = await page.goto(`${O}/api/shopify/status`, { waitUntil: 'load' });
    c.check('/api/shopify/status en la barra de direcciones → 200 con la puerta', bar.status() === 200 && (await page.title()) === GATE_TITLE, `${bar.status()} «${await page.title()}»`);
    await context.close();
    const health = await raw('GET', `${O}/api/health`);
    c.check('/api/health desde loopback → 200 {"status":"ok"}', health.status === 200 && health.body === '{"status":"ok"}', brief(health));
    c.note('Producción (dato del 2026-11-08 con curl, Accept */*): GET / 200 puerta; /products, /menudrop, /product/x, /legal/privacidad 404 text/plain; HEAD / 404; /api/health 404 JSON. Coincide con lo observado aquí: curl no manda Accept text/html, así que solo / cuenta como documento.');
    cards.push(c);
  }
  {
    const c = card('R-GATE-08', 'api');
    const css = await raw('GET', `${O}/access-gate/style.css`);
    c.check('style.css → 200 text/css no-store', css.status === 200 && /text\/css/.test(css.headers['content-type']) && css.headers['cache-control'] === 'no-store', brief(css, ['content-type', 'cache-control']));
    const expected = { logo: ['logo512.png', /image\/png/], grafitero: ['grafitero-spray-', /image\/webp/], paquete: ['abrazando-paquete-', /image\/webp/], corredor: ['corriendo-bolsa-', /image\/webp/], 'font-display': ['luckiest-guy-latin-400-', /font\/woff2/], 'font-hand': ['fredoka-latin-300-700-', /font\/woff2/], 'font-body': ['archivo-latin-400-800-', /font\/woff2/] };
    const sizes = Object.fromEntries(
      [['logo512.png', fs.statSync(`${REPO}/dist/logo512.png`).size], ...fs.readdirSync(`${REPO}/dist/assets`).map((n) => [n, fs.statSync(`${REPO}/dist/assets/${n}`).size])]
    );
    let total = 0;
    for (const [name, [stem, type]] of Object.entries(expected)) {
      const r = await raw('GET', `${O}/access-gate/media/${name}`);
      total += r.bytes;
      const match = Object.entries(sizes).filter(([n, s]) => n.startsWith(stem) && s === r.bytes).map(([n]) => n);
      c.check(`media/${name} → 200 ${type.source} no-store (${match.join('/') || '¿?'})`, r.status === 200 && type.test(r.headers['content-type']) && r.headers['cache-control'] === 'no-store' && match.length > 0 && !match.some((m) => m.includes('-hd-')), `${r.status} ${r.headers['content-type']} ${r.bytes} B → ${match.join('/')}`);
    }
    c.check('peso total ≈ 166 kB', total > 150_000 && total < 180_000, `${Math.round(total / 1024)} KiB (${total} B)`);
    const bad = await raw('GET', `${O}/access-gate/media/app`);
    c.check('media/app → 404 «Not found.»', bad.status === 404 && bad.body === 'Not found.', brief(bad));
    // Recarga en el navegador: todo se vuelve a pedir a la red.
    const { page, context } = await openPage(browser, 'escritorio', {});
    await page.goto(`${O}/`, { waitUntil: 'networkidle' });
    const reqs = [];
    page.on('requestfinished', async (rq) => { if (rq.url().includes('/access-gate/')) { const rs = await rq.response(); reqs.push(`${rs.status()}${rs.fromServiceWorker() ? '(sw)' : ''} ${rq.url().split('/access-gate/')[1]}`); } });
    await page.reload({ waitUntil: 'networkidle' });
    c.check('al recargar se vuelven a pedir CSS y medios', reqs.length >= 5, reqs.join(', '));
    await context.close();
    c.note('Variante «sin dist/» no ejecutada: exigiría borrar o mover dist/ del repo.');
    cards.push(c);
  }
  {
    const c = card('R-SYS-02', 'api');
    let ip = '172.17.0.1';
    let h = await raw('GET', `http://${ip}:${PORT}/api/health`).catch(() => null);
    if (!h) { ip = 'IP-LOCAL-DEL-EQUIPO'; h = await raw('GET', `http://${ip}:${PORT}/api/health`); }
    c.check(`/api/health desde ${ip} (no loopback) → 404 JSON «Ruta no encontrada.» con no-store y CDN no-store`, h.status === 404 && h.body === '{"message":"Ruta no encontrada."}' && h.headers['cache-control'] === 'no-store' && h.headers['cloudflare-cdn-cache-control'] === 'no-store', brief(h));
    const o = await raw('GET', `http://${ip}:${PORT}/api/loquesea`);
    c.check('/api/loquesea desde fuera → 404 text/plain «Not found.» (puerta)', o.status === 404 && o.body === 'Not found.', brief(o));
    c.note('Simulación local: petición a una IP propia no loopback (req.socket.remoteAddress no es loopback). Producción (dato del 2026-11-08): /api/health → 404 {"message":"Ruta no encontrada."}, igual.');
    cards.push(c);
  }
  {
    const c = card('R-HOOK-07', 'api-puerta');
    const r = await raw('POST', `${O}/api/shopify/webhooks`, { headers: { 'Content-Type': 'application/json' }, body: '{}' });
    c.check('puerta activa y sin acceso: POST /api/shopify/webhooks → 404 text/plain «Not found.»', r.status === 404 && r.body === 'Not found.', brief(r));
    cards.push(c);
  }
  {
    const c = card('R-SYS-04', 'api-puerta');
    const unlock = await postGate(PW);
    const cookie = (unlock.headers['set-cookie'] || [])[0]?.split(';')[0];
    c.check('desbloqueo por API', unlock.status === 303 && Boolean(cookie), `${unlock.status}`);
    for (const p of ['/', '/index.html', '/crew', `/assets/${jsAsset}`, '/manifest.json', '/theme-init.js', '/products/rocky-racing.webp', '/music/barro.m4a']) {
      const r = await raw('GET', `${O}${p}`, { headers: { Cookie: cookie } });
      c.check(`${p} con acceso → private, no-store + CDN no-store + Vary: Cookie`, r.status === 200 && r.headers['cache-control'] === 'private, no-store' && r.headers['cloudflare-cdn-cache-control'] === 'no-store' && /Cookie/.test(r.headers.vary || ''), `${r.status} cc=${r.headers['cache-control']} cdn=${r.headers['cloudflare-cdn-cache-control']} vary=${r.headers.vary}`);
    }
    cards.push(c);
  }

  // ================= Pasada de teclado: puerta =================
  await restartGate();
  {
    const c = card('TECLADO-puerta', 'escritorio');
    const { page, issues, context } = await openPage(browser, 'escritorio', {});
    await page.goto(`${O}/`, { waitUntil: 'networkidle' });
    const st = () => page.evaluate(() => {
      const a = document.activeElement;
      const cs = getComputedStyle(a);
      return { tag: a.tagName, id: a.id, text: (a.textContent || '').trim(), fv: a.matches(':focus-visible'), outline: `${cs.outlineStyle} ${cs.outlineWidth} ${cs.outlineColor}`, shadow: cs.boxShadow, border: cs.borderColor };
    });
    const s0 = await st();
    c.check('foco inicial en el campo, visible (borde azul + halo)', s0.id === 'access-password' && s0.fv && /47, 111, 219/.test(s0.border) && /47, 111, 219/.test(s0.shadow), JSON.stringify(s0));
    await page.keyboard.press('Tab');
    const s1 = await st();
    c.check('Tab → «Entrar» con contorno visible', s1.tag === 'BUTTON' && s1.text === 'Entrar' && s1.fv && /solid 4px/.test(s1.outline), JSON.stringify(s1));
    await page.keyboard.press('Tab');
    const s2 = await st();
    c.note(`Tab desde «Entrar» → ${s2.tag}${s2.id ? '#' + s2.id : ''} (no hay más elementos enfocables)`);
    await page.keyboard.press('Shift+Tab');
    const s3 = await st();
    await page.keyboard.press('Shift+Tab');
    const s4 = await st();
    c.note(`Shift+Tab → ${s3.tag} «${s3.text || s3.id}», otra vez → ${s4.tag} ${s4.id}`);
    await page.locator('#access-password').focus();
    await page.keyboard.type('mala-por-teclado');
    const [resp] = await Promise.all([page.waitForResponse((r) => r.request().method() === 'POST'), page.keyboard.press('Enter')]);
    await page.waitForLoadState('domcontentloaded');
    const s5 = await st();
    const aria = await page.evaluate(() => ({ describedby: document.querySelector('#access-password').getAttribute('aria-describedby'), invalid: document.querySelector('#access-password').getAttribute('aria-invalid'), msg: document.querySelector('.gate-message').textContent.trim() }));
    c.check('Enter envía el formulario (401) y el foco vuelve al campo', resp.status() === 401 && s5.id === 'access-password', `${resp.status()} foco=${s5.id}`);
    c.note(`Tras el fallo: mensaje «${aria.msg}» en role=status presente desde la carga; campo sin aria-invalid (${aria.invalid}) ni aria-describedby (${aria.describedby})`);
    c.note(summarize(issues));
    await context.close();
    cards.push(c);
  }
} finally {
  await browser.close();
  await stopGate();
  fs.writeFileSync(new URL('./obs/gate-server.log', import.meta.url), logs.join('\n----\n'));
  saveObs('gate', cards);
}
