function evid(cards, max = 1900) {
  const parts = [];
  for (const c of cards) {
    const fails = c.checks.filter((x) => !x.ok);
    const oks = c.checks.filter((x) => x.ok && x.label !== 'nota');
    const notes = c.checks.filter((x) => x.label === 'nota');
    for (const f of fails) parts.push(`NO CUMPLE: ${f.label} → ${f.observed.slice(0, 260)}`);
    for (const o of oks) parts.push(`${o.label}: ${o.observed.slice(0, 140)}`);
    for (const n of notes) parts.push(`nota: ${n.observed.slice(0, 220)}`);
  }
  const s = parts.join(' ; ');
  return s.length > max ? `${s.slice(0, max)}…` : s;
}
