function record(id, check, ok, evidence, kind = ok ? 'OK' : 'FALLO') {
  results.push({ id, check, result: kind, evidence: String(evidence).slice(0, 600) });
  process.stdout.write(`${kind.padEnd(9)} ${id} ${check} — ${String(evidence).slice(0, 220)}\n`);
}

async function step(id, check, fn) {
  try {
    const out = await fn();
    if (out && typeof out === 'object' && 'ok' in out) record(id, check, out.ok, out.evidence, out.kind);
    else record(id, check, true, out ?? '');
  } catch (error) {
    record(id, check, false, `${error?.message || error}`.split('\n')[0]);
  }
}

      const s = await (await api.get(`${B.origin}/api/shopify/status`)).json();
      return { ok: Object.values(s.capabilities).every(Boolean) && s.mode === 'shopify', evidence: JSON.stringify(s) };
    });

    await step('B2', 'Login: /mi-crew → «Entrar en la Crew» → authorize (simulado) → callback → /mi-crew', async () => {
      await page.goto(`${B.origin}/mi-crew`);
      const callback = page.waitForResponse((r) => r.url().includes('/api/shopify/account/callback'), { timeout: 15_000 });
      await page.getByRole('link', { name: 'Entrar en la Crew' }).click();
      const cb = await callback;
      await page.waitForURL(`${B.origin}/mi-crew`, { timeout: 15_000 });
      await page.locator('h1.page-title', { hasText: 'Ana Prueba' }).waitFor({ timeout: 15_000 });
      await page.screenshot({ path: path.join(SHOTS, 'b2-mi-crew.png') });
      return { ok: cb.status() === 302 && routes.authorizations.length === 1, evidence: `callback → ${cb.status()} a ${cb.headers().location}; authorize simulados: ${routes.authorizations.length}; título «Ana Prueba» visible` };
    });

    await step('B3', 'GET /api/shopify/account → loggedIn: true', async () => {
      const body = await (await api.get(`${B.origin}/api/shopify/account`)).json();
      return { ok: body.loggedIn === true && body.customer?.id === 'gid://shopify/Customer/7035000001', evidence: JSON.stringify({ loggedIn: body.loggedIn, displayName: body.customer?.displayName, id: body.customer?.id }) };
    });

    await step('B4', 'Compra Crew: carrito → orders/paid firmado → XP y tickets acreditados', async () => {
      const before = (await (await api.get(`${B.origin}/api/shopify/account/crew`)).json()).profile;
      for (const [alias, quantity] of [['rocky-racing/M', 2], ['dwag/L', 1]]) {
        const res = await api.post(`${B.origin}/api/shopify/cart/lines`, { headers: { Origin: B.origin }, data: { variantId: variantId(alias), quantity, operationId: opId() } });
        if (res.status() !== 200) throw new Error(`añadir ${alias} → ${res.status()}`);
      }
      const cart = (await (await api.get(`${B.origin}/api/shopify/cart`)).json()).cart;
      const payload = orderPaidFromCart(cart, { customer: 'ana' });
      const hook = await sendShopifyWebhook({ baseUrl: B.httpOrigin, payload });
      const after = (await (await api.get(`${B.origin}/api/shopify/account/crew`)).json()).profile;
      const ok = hook.status === 200 && before.xp === 0 && after.xp === 110 && after.ticketBalance === 11 && after.level.name === 'Del Barrio';
      return { ok, evidence: `carrito ${cart.cost.totalAmount.amount} EUR; webhook ${hook.status} ${JSON.stringify(hook.body)}; XP ${before.xp}→${after.xp}, tickets ${after.ticketBalance}, nivel «${after.level.name}», compras ${after.purchases.length}` };
    });

    await step('B5', 'Rocky IA con perfil Crew verificado (sesión iniciada)', async () => {
      const res = await api.post(`${B.origin}/api/chat`, { headers: { Origin: B.origin }, data: { message: '¿Qué nivel tengo en la crew?' } });
      const body = await res.json();
      return { ok: res.status() === 200 && body.message.includes('Del Barrio') && body.message.includes('110 XP'), evidence: `${res.status()} «${body.message}»` };
    });

    await step('B6', 'Logout desde Mi Crew → end_session simulado → vuelta a la web → loggedIn: false', async () => {
      await page.goto(`${B.origin}/mi-crew`);
      await page.locator('.crew-logout').click();
      await page.waitForURL(`${B.origin}/`, { timeout: 15_000 });
      const body = await (await api.get(`${B.origin}/api/shopify/account`)).json();
      const logout = routes.logouts[0];
      return { ok: body.loggedIn === false && logout?.hasIdTokenHint === true, evidence: `end_session con id_token_hint=${logout?.hasIdTokenHint}, vuelta a ${logout?.back}; account → ${JSON.stringify(body)}` };
    });
    record('B6b', 'Rutas externas bloqueadas / errores en el recorrido HTTPS', true, `bloqueadas: ${[...new Set(routes.blocked)].join(' | ') || 'ninguna'}; ${JSON.stringify(issues).slice(0, 400)}`, 'OBSERVADO');
    await context.close();
  }

  await step('B7', 'ID token con firma ajena: el callback lo rechaza (401) y no hay sesión', async () => {
    B.setControl({ customerAccount: { idToken: 'badSignature' } });
    const { context, page } = await openE2Page(browser, B);
    const callback = page.waitForResponse((r) => r.url().includes('/api/shopify/account/callback'), { timeout: 15_000 });
    await page.goto(`${B.origin}/api/shopify/account/login?returnPath=%2Fmi-crew`);
    const cb = await callback;
    const cbBody = await cb.text();
    const account = await (await context.request.get(`${B.origin}/api/shopify/account`)).json();
    B.resetControl();
    await context.close();
    return { ok: cb.status() === 401 && account.loggedIn === false, evidence: `callback → ${cb.status()} ${cbBody.slice(0, 120)}; account → ${JSON.stringify(account)}` };
  });

  await step('B8', 'El cliente cancela en Shopify (error=access_denied): respuesta del callback', async () => {
    const { context, page } = await openE2Page(browser, B, { decision: 'deny' });
    const callback = page.waitForResponse((r) => r.url().includes('/api/shopify/account/callback'), { timeout: 15_000 });
    await page.goto(`${B.origin}/api/shopify/account/login?returnPath=%2Fmi-crew`);
    const cb = await callback;
    const text = await cb.text();
    await context.close();
    return { ok: true, kind: 'OBSERVADO', evidence: `callback → ${cb.status()} ${text.slice(0, 160)}` };
  });

  await step('B9', 'Sin llamadas no previstas en la instancia HTTPS', async () => {
    const unexpected = B.unexpectedCalls();
    const ops = [...new Set(B.calls().map((c) => `${c.service}:${c.op}`))];
    return { ok: unexpected.length === 0, evidence: `operaciones: ${ops.join(', ')}; no previstas: ${unexpected.length}` };
  });
  await B.stop();

  // ------------------------------------- HTTPS con NODE_ENV=production (opcional)
  await step('E1', 'HTTPS con NODE_ENV=production y TRUST_PROXY_HOPS=1: login y cookie __Host-', async () => {
    const P = await startE2Instance(3409, { ...EXTRA, NODE_ENV: 'production', TRUST_PROXY_HOPS: '1' }, { https: true, httpsPort: 3410 });
    instances.push(P);
    const { context, page } = await openE2Page(browser, P);
    await page.goto(`${P.origin}/api/shopify/account/login?returnPath=%2Fmi-crew`);
    await page.waitForURL(`${P.origin}/mi-crew`, { timeout: 15_000 });
    const account = await (await context.request.get(`${P.origin}/api/shopify/account`)).json();
    const cookies = (await context.cookies(P.origin)).map((c) => `${c.name}${c.secure ? ' (Secure)' : ''}`);
    const hsts = (await context.request.get(`${P.origin}/api/health`)).headers()['strict-transport-security'];
    await context.close();
    await P.stop();
    return { ok: account.loggedIn === true && cookies.some((c) => c.startsWith('__Host-rocky_session')), evidence: `loggedIn=${account.loggedIn}; cookies: ${cookies.join(', ')}; HSTS: ${hsts}` };
  });

  await browser.close();
}

try {
  await main();
} catch (error) {
  record('X', 'Error no controlado en la verificación', false, error?.stack || error);
} finally {
  for (const instance of instances) await instance.stop().catch(() => {});
  fs.writeFileSync(OUT, `${JSON.stringify({ at: new Date().toISOString(), results }, null, 2)}\n`);
  const failed = results.filter((r) => r.result === 'FALLO').length;
  process.stdout.write(`\n${results.length} comprobaciones, ${failed} fallos. Resultado en ${OUT}\n`);
  process.exitCode = failed ? 1 : 0;
}
