// Las imágenes del catálogo apuntan al CDN de Shopify (lo único que admite la
// CSP); en Playwright, installE2Routes las sirve desde dist/products/.
export const CDN_PREFIX = 'https://cdn.shopify.com/s/files/1/0903/5003/5035/files/';
const cdn = (file) => `${CDN_PREFIX}${file}?v=1730000000`;

const DROP_4 = { handle: 'rocky-drop-4', title: 'ROCKY DROP 4' };
const DROP_3 = { handle: 'rocky-drop-3', title: 'ROCKY DROP 3' };

const pad = (n) => String(n).padStart(2, '0');
const productGid = (n) => `gid://shopify/Product/80350000${pad(n)}`;
const variantGid = (n, k) => `gid://shopify/ProductVariant/90350${pad(n)}${pad(k)}`;

// stock: unidades por talla. 0 = agotada. Las variantes no tienen imagen propia
// (como en la tienda real cuando no se asigna), así que la ficha usa la del producto.
function product(n, { handle, title, description, price, image = null, drop = DROP_4, stock }) {
  const sizes = Object.keys(stock);
  const isDefault = sizes.length === 1 && sizes[0] === 'Default Title';
  return {
    n,
    id: productGid(n),
    handle,
    title,
    description,
    image: image ? { url: cdn(image), altText: title } : null,
    collections: drop ? [drop] : [],
    variants: sizes.map((size, index) => ({
      id: variantGid(n, index + 1),
      alias: `${handle}/${size}`,
      title: size,
      selectedOptions: [{ name: isDefault ? 'Title' : 'Talla', value: size }],
      price,
      inventory: stock[size],
    })),
  };
}

export const VARIANTS = new Map(
  PRODUCTS.flatMap((p) => p.variants.map((v) => [v.id, { ...v, product: p }]))
);
export const VARIANT_BY_ALIAS = new Map([...VARIANTS.values()].map((v) => [v.alias, v]));

// Acepta un GID de variante o un alias «handle/talla» (p. ej. rockydz-boyz/L).
export function findVariant(idOrAlias) {
  return VARIANTS.get(idOrAlias) || VARIANT_BY_ALIAS.get(idOrAlias) || null;
}
